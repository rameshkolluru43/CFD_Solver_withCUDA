16 double Pressure_From_U(const V_D &U)
17 {
18 const double rho = U[0];
19 if (rho <= 1e-14 || !std::isfinite(rho))
20 return -1.0;
21 const double u = U[1] / rho;
22 const double v = U[2] / rho;
23 const double p = gamma_M_1 * (U[3] - 0.5 * rho * (u * u + v * v));
24 return p;
25 }

28 void WenoPrimitiveFromU(const V_D &U_Vect, V_D &G_Primitive)
29 {
30 Vector_Reset(G_Primitive);
31 const double Rho = U_Vect[0];
32 CFD_RequireFinitePositive(Rho, "WenoPrimitiveFromU density");
33 const double inv_Density = 1.0 / Rho;
34 double v1 = U_Vect[1] * inv_Density;
35 double v2 = U_Vect[2] * inv_Density;
36 if (fabs(v1) <= 1e-10)
37 v1 = 0.0;
38 if (fabs(v2) <= 1e-10)
39 v2 = 0.0;
40 const double vmag = (v1 * v1 + v2 * v2);
41 const double Pressure = gamma_M_1 * (U_Vect[3] - 0.5 * Rho * vmag);
42 CFD_RequireFinitePositive(Pressure, "WenoPrimitiveFromU pressure");
43 double Temperature;
44 if (Non_Dimensional_Form)
45 Temperature = (Pressure / (Rho * R_ref));
46 else
47 Temperature = (Pressure * inv_Density / R_GC);
48 const double C = sqrt(gamma * Pressure * inv_Density);
49 CFD_RequireFinitePositive(C, "WenoPrimitiveFromU sound speed");
50 const double Mloc = sqrt(vmag) / C;
51
52 G_Primitive[0] = Rho;
53 G_Primitive[1] = v1;
54 G_Primitive[2] = v2;
55 G_Primitive[3] = Temperature;
56 G_Primitive[4] = Pressure;
57 G_Primitive[5] = C;
58 G_Primitive[6] = U_Vect[3];
59 G_Primitive[7] = Mloc;
60 G_Primitive[8] = 0.0;
61 G_Primitive[9] = 0.0;
62 G_Primitive[10] = Pressure * pow((1.0 + 0.5 * gamma_M_1 * Mloc * Mloc), gamma / gamma_M_1);
63
64 if (std::isnan(C))
65 {
66 throw runtime_error("WenoPrimitiveFromU: non-finite sound speed");
67 }
68 }

71{
72 // Check for NaN or infinite values in input
73 if (!std::isfinite(a) || !std::isfinite(b) || !std::isfinite(c) || !std::isfinite(d) || !std::isfinite(e))
74 {
75 std::cout << "Warning: Non-finite values in WENO reconstruction input" << std::endl;
76 U = c; // Fall back to central value
77 return;
78 }
79
80 double d0 = 0.0, d1 = 0.0, d2 = 0.0, b0 = 0.0, b1 = 0.0, b2 = 0.0, a0 = 0.0, a1 = 0.0, a2 = 0.0;

83 switch (shift)
84 {
85 case 0: // for Left Values
86 d0 = 3.0 / 10.0;
87 d1 = 6.0 / 10.0;
88 d2 = 1.0 / 10.0;
89
90 v2 = (2.0 * a - 7.0 * b + 11.0 * c) / 6.0;
91 v1 = (-1.0 * b + 5.0 * c + 2.0 * d) / 6.0;
92 v0 = (2.0 * c + 5.0 * d - 1.0 * e) / 6.0;
93
94 break;
95 case 1: // for Right Values
96 d0 = 1.0 / 10.0;
97 d1 = 6.0 / 10.0;
98 d2 = 3.0 / 10.0;
99
100 v2 = (-1.0 * a + 5.0 * b + 2.0 * c) / 6.0;
101 v1 = (2.0 * b + 5.0 * c - 1.0 * d) / 6.0;
102 v0 = (11.0 * c - 7.0 * d + 2.0 * e) / 6.0;
103
104 break;
105 }
106
107 {
108 const double t2 = (a - 2.0 * b + c);
109 const double t2b = (a - 4.0 * b + 3.0 * c);
110 b2 = (13.0 / 12.0) * (t2 * t2) + (1.0 / 4.0) * (t2b * t2b);
111 }
112 {
113 const double t1 = (b - 2.0 * c + d);
114 const double t1b = (b - d);
115 b1 = (13.0 / 12.0) * (t1 * t1) + (1.0 / 4.0) * (t1b * t1b);
116 }
117 {
118 const double t0 = (c - 2.0 * d + e);
119 const double t0b = (3.0 * c - 4.0 * d + e);
120 b0 = (13.0 / 12.0) * (t0 * t0) + (1.0 / 4.0) * (t0b * t0b);
121 }
122
123 {
124 const double s0 = epsilon + b0;
125 const double s1 = epsilon + b1;
126 const double s2 = epsilon + b2;
127 a0 = d0 / (s0 * s0);
128 a1 = d1 / (s1 * s1);
129 a2 = d2 / (s2 * s2);
130 }
131
132 sum = a0 + a1 + a2;
133
134 // Check for degenerate case where sum is too small
135 if (sum < epsilon)
136 {
137 // Fall back to simple average of the three candidate values
138 U = (v0 + v1 + v2) / 3.0;
139 }
140 else
141 {
142 w0 = a0 / sum;
143 w1 = a1 / sum;
144 w2 = a2 / sum;
145 U = w0 * v0 + w1 * v1 + w2 * v2;
146 }
147
148 // Final validation
149 if (!std::isfinite(U))
150 {
151 std::cout << "Warning: Non-finite result in WENO reconstruction, using central value" << std::endl;
152 U = c;
153 }

180 double denom = sqrt_dL + sqrt_dR;

182 if (denom < 1e-14)

185 u_RL = 0.5 * (uL + uR);
186 v_RL = 0.5 * (vL + vR);
187 a_RL = 0.5 * (aL + aR);

218 L[4] = 2.0 * t2 * u_RL;

222 L[5] = ny;
223 L[9] = -nx;

227 L[10] = -t2 * v_RL + a_RL * t1 * ny;
228 L[14] = t2;
229 L[3] = t2 * ek + a_RL * vn * t1;
230 L[7] = -t2 * u_RL - a_RL * t1 * nx;
231 L[11] = -t2 * v_RL - a_RL * t1 * ny;
232 L[15] = t2;
233
234 // Print(L);

236 {
237 if (isnan(L[i]))

242

244 IL[0] = 1.0;
245 IL[4] = 0.0;
246 IL[8] = 1.0;
247 IL[12] = 1.0;
248 IL[1] = u_RL;
249 IL[5] = ny;

251 IL[13] = u_RL - a_RL * nx;
252 IL[2] = v_RL;
253 IL[6] = -nx;
254 IL[10] = v_RL + a_RL * ny;
255 IL[14] = v_RL - a_RL * ny;
256 // IL[1] = vn; IL[5] = ny; IL[9] = vn + a_RL*nx; IL[13] = vn - a_RL*nx;
257 // IL[2] = vn; IL[6] = -nx; IL[10] = vn + a_RL*ny; IL[14] = vn - a_RL*ny;
258 IL[3] = ek;
259 IL[7] = u_RL * ny - v_RL * nx;
260 IL[11] = h + a_RL * vn;
261 IL[15] = h - a_RL * vn;
262

513void Calculate_Face_WENO_Flux(int Cell_No, int N_Cell_No, const int &Face_No, bool Is_Wall_Face, V_D &out_avg, V_D &out_diss)
514{

516 V_D d_U(4, 0.0), d_F(4, 0.0), Mod_Alpha(4, 0.0);
517 V_D ricca_diss(4, 0.0), llf_diss(4, 0.0);
518 V_D primL(11, 0.0), primR(11, 0.0);
519
520 double Rho_L = 0.0, P_L = 0.0, u_L = 0.0, v_L = 0.0, Vdotn_L = 0.0, C_L = 0.0, Vmag_L = 0.0;
521 double Rho_R = 0.0, P_R = 0.0, u_R = 0.0, v_R = 0.0, Vdotn_R = 0.0, C_R = 0.0, Vmag_R = 0.0;
522 double nx = 0.0, ny = 0.0, dl = 0.0;
523
524 const int index = Face_No * 2;
525
526 /* Wall faces: WENO5 needs a 5-point stencil; host ghosts collapse to one cell
527 and frequently produce negative pressure at M=6 (e.g. cell 5504). Use
528 robust 1O interior/ghost conservatives from the BC-filled neighbour. */
529 if (Is_Wall_Face || N_Cell_No < 0 || N_Cell_No >= static_cast<int>(U_Cells.size()))
530 {
531 U_L = U_Cells[Cell_No];
532 if (N_Cell_No >= 0 && N_Cell_No < static_cast<int>(U_Cells.size()))
533 U_R = U_Cells[N_Cell_No];
534 else
535 U_R = U_L;
536 }
537 else
538 {
539 WENO_Reconstruction_X(Cell_No, Face_No, U_L, U_R);
540
541 if (Face_No == 0 || Face_No == 1)
542 std::swap(U_L, U_R);
543
544 const double pL = Pressure_From_U(U_L);
545 const double pR = Pressure_From_U(U_R);
546 if (pL < 1e-12 || pR < 1e-12 || !std::isfinite(pL) || !std::isfinite(pR))
547 {
548 U_L = U_Cells[Cell_No];
549 U_R = U_Cells[N_Cell_No];
550 }
551 }
552
553 WenoPrimitiveFromU(U_L, primL);
554 Rho_L = primL[0];
555 P_L = primL[4];
556 u_L = primL[1];
557 v_L = primL[2];
558 C_L = primL[5];
559 WenoPrimitiveFromU(U_R, primR);
