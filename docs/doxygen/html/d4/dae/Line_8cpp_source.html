51void Line::generate(Point &sp, Point &ep, int &n, bool &both_sides)
52{
53 double beta_v = 1.01, alpha_v, a, b, y_val, eta_v;
54 no_of_points = n;
55 point_list.clear();
56 point_list.reserve(n);
57
58 ds_x = ep.Get_x() - sp.Get_x();
59 ds_y = ep.Get_y() - sp.Get_y();
60 ds_z = ep.Get_z() - sp.Get_z();
61 Length = sqrt(ds_x * ds_x + ds_y * ds_y + ds_z * ds_z);
62
63 del_s = Length / (n - 1);
64 vector<double> s_dist(n, 0.0);
65
66 for (int j = 0; j < n; j++)
67 {
68 eta_v = static_cast<double>(j) / (n - 1);
69
70 if (both_sides)
71 {
72 alpha_v = 0.5;
73 a = (eta_v - alpha_v) / (1.0 - alpha_v);
74 b = (beta_v + 1.0) / (beta_v - 1.0);
75 y_val = Length * (((2 * alpha_v + beta_v) * pow(b, a) + 2 * alpha_v - beta_v)
76 / ((2 * alpha_v + 1.0) * (pow(b, a) + 1.0)));
77 }

79 {

81 a = 1.0 - eta_v;
82 b = (beta_v + 1.0) / (beta_v - 1.0);
83 y_val = Length * ((beta_v + 1.0) - (beta_v - 1.0) * pow(b, a))
84 / (pow(b, a) + 1.0);
85 }
86 s_dist[j] = y_val;
87 }

89 distribute_points(sp, s_dist);

148/* ======================================================================
149 Boundary layer distribution using geometric growth.
150 first_cell_height: physical height of the first cell
151 growth_rate: ratio of successive cell heights (typically 1.1 - 1.3)
152
153 The distribution grows geometrically from the start point up to the
154 point where the total length is reached. If the geometric series
155 doesn't reach the full length, the remaining points are uniformly
156 distributed.
157 ====================================================================== */
158void Line::generate_boundary_layer(Point &sp, Point &ep, int &n,
159 double first_cell_height,
160 double growth_rate)
161{
162 Point Temp_point;
163 no_of_points = n;
164 point_list.clear();
165 point_list.reserve(n);
166
167 ds_x = ep.Get_x() - sp.Get_x();
168 ds_y = ep.Get_y() - sp.Get_y();
169 ds_z = ep.Get_z() - sp.Get_z();
170 Length = sqrt(ds_x * ds_x + ds_y * ds_y + ds_z * ds_z);
171
172 vector<double> s_dist(n, 0.0);
173 s_dist[0] = 0.0;
174
175 if (fabs(growth_rate - 1.0) < 1e-10)
176 {
177 double ds = Length / (n - 1);
178 for (int i = 1; i < n; i++)
179 s_dist[i] = i * ds;
180 }
181 else
182 {
183 double dh = first_cell_height;
184 double total_geometric = first_cell_height * (pow(growth_rate, n - 1) - 1.0)
185 / (growth_rate - 1.0);
186
187 if (total_geometric <= Length)
188 {
189 double scale = Length / total_geometric;
190 dh = first_cell_height * scale;
191 s_dist[1] = dh;
192 for (int i = 2; i < n; i++)
193 {
194 dh *= growth_rate;
195 s_dist[i] = s_dist[i - 1] + dh;
196 }
197 }
198 else
199 {
200 dh = first_cell_height;
201 for (int i = 1; i < n; i++)
202 {
203 s_dist[i] = s_dist[i - 1] + dh;
204 if (s_dist[i] >= Length)
205 {
206 double uniform_ds = (Length - s_dist[i - 1]) / (n - i);
207 for (int k = i; k < n; k++)
208 s_dist[k] = s_dist[i - 1] + (k - i + 1) * uniform_ds;
209 break;
210 }
211 dh *= growth_rate;
212 }
213 double scale = Length / s_dist[n - 1];
214 for (int i = 1; i < n; i++)
215 s_dist[i] *= scale;
216 }
217 }
218
219 distribute_points(sp, s_dist);
220
221 cout << "Created Line (boundary layer): " << point_list.size() << " points\n"
222 << " First cell height: " << first_cell_height << "\n"
223 << " Growth rate: " << growth_rate << "\n"
224 << " Actual first dh: " << s_dist[1] << "\n"
225 << " Actual last dh: " << s_dist[n - 1] - s_dist[n - 2] << endl;
226}

228/* ======================================================================
229 Hyperbolic tangent stretching.
230 stretching_factor: controls clustering intensity.
231 delta > 0: cluster at both ends
232 Larger delta = more uniform; smaller delta = more clustering.
233 Based on: s(eta) = 1 + tanh(delta*(eta - 0.5)) / tanh(delta/2)
234 ====================================================================== */
235void Line::generate_tanh(Point &sp, Point &ep, int &n,
236 double stretching_factor)
237{
238 Point Temp_point;
239 no_of_points = n;
240 point_list.clear();
241 point_list.reserve(n);
242
243 ds_x = ep.Get_x() - sp.Get_x();
244 ds_y = ep.Get_y() - sp.Get_y();
245 ds_z = ep.Get_z() - sp.Get_z();
246 Length = sqrt(ds_x * ds_x + ds_y * ds_y + ds_z * ds_z);
247
248 vector<double> s_dist(n, 0.0);
249 double delta = stretching_factor;
250 double tanh_half = tanh(delta * 0.5);
251
252 for (int j = 0; j < n; j++)
253 {
254 double eta = static_cast<double>(j) / (n - 1);
255 double s_norm = 0.5 * (1.0 + tanh(delta * (eta - 0.5)) / tanh_half);
256 s_dist[j] = s_norm * Length;
257 }
258
259 distribute_points(sp, s_dist);
260
261 cout << "Created Line (tanh): " << point_list.size() << " points\n"
262 << " Stretching factor: " << delta << "\n"
263 << " First spacing: " << s_dist[1] << "\n"
264 << " Mid spacing: " << s_dist[n / 2 + 1] - s_dist[n / 2] << "\n"
265 << " Last spacing: " << s_dist[n - 1] - s_dist[n - 2] << endl;
266}
