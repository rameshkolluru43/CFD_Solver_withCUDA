41__device__ void condition_for_movers(double d_U, double d_F, double L_Max, double L_Min, double &Alpha)
42{
43 const double epsilon = 1e-10;
44 if (fabs(d_F) < epsilon && fabs(d_U) > epsilon)
45 Alpha = 0.0;
46 else if (fabs(d_F) < epsilon && fabs(d_U) < epsilon)
47 Alpha = L_Min;
48 else if (fabs(d_F) > epsilon && fabs(d_U) > epsilon)
49 {
50 Alpha = fabs(d_F / d_U);
51 if (Alpha >= L_Max)
52 Alpha = d_sign(Alpha) * L_Max;
53 else if (Alpha <= L_Min)
54 Alpha = d_sign(Alpha) * L_Min;
55 }
56 else
57 Alpha = L_Min;
58}

127__device__ void dissipative_movers(
128 int cell, int neigh,
129 const double *d_prim, const double *d_U,
130 double nx, double ny, double dl,
131 int dissipation_type, bool is_movers_1, bool entropy_fix_on, double gamma,
132 double out_diss[kNv])
133{
134 const int Lp = cell * 6;
135 const int Rp = neigh * 6;
136 const double rho_L = d_prim[Lp + 0];
137 const double u_L = d_prim[Lp + 1];
138 const double v_L = d_prim[Lp + 2];
139 const double P_L = d_prim[Lp + 4];
140 const double C_L = d_prim[Lp + 5];
141
142 const double rho_R = d_prim[Rp + 0];
143 const double u_R = d_prim[Rp + 1];
144 const double v_R = d_prim[Rp + 2];
145 const double P_R = d_prim[Rp + 4];
146 const double C_R = d_prim[Rp + 5];
147
148 const int Lc = cell * kNv;
149 const int Rc = neigh * kNv;
150 const double d_Uv[kNv] = {
151 d_U[Rc + 0] - d_U[Lc + 0],
152 d_U[Rc + 1] - d_U[Lc + 1],
153 d_U[Rc + 2] - d_U[Lc + 2],
154 d_U[Rc + 3] - d_U[Lc + 3]};
155
156 const double Vdotn_L = u_L * nx + v_L * ny;
157 const double Vdotn_R = u_R * nx + v_R * ny;
158 const double Vmag_L = 0.5 * (u_L * u_L + v_L * v_L);
159 const double Vmag_R = 0.5 * (u_R * u_R + v_R * v_R);
160
161 const double d_F[kNv] = {
162 (rho_R * Vdotn_R - rho_L * Vdotn_L) * dl,
163 (rho_R * u_R * Vdotn_R + P_R * nx - rho_L * u_L * Vdotn_L + P_L * nx) * dl,
164 (rho_R * v_R * Vdotn_R + P_R * ny - rho_L * v_L * Vdotn_L + P_L * ny) * dl,
165 ((((P_R / (gamma - 1.0)) + rho_R * Vmag_R) + P_R) * Vdotn_R -
166 (((P_L / (gamma - 1.0)) + rho_L * Vmag_L) + P_L) * Vdotn_L) * dl};
167
168 double S[6];
169 S[0] = fabs(Vdotn_L - C_L) * dl;
170 S[1] = fabs(Vdotn_L + C_L) * dl;
171 S[2] = fabs(Vdotn_L) * dl;
172 S[3] = fabs(Vdotn_R - C_R) * dl;
173 S[4] = fabs(Vdotn_R + C_R) * dl;
174 S[5] = fabs(Vdotn_R) * dl;
175
176 double Max1, Max2, Min1, Min2, Lambda_Max, Lambda_Min;
177 d_max3(S[0], S[1], S[2], Max1);
178 d_max3(S[3], S[4], S[5], Max2);
179 Lambda_Max = fmax(Max1, Max2);
180 d_min3(S[0], S[1], S[2], Min1);
181 d_min3(S[3], S[4], S[5], Min2);
182 Lambda_Min = fmax(Min1, Min2);
183
184 double alpha[kNv];
185 condition_for_movers(d_Uv[3], d_F[3], Lambda_Max, Lambda_Min, alpha[3]);
186 alpha[0] = alpha[3];
187 alpha[1] = alpha[3];
188 alpha[2] = alpha[3];
189
190 if (!is_movers_1)
191 {
192 condition_for_movers(d_Uv[0], d_F[0], Lambda_Max, Lambda_Min, alpha[0]);
193 condition_for_movers(d_Uv[1], d_F[1], Lambda_Max, Lambda_Min, alpha[1]);
194 condition_for_movers(d_Uv[2], d_F[2], Lambda_Max, Lambda_Min, alpha[2]);
195 }
196
197 if (entropy_fix_on)
198 {
199 entropy_fix(alpha[0], Lambda_Max);
200 entropy_fix(alpha[1], Lambda_Max);
201 entropy_fix(alpha[2], Lambda_Max);
202 entropy_fix(alpha[3], Lambda_Max);
203 }
204
205 for (int i = 0; i < kNv; ++i)
206 out_diss[i] = 0.5 * alpha[i] * d_Uv[i];
207}

212 double nx, double ny, double dl, double gamma,
213 double out_diss[kNv])
214{
215 const int Lp = cell * 6;
216 const int Rp = neigh * 6;
217 const double rho_L = d_prim[Lp + 0];
218 const double u_L = d_prim[Lp + 1];
219 const double v_L = d_prim[Lp + 2];
220 const double P_L = d_prim[Lp + 4];
221
222 const double rho_R = d_prim[Rp + 0];
223 const double u_R = d_prim[Rp + 1];
224 const double v_R = d_prim[Rp + 2];
225 const double P_R = d_prim[Rp + 4];

227 const double Vdotn_L = u_L * nx + v_L * ny;
228 const double Vdotn_R = u_R * nx + v_R * ny;

230 const double Vmag_R = 0.5 * (u_R * u_R + v_R * v_R);
231
232 double d_Uv[kNv] = {
233 rho_R - rho_L,
234 rho_R * u_R - rho_L * u_L,
235 rho_R * v_R - rho_L * v_L,
236 ((P_R / (gamma - 1.0)) + 0.5 * rho_R * (u_R * u_R + v_R * v_R)) -
237 ((P_L / (gamma - 1.0)) + 0.5 * rho_L * (u_L * u_L + v_L * v_L))};
238
239 const double d_F[kNv] = {
240 (rho_R * Vdotn_R - rho_L * Vdotn_L) * dl,
241 (rho_R * u_R * Vdotn_R + P_R * nx - rho_L * u_L * Vdotn_L + P_L * nx) * dl,
242 (rho_R * v_R * Vdotn_R + P_R * ny - rho_L * v_L * Vdotn_L + P_L * ny) * dl,
243 ((((P_R / (gamma - 1.0)) + rho_R * Vmag_R) + P_R) * Vdotn_R -
244 (((P_L / (gamma - 1.0)) + rho_L * Vmag_L) + P_L) * Vdotn_L) * dl};
245
246 double dP = fabs(P_L - P_R);
247 const double P_I = 0.5 * (P_L + P_R);
248 const double Rho_I = 0.5 * (rho_L + rho_R);
249
250 for (int i = 0; i < kNv; ++i)
251 {
252 double alpha = 0.0;
253 condition_for_ricca(d_Uv[i], d_F[i], Vdotn_L, Vdotn_R, dP, Rho_I, P_I, alpha, gamma);
254 out_diss[i] = 0.5 * alpha * dl * d_Uv[i];
255 }
256}

258__device__ void dissipative_movers_nwsc(
259 int cell, int neigh,
260 const double *d_prim, const double *d_U,
261 double nx, double ny, double dl,
262 double out_diss[kNv])
263{
264 const int Lp = cell * 6;
265 const int Rp = neigh * 6;
266 const double rho_L = d_prim[Lp + 0];
267 const double u_L = d_prim[Lp + 1];
268 const double v_L = d_prim[Lp + 2];
269 const double P_L = d_prim[Lp + 4];
270
271 const double rho_R = d_prim[Rp + 0];
272 const double u_R = d_prim[Rp + 1];
273 const double v_R = d_prim[Rp + 2];
274 const double P_R = d_prim[Rp + 4];
275
276 const int Lc = cell * kNv;
277 const int Rc = neigh * kNv;
278 const double d_Uv[kNv] = {
279 d_U[Rc + 0] - d_U[Lc + 0],
280 d_U[Rc + 1] - d_U[Lc + 1],
281 d_U[Rc + 2] - d_U[Lc + 2],
282 d_U[Rc + 3] - d_U[Lc + 3]};
283
284 const double Vdotn_L = u_L * nx + v_L * ny;
285 const double Vdotn_R = u_R * nx + v_R * ny;
286 const double Vmag_L = 0.5 * (u_L * u_L + v_L * v_L);
287 const double Vmag_R = 0.5 * (u_R * u_R + v_R * v_R);
288
289 const double d_F[kNv] = {
290 (rho_R * Vdotn_R - rho_L * Vdotn_L) * dl,
291 (rho_R * u_R * Vdotn_R + P_R * nx - rho_L * u_L * Vdotn_L + P_L * nx) * dl,
292 (rho_R * v_R * Vdotn_R + P_R * ny - rho_L * v_L * Vdotn_L + P_L * ny) * dl,
293 ((((P_R / (gamma - 1.0)) + rho_R * Vmag_R) + P_R) * Vdotn_R -
294 (((P_L / (gamma - 1.0)) + rho_L * Vmag_L) + P_L) * Vdotn_L) * dl};
295
296 double alpha[kNv];
297 for (int i = 0; i < kNv; ++i)
298 condition_for_movers_nwsc(d_Uv[i], d_F[i], alpha[i]);
299
300 const double d_P = P_R - P_L;
301 const double P_I = 0.5 * (P_R + P_L);
302 const double Alpha_P = 0.5 * (fabs(Vdotn_R * dl) + fabs(Vdotn_L * dl));
303 const double beta = 0.21;
304 const double Sensor = (fabs(P_I) > 1e-14) ? beta * fabs(d_P / (2.0 * P_I)) : 0.0;
305
306 for (int i = 0; i < kNv; ++i)
307 out_diss[i] = 0.5 * (Sensor * alpha[i] + Alpha_P * d_Uv[i]);
308}

312__device__ double d_weno5_scalar(double a, double b, double c, double d, double e, int shift)
313{
314 if (!isfinite(a) || !isfinite(b) || !isfinite(c) || !isfinite(d) || !isfinite(e))
315 return c;
316 const double epsilon = 1e-6;
317 double d0, d1, d2, v0, v1, v2;
318 if (shift == 0)
319 {
320 d0 = 0.3;
321 d1 = 0.6;
322 d2 = 0.1;
323 v2 = (2.0 * a - 7.0 * b + 11.0 * c) / 6.0;
324 v1 = (-b + 5.0 * c + 2.0 * d) / 6.0;
325 v0 = (2.0 * c + 5.0 * d - e) / 6.0;
326 }
327 else
328 {
329 d0 = 0.1;
330 d1 = 0.6;
331 d2 = 0.3;
332 v2 = (-a + 5.0 * b + 2.0 * c) / 6.0;
333 v1 = (2.0 * b + 5.0 * c - d) / 6.0;
334 v0 = (11.0 * c - 7.0 * d + 2.0 * e) / 6.0;
335 }
336 const double t2 = a - 2.0 * b + c;
337 const double t2b = a - 4.0 * b + 3.0 * c;
338 const double b2 = (13.0 / 12.0) * t2 * t2 + 0.25 * t2b * t2b;
339 const double t1 = b - 2.0 * c + d;
340 const double t1b = b - d;
341 const double b1 = (13.0 / 12.0) * t1 * t1 + 0.25 * t1b * t1b;
342 const double t0 = c - 2.0 * d + e;
343 const double t0b = 3.0 * c - 4.0 * d + e;
344 const double b0 = (13.0 / 12.0) * t0 * t0 + 0.25 * t0b * t0b;
345 const double s0 = epsilon + b0;
346 const double s1 = epsilon + b1;
347 const double s2 = epsilon + b2;
348 const double a0 = d0 / (s0 * s0);
349 const double a1 = d1 / (s1 * s1);
350 const double a2 = d2 / (s2 * s2);
351 const double sum = a0 + a1 + a2;
352 double U;
353 if (sum < epsilon)
354 U = (v0 + v1 + v2) / 3.0;
355 else
356 U = (a0 * v0 + a1 * v1 + a2 * v2) / sum;
357 return isfinite(U) ? U : c;
358}

360__device__ void d_slip_reflect_U(const double U_in[kNv], double nx, double ny, double gamma, double U_out[kNv])
361{
362 const double rho = U_in[0];
363 if (!(rho > 1e-14) || !isfinite(rho))
364 {
365 for (int v = 0; v < kNv; ++v)
366 U_out[v] = U_in[v];
367 return;
368 }
369 const double u = U_in[1] / rho;
370 const double vv = U_in[2] / rho;
371 const double un = u * nx + vv * ny;
372 const double ug = u - 2.0 * un * nx;
373 const double vg = vv - 2.0 * un * ny;
374 double p = (gamma - 1.0) * (U_in[3] - 0.5 * rho * (u * u + vv * vv));
375 if (!(p > 0.0) || !isfinite(p))
376 p = 1e-12;
377 U_out[0] = rho;
378 U_out[1] = rho * ug;
379 U_out[2] = rho * vg;
380 U_out[3] = p / (gamma - 1.0) + rho * 0.5 * (ug * ug + vg * vg);
381}

483__device__ bool d_get_LR(double dL, double uL, double vL, double aL,
484 double dR, double uR, double vR, double aR,
485 double nx, double ny, double gamma,
486 double L[16], double IL[16])
487{
488 const double sqrt_dL = sqrt(fmax(dL, 1e-14));
489 const double sqrt_dR = sqrt(fmax(dR, 1e-14));
490 const double denom = sqrt_dL + sqrt_dR;
491 double u_RL, v_RL, a_RL;
492 if (denom < 1e-14)
493 {
494 u_RL = 0.5 * (uL + uR);
495 v_RL = 0.5 * (vL + vR);
496 a_RL = 0.5 * (aL + aR);
497 }
498 else
499 {
500 u_RL = (uL * sqrt_dL + uR * sqrt_dR) / denom;
501 v_RL = (vL * sqrt_dL + vR * sqrt_dR) / denom;
502 a_RL = (aL * sqrt_dL + aR * sqrt_dR) / denom;
503 }
504 if (!(a_RL > 1e-14) || !isfinite(a_RL))
505 a_RL = 1e-14;
506
507 const double gamma_m1 = gamma - 1.0;
508 const double vn = u_RL * nx + v_RL * ny;
509 const double ek = 0.5 * (u_RL * u_RL + v_RL * v_RL);
510 const double h = (a_RL * a_RL / gamma_m1) + ek;
511 const double t1 = 0.5 / (a_RL * a_RL);
512 const double t2 = gamma_m1 * t1;
513
514 L[0] = 1.0 - 2.0 * t2 * ek;
515 L[4] = 2.0 * t2 * u_RL;
516 L[8] = 2.0 * t2 * v_RL;
517 L[12] = -2.0 * t2;
518 L[1] = v_RL * nx - u_RL * ny;

656__device__ double d_positivity_theta(const double U0[kNv], const double Uw[kNv],
657 double gamma, double eps)
658{
659 double theta = 1.0;
660
661 if (!(Uw[0] > eps) || !isfinite(Uw[0]))
662 {
663 if (U0[0] > eps && isfinite(U0[0]) && (U0[0] - Uw[0]) > 1e-30)
664 theta = fmin(theta, (U0[0] - eps) / (U0[0] - Uw[0]));
665 else
666 theta = 0.0;
667 }
668 theta = fmax(0.0, fmin(1.0, theta));
669
670 if (!(d_blend_pressure(U0, Uw, theta, gamma) > eps))
671 {
672 double lo = 0.0, hi = theta;
673 for (int it = 0; it < 14; ++it)
674 {
675 const double mid = 0.5 * (lo + hi);
676 if (d_blend_pressure(U0, Uw, mid, gamma) > eps)
677 lo = mid;
678 else
679 hi = mid;
680 }
681 theta = lo;
682 }
683 if (!isfinite(theta) || theta < 0.0)
684 theta = 0.0;
685 if (theta > 1.0)
686 theta = 1.0;
687 return theta;
688}

812__global__ void net_flux_weno_ricca_kernel(
813 const int *leaf_cells,
814 int n_leaf,
815 const int *d_num_faces,
816 const int *d_neighbours,
817 const double *d_face_normals,
818 const double *d_face_areas,
819 const int *d_wall_face,
820 const double *d_U,
821 const double *d_prim,
822 double *d_net_flux,
823 int n_cells,
824 int n_phys,
825 int dissipation_type,
826 bool is_char,
827 double gamma,
828 double gamma1)
829{
830 const int li = blockIdx.x * blockDim.x + threadIdx.x;
831 if (li >= n_leaf)
832 return;
833 const int cell = leaf_cells[li];
834 if (cell < 0 || cell >= n_phys)
835 return;
836
837 for (int v = 0; v < kNv; ++v)
838 d_net_flux[cell * kNv + v] = 0.0;
839
840 const int nf = d_num_faces[cell];
841 if (nf != 4)
842 return;
843
844 for (int face = 0; face < 4; ++face)
845 {
846 const int neigh = d_neighbours[cell * kMaxFaces + face];
847 if (neigh < 0 || neigh >= n_cells)
848 continue;
849
850 const bool is_wall = d_wall_face[cell * kMaxFaces + face] != 0;
851 double U_L[kNv], U_R[kNv];
852 double U_cell[kNv], U_neigh[kNv];
853 d_load_cell_U(cell, d_U, n_cells, U_cell);
854 d_load_cell_U(neigh, d_U, n_cells, U_neigh);
855
856 /* Host WENO2D: walls / invalid neighbours use 1O cell averages (no WENO). */
857 if (is_wall || neigh >= n_phys)
858 {
859 for (int v = 0; v < kNv; ++v)
860 {
861 U_L[v] = U_cell[v];
862 U_R[v] = U_neigh[v];
863 }
864 }
865 else
866 {
867 d_weno_interior_face(cell, face, d_neighbours, d_wall_face, d_face_normals, d_U,
868 d_prim, n_phys, n_cells, gamma, is_char, U_L, U_R);
869 }
870
871 /* Zhang–Shu: convex-blend WENO faces toward 1O cell averages if ρ/p ≤ ε. */
872 {
873 constexpr double pos_eps = 1e-8;
874 d_positivity_limit_face(U_cell, U_L, gamma, pos_eps);
875 d_positivity_limit_face(U_neigh, U_R, gamma, pos_eps);
876 const double pL = d_pressure_from_U(U_L, gamma);
877 const double pR = d_pressure_from_U(U_R, gamma);
878 if (!(U_L[0] > pos_eps) || !(U_R[0] > pos_eps) || !(pL > pos_eps) || !(pR > pos_eps) ||
879 !isfinite(pL) || !isfinite(pR))
880 {
881 for (int v = 0; v < kNv; ++v)
882 {
883 U_L[v] = U_cell[v];
884 U_R[v] = U_neigh[v];
885 }
886 }
887 }
888
889 double rho_L = U_L[0], u_L = U_L[1] / fmax(U_L[0], 1e-14), v_L = U_L[2] / fmax(U_L[0], 1e-14);
890 double rho_R = U_R[0], u_R = U_R[1] / fmax(U_R[0], 1e-14), v_R = U_R[2] / fmax(U_R[0], 1e-14);
891 double P_L = d_pressure_from_U(U_L, gamma);
892 double P_R = d_pressure_from_U(U_R, gamma);
893 if (!(P_L > 1e-12))
894 P_L = 1e-12;
895 if (!(P_R > 1e-12))
896 P_R = 1e-12;
897
898 const double nx = d_face_normals[(cell * kMaxFaces + face) * 2 + 0];
899 const double ny = d_face_normals[(cell * kMaxFaces + face) * 2 + 1];
900 const double dl = d_face_areas[cell * kMaxFaces + face];
901 if (!(dl > 0.0))
902 continue;
903
904 if (is_wall)
905 P_R = P_L;
906
907 double avg[kNv], diss[kNv];
908 face_average_convective(rho_L, u_L, v_L, P_L, rho_R, u_R, v_R, P_R, nx, ny, dl, is_wall, gamma1, avg);
909
910 if (dissipation_type == 4)
911 {
912 /* RICCA on reconstructed face conservatives (host WENO path). */
913 const double Vdotn_L = u_L * nx + v_L * ny;
914 const double Vdotn_R = u_R * nx + v_R * ny;
915 const double Vmag_L = 0.5 * (u_L * u_L + v_L * v_L);
916 const double Vmag_R = 0.5 * (u_R * u_R + v_R * v_R);
917 double d_Uv[kNv] = {U_R[0] - U_L[0], U_R[1] - U_L[1], U_R[2] - U_L[2], U_R[3] - U_L[3]};
918 /* Match host WENO2D / 1O RICCA: dF = F_R - F_L with +P on both sides. */
919 const double d_F[kNv] = {
920 (rho_R * Vdotn_R - rho_L * Vdotn_L) * dl,
921 (rho_R * u_R * Vdotn_R + P_R * nx - rho_L * u_L * Vdotn_L + P_L * nx) * dl,
922 (rho_R * v_R * Vdotn_R + P_R * ny - rho_L * v_L * Vdotn_L + P_L * ny) * dl,
923 ((((P_R / (gamma - 1.0)) + rho_R * Vmag_R) + P_R) * Vdotn_R -
924 (((P_L / (gamma - 1.0)) + rho_L * Vmag_L) + P_L) * Vdotn_L) *
925 dl};
926 double dP = fabs(P_L - P_R);
927 const double P_I = 0.5 * (P_L + P_R);
928 const double Rho_I = 0.5 * (rho_L + rho_R);
929 for (int i = 0; i < kNv; ++i)
930 {
931 double alpha = 0.0;
932 condition_for_ricca(d_Uv[i], d_F[i], Vdotn_L, Vdotn_R, dP, Rho_I, P_I, alpha, gamma);
933 diss[i] = 0.5 * alpha * dl * d_Uv[i];
934 if (!isfinite(diss[i]))
935 diss[i] = 0.0;
936 }
937 }
938 else
939 {
940 for (int i = 0; i < kNv; ++i)
941 diss[i] = 0.0;
942 }
943
944 for (int v = 0; v < kNv; ++v)
945 d_net_flux[cell * kNv + v] += avg[v] - diss[v];
946 }
947}

1013 d_net_flux[cell * kNv + v] += avg[v] - diss[v];
1014 }
1015}

1017__device__ void conserv_to_prim(const double *U, double *prim, double gamma)
1018{
1019 double rho = U[0];
1020 if (rho < 1e-12)
1021 rho = 1e-12;
1022 const double u = U[1] / rho;
1023 const double v = U[2] / rho;
1024 const double ke = 0.5 * (u * u + v * v);
1025 double p = (gamma - 1.0) * (U[3] - rho * ke);
1026 if (p < 1e-12)
1027 p = 1e-12;
1028 prim[0] = rho;
1029 prim[1] = u;
1030 prim[2] = v;
1031 prim[3] = 0.0;
1032 prim[4] = p;
1033 prim[5] = sqrt(gamma * p / rho);
1034}
1035
1036/* Repair conservatives in-place so prim and U stay thermodynamically consistent.
1037 Blind p-clamp in conserv_to_prim alone leaves U[3] negative-internal-energy and
1038 the next WENO step amplifies until |V|→∞ and dt collapses. */
1039__device__ void repair_conserved(double *U, double gamma)
1040{
1041 const double rho_floor = 1e-8;
1042 const double p_floor = 1e-8;
1043 double rho = U[0];
1044 const bool rho_bad = !isfinite(rho) || rho < rho_floor;
1045 if (rho_bad)
1046 {
1047 /* Tiny/NaN density with leftover energy → a→∞; reset to a calm floor state. */
1048 U[0] = rho_floor;
1049 U[1] = 0.0;
1050 U[2] = 0.0;
1051 U[3] = p_floor / (gamma - 1.0);
1052 return;
1053 }
1054 double u = U[1] / rho;
1055 double v = U[2] / rho;
1056 if (!isfinite(u))
1057 u = 0.0;
1058 if (!isfinite(v))
1059 v = 0.0;
1060 /* Cap extreme velocities that already indicate a failed update. */
1061 const double v_cap = 20.0;
1062 const double speed = sqrt(u * u + v * v);
1063 if (speed > v_cap)
1064 {
1065 const double s = v_cap / speed;
1066 u *= s;
1067 v *= s;
1068 }
1069 U[1] = rho * u;
1070 U[2] = rho * v;
1071 double ke = 0.5 * (u * u + v * v);
1072 double p = (gamma - 1.0) * (U[3] - rho * ke);
1073 if (!isfinite(p) || p < p_floor)
1074 {
1075 p = p_floor;
1076 /* Near-vacuum: drop unsupported KE so a stays finite. */
1077 if (ke > 2.0)
1078 {
1079 u = 0.0;
1080 v = 0.0;
1081 ke = 0.0;
1082 U[1] = 0.0;
1083 U[2] = 0.0;
1084 }
1085 }
1086 U[3] = p / (gamma - 1.0) + rho * ke;
1087}

1089__global__ void inlet_ghost_kernel(const int *list, int n, double *U, double *prim,
1090 double rho, double u, double v, double p, double gamma)

1116__global__ void wall_ghost_kernel(const int *list, int n, double *U, double *prim,
1117 const double *face_normals, double gamma)
1118{

1120 if (i >= n)
1121 return;

1123 const int f = list[i * 3 + 1];
1124 const int g = list[i * 3 + 2];
1125 const double n1 = face_normals[(c * kMaxFaces + f) * 2 + 0];
1126 const double n2 = face_normals[(c * kMaxFaces + f) * 2 + 1];

1129 const double v = prim[c * 6 + 2];

1133 const double vg = v - 2.0 * un * n2;
1134 const double ke = 0.5 * (ug * ug + vg * vg);
1135 U[g * kNv + 0] = rho;
1136 U[g * kNv + 1] = rho * ug;
1137 U[g * kNv + 2] = rho * vg;

1139 conserv_to_prim(&U[g * kNv], &prim[g * 6], gamma);

1142__global__ void cell_dt_kernel(const int *leaf, int n_leaf, const int *num_faces,
1143 const double *face_normals, const double *face_areas,
1144 const double *prim, const double *inv_area_unused,
1145 const double *cell_area, double *dt_out, double cfl)
1146{
1147 (void)inv_area_unused;
1148 const int li = blockIdx.x * blockDim.x + threadIdx.x;
1149 if (li >= n_leaf)
1150 return;
1151 const int c = leaf[li];
1152 const double u = prim[c * 6 + 1];
1153 const double v = prim[c * 6 + 2];
1154 const double a = prim[c * 6 + 5];
1155 const int nf = num_faces[c];
1156 double denom = 0.0;
1157 for (int f = 0; f < nf && f < kMaxFaces; ++f)
1158 {
1159 const double nx = face_normals[(c * kMaxFaces + f) * 2 + 0];
1160 const double ny = face_normals[(c * kMaxFaces + f) * 2 + 1];
1161 const double dl = face_areas[c * kMaxFaces + f];
1162 denom += (fabs(u * nx + v * ny) + a) * dl;

1165}

1167__global__ void update_and_error_kernel(const int *leaf, int n_leaf, double *U, double *prim,
1168 const double *net_flux, const double *inv_area,
1169 double min_dt, double *err_sq, double gamma)
1170{
1171 const int li = blockIdx.x * blockDim.x + threadIdx.x;
1172 if (li >= n_leaf)
1173 return;
1174 const int c = leaf[li];
1175 const double s = -min_dt * inv_area[c];
1176 double local_e[kNv] = {0.0, 0.0, 0.0, 0.0};
1177 double Uold[kNv], Utry[kNv];
1178 for (int v = 0; v < kNv; ++v)
1179 Uold[v] = U[c * kNv + v];
1180 for (int v = 0; v < kNv; ++v)
1181 {
1182 const double dU = s * net_flux[c * kNv + v];
1183 Utry[v] = Uold[v] + dU;
1184 }
1185 /* Zhang–Shu cell update: U ← (1-θ)Uold + θ Utry with ρ,p ≥ ε (θ≤1). */
1186 double th = 1.0;
1187 {
1188 constexpr double pos_eps = 1e-8;
1189 for (int v = 0; v < kNv; ++v)
1190 {
1191 if (!isfinite(Utry[v]))
1192 {
1193 th = 0.0;
1194 break;
1195 }
1196 }
1197 if (th > 0.0)
1198 th = d_positivity_theta(Uold, Utry, gamma, pos_eps);
1199 for (int v = 0; v < kNv; ++v)
1200 U[c * kNv + v] = (1.0 - th) * Uold[v] + th * Utry[v];
1201 }
1202 /* Residual from *applied* update (θ·dU), not the rejected trial. */
1203 for (int v = 0; v < kNv; ++v)
1204 {
1205 const double dU_app = th * (Utry[v] - Uold[v]);
1206 const double denom = Uold[v];
1207 double t = (!isfinite(dU_app) || !isfinite(denom)) ? 0.0
1208 : (fabs(denom) < 1e-12 ? fabs(dU_app) : fabs(dU_app) / fabs(denom));
1209 local_e[v] = t * t;
1210 }
1211 repair_conserved(&U[c * kNv], gamma);
1212 /* Last-resort safety net if repair still leaves a runaway / vacuum state. */
1213 {
1214 const double rho = U[c * kNv + 0];
1215 bool bad = !isfinite(rho) || !(rho > 1e-10);
1216 if (!bad)
1217 {
1218 const double u = U[c * kNv + 1] / rho;
1219 const double v = U[c * kNv + 2] / rho;
1220 const double ke = 0.5 * (u * u + v * v);
1221 const double p = (gamma - 1.0) * (U[c * kNv + 3] - rho * ke);
1222 const double a2 = (rho > 0.0) ? (gamma * p / rho) : 0.0;
1223 const double speed = sqrt(u * u + v * v);
1224 bad = !isfinite(u) || !isfinite(v) || !isfinite(p) || !isfinite(a2) ||
1225 !(p > 1e-10) || speed > 20.0 || a2 > 20.0 * 20.0;
1226 }
1227 if (bad)
1228 {
1229 for (int v = 0; v < kNv; ++v)
1230 U[c * kNv + v] = Uold[v];
1231 repair_conserved(&U[c * kNv], gamma);
1232 }
1233 }
1234 conserv_to_prim(&U[c * kNv], &prim[c * 6], gamma);
1235 for (int v = 0; v < kNv; ++v)
1236 atomicAdd(&err_sq[v], local_e[v]);
1237}

1564static void diagnose_dt_collapse(const std::vector<double> &h_dt, double min_dt, double ref_dt, const char *tag)
1565{
1566 const int n_leaf = static_cast<int>(h_dt.size());
1567 if (n_leaf <= 0 || g_buf.h_leaf.size() != static_cast<size_t>(n_leaf))
1568 return;
1569
1570 int imin = 0;
1571 double max_dt = h_dt[0];
1572 int n_small = 0;
1573 const double thresh = (ref_dt > 0.0) ? 0.1 * ref_dt : 1e-6;
1574 for (int i = 0; i < n_leaf; ++i)
1575 {
1576 if (h_dt[i] < h_dt[imin])
1577 imin = i;
1578 if (h_dt[i] > max_dt)
1579 max_dt = h_dt[i];
1580 if (h_dt[i] < thresh)

1586 cudaMemcpy(U4.data(), g_buf.d_U + c * kNv, 4 * sizeof(double), cudaMemcpyDeviceToHost);

1588
1589 const double rho = U4[0];
1590 const double u = (fabs(rho) > 1e-30) ? U4[1] / rho : 0.0;
1591 const double v = (fabs(rho) > 1e-30) ? U4[2] / rho : 0.0;
1592 const double ke = 0.5 * (u * u + v * v);
1593 const double praw = (GAMMA_CONST - 1.0) * (U4[3] - rho * ke);
1594 const double speed = sqrt(u * u + v * v);
1595
1596 /* Global pathology counts (sample all leaf cells' raw pressure from U). */
1597 std::vector<double> h_U(g_buf.n_cells * kNv);
1598 std::vector<double> h_prim(g_buf.n_cells * 6);
1599 cudaMemcpy(h_U.data(), g_buf.d_U, g_buf.n_cells * kNv * sizeof(double), cudaMemcpyDeviceToHost);
1600 cudaMemcpy(h_prim.data(), g_buf.d_prim, g_buf.n_cells * 6 * sizeof(double), cudaMemcpyDeviceToHost);
1601
1602 int n_neg_p = 0, n_tiny_rho = 0, n_fast = 0, n_clamped_p = 0;
1603 double max_speed = 0.0, max_a = 0.0, min_praw = 1e300;

1605 for (int li = 0; li < n_leaf; ++li)
1606 {
1607 const int ci = g_buf.h_leaf[li];
1608 const double r = h_U[ci * kNv + 0];
1609 const double uu = (fabs(r) > 1e-30) ? h_U[ci * kNv + 1] / r : 0.0;
1610 const double vv = (fabs(r) > 1e-30) ? h_U[ci * kNv + 2] / r : 0.0;
1611 const double pr = (GAMMA_CONST - 1.0) * (h_U[ci * kNv + 3] - 0.5 * r * (uu * uu + vv * vv));
1612 const double spd = sqrt(uu * uu + vv * vv);
1613 const double a = h_prim[ci * 6 + 5];

1615 {
1616 ++n_neg_p;

1618 c_neg = ci;
1619 }
1620 if (!(r > 1e-6) || !std::isfinite(r))
1621 ++n_tiny_rho;
1622 if (pr < 1e-12 && r > 1e-6)
1623 ++n_clamped_p; /* would hit conserv_to_prim floor */

1625 {
1626 max_speed = spd + a;
1627 c_fast = ci;
1628 }
1629 if (a > max_a)
1630 max_a = a;
1631 if (pr < min_praw)
1632 min_praw = pr;
1633 if (spd + a > 50.0)
1634 ++n_fast;
1635 }

1637 std::cerr << std::scientific;
1638 std::cerr << "[GPU-DT-DEBUG] " << tag
1639 << " min_dt=" << min_dt << " ref_dt=" << ref_dt
1640 << " max_dt=" << max_dt << " n_dt<0.1*ref=" << n_small << "\n";
1641 std::cerr << "[GPU-DT-DEBUG] worst-dt cell=" << c
1642 << " U=[" << U4[0] << "," << U4[1] << "," << U4[2] << "," << U4[3] << "]"
1643 << " praw=" << praw << " |V|=" << speed
1644 << " prim(rho,u,v,p,a)=[" << prim6[0] << "," << prim6[1] << "," << prim6[2]
1645 << "," << prim6[4] << "," << prim6[5] << "]\n";
