173#endif

197 }
198 }
199 /* Print every 1000; write VTK/solution only every 10000 (I/O was dominating runtime). */
200 if (iterations % 1000 == 0)
201 {
202 if (CFD_MPI_Is_Root())
203 {
204 cout << setw(10) << iterations
205 << setw(15) << Min_dt
206 << setw(15) << Error[0]
207 << setw(15) << Error[1]

229 if (CFD_MPI_Is_Root())

231 cout << "Residuals below " << kResTol << " at iteration " << iterations

233#ifdef USE_CUDA

235 Resident_GPU_Explicit_Download_Host();
236#endif

240 }

242
243#ifdef USE_CUDA
244 if (use_resident_gpu)
245 Resident_GPU_Explicit_Download_Host();
246#endif
247 if (CFD_MPI_Is_Root())
248 {
249 Write_Final_Solution_Output(Error_Filename, Sol_Filename, Solution_Data_Type);
250 cout << "Inviscid solver completed successfully after " << iterations << " iterations" << endl;
251 cout << "Final Min_dt=" << Min_dt
252 << " errors(rho,rhou,rhov,rhoEt)="
253 << Error[0] << " " << Error[1] << " " << Error[2] << " " << Error[3] << endl;
254 }
255#ifdef USE_CUDA
256 if (use_resident_gpu)
257 Resident_GPU_Explicit_Shutdown();
258#endif
259 return true;
260 }
261 catch (const std::exception &e)
262 {
263 cerr << "Exception in Inviscid_Solver: " << e.what() << endl;
264 return false;
265 }
266 catch (...)
267 {
268 cerr << "Unknown exception occurred in Inviscid_Solver" << endl;
269 return false;
270 }
