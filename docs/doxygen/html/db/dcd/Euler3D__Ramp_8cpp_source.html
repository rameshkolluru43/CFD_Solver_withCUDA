50void quad_area_normal(const double p0[3], const double p1[3], const double p2[3],
51 const double p3[3], double n_unit[3], double &area)
52{
53 double d1[3], d2[3], c1[3], c2[3];
54 sub3(p2, p0, d1);
55 sub3(p3, p1, d2);
56 cross(d1, d2, c1);
57 /* alternate triangulation average */
58 sub3(p1, p0, d1);
59 sub3(p2, p0, d2);
60 cross(d1, d2, c1);
61 sub3(p2, p0, d1);
62 sub3(p3, p0, d2);
63 cross(d1, d2, c2);
64 double n[3] = {0.5 * (c1[0] + c2[0]), 0.5 * (c1[1] + c2[1]), 0.5 * (c1[2] + c2[2])};
65 area = norm3(n);
66 if (area < 1e-30)
67 {
68 n_unit[0] = n_unit[1] = n_unit[2] = 0.0;
69 area = 0.0;
70 return;
71 }
72 n_unit[0] = n[0] / area;
73 n_unit[1] = n[1] / area;
74 n_unit[2] = n[2] / area;
75}

77double hex_volume(const double p[8][3])
78{
79 /* Decompose into 5 or 6 tets from vertex 0 — use average of two decompositions. */
80 auto tet = [](const double a[3], const double b[3], const double c[3], const double d[3]) {
81 double ab[3], ac[3], ad[3], cr[3];
82 sub3(b, a, ab);
83 sub3(c, a, ac);
84 sub3(d, a, ad);
85 cross(ab, ac, cr);
86 return std::fabs(cr[0] * ad[0] + cr[1] * ad[1] + cr[2] * ad[2]) / 6.0;
87 };
88 const double v1 = tet(p[0], p[1], p[3], p[4]) + tet(p[1], p[2], p[3], p[6]) +
89 tet(p[1], p[3], p[4], p[6]) + tet(p[1], p[4], p[5], p[6]) +
90 tet(p[3], p[4], p[6], p[7]);
91 return v1;
92}

94double y_wall(double x, double x_ramp, double x_plateau, double angle_deg)
95{
96 const double ang = angle_deg * 3.14159265358979323846 / 180.0;
97 const double slope = std::tan(ang);
98 if (x <= x_ramp)
99 return 0.0;
100 if (x <= x_plateau)
101 return slope * (x - x_ramp);
102 return slope * (x_plateau - x_ramp);
103}

106{

118Mesh make_ramp15_mesh(const Config &cfg)
119{
120 if (cfg.nx < 4 || cfg.ny < 4 || cfg.nz < 1)
121 throw std::runtime_error("Ramp mesh needs nx,ny >= 4 and nz >= 1");
122
123 Mesh m;
124 m.nx = cfg.nx;
125 m.ny = cfg.ny;
126 m.nz = cfg.nz;
127 m.Lx = cfg.Lx;
128 m.Ly = cfg.Ly;
129 m.Lz = cfg.Lz;
130 m.n_cells = m.nx * m.ny * m.nz;
131 m.dx = cfg.Lx / m.nx;
132 m.dy = cfg.Ly / m.ny;
133 m.dz = cfg.Lz / m.nz;
134
135 const int npx = m.nx + 1, npy = m.ny + 1, npz = m.nz + 1;
136 const int n_nodes = npx * npy * npz;
137 m.nodes.assign(static_cast<std::size_t>(n_nodes) * 3, 0.0);
138 m.has_nodes = true;
139
140 const double x_ramp = cfg.ramp_x_start;
141 const double x_plateau = cfg.ramp_x_end;
142 const double ang = cfg.ramp_angle_deg;
143
144 for (int k = 0; k < npz; ++k)
145 for (int j = 0; j < npy; ++j)
146 for (int i = 0; i < npx; ++i)
147 {
148 const double x = cfg.Lx * (static_cast<double>(i) / m.nx);
149 const double z = cfg.Lz * (static_cast<double>(k) / m.nz);
150 const double yw = y_wall(x, x_ramp, x_plateau, ang);
151 const double eta = static_cast<double>(j) / m.ny;
152 const double y = yw + eta * (cfg.Ly - yw);
153 const int id = nid(i, j, k, npx, npy);
154 m.nodes[id * 3 + 0] = x;
155 m.nodes[id * 3 + 1] = y;
156 m.nodes[id * 3 + 2] = z;
157 }
158
159 m.xc.resize(m.n_cells);
160 m.yc.resize(m.n_cells);
161 m.zc.resize(m.n_cells);
162 m.volume.assign(m.n_cells, 0.0);
163 m.neigh.assign(m.n_cells * kFaces, -1);
164 m.nxyz.assign(m.n_cells * kFaces * 3, 0.0);
165 m.area.assign(m.n_cells * kFaces, 0.0);
166 m.wall_face.assign(m.n_cells * kFaces, 0);
167 m.bc_type.assign(m.n_cells * kFaces, BC_INTERIOR);
168
169 const int bc_face[kFaces] = {
170 parse_bc_local(cfg.bc_xmin), parse_bc_local(cfg.bc_xmax), parse_bc_local(cfg.bc_ymin),
171 parse_bc_local(cfg.bc_ymax), parse_bc_local(cfg.bc_zmin), parse_bc_local(cfg.bc_zmax)};
172
173 auto node = [&](int i, int j, int k, double out[3]) {
174 const int id = nid(i, j, k, npx, npy);
175 out[0] = m.nodes[id * 3 + 0];
176 out[1] = m.nodes[id * 3 + 1];
177 out[2] = m.nodes[id * 3 + 2];
178 };
179
180 for (int k = 0; k < m.nz; ++k)
181 for (int j = 0; j < m.ny; ++j)
182 for (int i = 0; i < m.nx; ++i)
183 {
184 const int c = cid(i, j, k, m.nx, m.ny);
185 double p[8][3];
186 node(i, j, k, p[0]);
187 node(i + 1, j, k, p[1]);
188 node(i + 1, j + 1, k, p[2]);
189 node(i, j + 1, k, p[3]);
190 node(i, j, k + 1, p[4]);
191 node(i + 1, j, k + 1, p[5]);
192 node(i + 1, j + 1, k + 1, p[6]);
193 node(i, j + 1, k + 1, p[7]);
194
195 m.volume[c] = hex_volume(p);
196 if (!(m.volume[c] > 1e-30))
197 throw std::runtime_error("Non-positive hex volume in ramp mesh");
198
199 m.xc[c] = m.yc[c] = m.zc[c] = 0.0;
200 for (int v = 0; v < 8; ++v)
201 {
202 m.xc[c] += p[v][0];
203 m.yc[c] += p[v][1];
204 m.zc[c] += p[v][2];
205 }
206 m.xc[c] *= 0.125;
207 m.yc[c] *= 0.125;
208 m.zc[c] *= 0.125;
209
210 /* Face corners ordered for outward normal (right-hand rule). */
211 const int face_nodes[kFaces][4] = {
212 {0, 3, 7, 4}, /* -x */
213 {1, 2, 6, 5}, /* +x */
214 {0, 1, 5, 4}, /* -y */
215 {3, 2, 6, 7}, /* +y */
216 {0, 1, 2, 3}, /* -z */
217 {4, 5, 6, 7} /* +z */
218 };
219 /* Fix -x / -y / -z winding if needed by flipping against cell center. */
220 for (int f = 0; f < kFaces; ++f)
221 {
222 double n[3], area;
223 quad_area_normal(p[face_nodes[f][0]], p[face_nodes[f][1]],
224 p[face_nodes[f][2]], p[face_nodes[f][3]], n, area);
225 double mid[3] = {0, 0, 0};
226 for (int q = 0; q < 4; ++q)
227 for (int d = 0; d < 3; ++d)
228 mid[d] += p[face_nodes[f][q]][d];
229 mid[0] *= 0.25;
230 mid[1] *= 0.25;
231 mid[2] *= 0.25;
232 const double outward =
233 n[0] * (mid[0] - m.xc[c]) + n[1] * (mid[1] - m.yc[c]) +
234 n[2] * (mid[2] - m.zc[c]);
235 if (outward < 0.0)
236 {
237 n[0] = -n[0];
238 n[1] = -n[1];
239 n[2] = -n[2];
240 }
241 m.area[c * kFaces + f] = area;
242 m.nxyz[(c * kFaces + f) * 3 + 0] = n[0];
243 m.nxyz[(c * kFaces + f) * 3 + 1] = n[1];
244 m.nxyz[(c * kFaces + f) * 3 + 2] = n[2];
245 }
246
247 const int ni[kFaces][3] = {{i - 1, j, k}, {i + 1, j, k}, {i, j - 1, k},
248 {i, j + 1, k}, {i, j, k - 1}, {i, j, k + 1}};
249 for (int f = 0; f < kFaces; ++f)
250 {
251 const int ii = ni[f][0], jj = ni[f][1], kk = ni[f][2];
252 if (ii >= 0 && ii < m.nx && jj >= 0 && jj < m.ny && kk >= 0 && kk < m.nz)
253 m.neigh[c * kFaces + f] = cid(ii, jj, kk, m.nx, m.ny);
254 else
255 {
256 m.wall_face[c * kFaces + f] = 1;
257 m.bc_type[c * kFaces + f] = bc_face[f];
258 }
259 }
260 }
261
262 return m;
263}
