30double theta_beta_M_residual(double beta, double M, double gamma, double theta)
31{
32 const double sb2 = std::sin(beta) * std::sin(beta);
33 const double num = 2.0 * (1.0 / std::tan(beta)) * (M * M * sb2 - 1.0);
34 const double den = M * M * (gamma + std::cos(2.0 * beta)) + 2.0;
35 return std::atan(num / den) - theta;
36}

38double solve_shock_angle(double M, double gamma, double theta)
39{
40 const double mu = std::asin(std::min(1.0, 1.0 / std::max(M, 1.01)));
41 double lo = mu + 1.0e-6;
42 double hi = 0.5 * 3.141592653589793 - 1.0e-4;
43 for (int n = 0; n < 80; ++n)
44 {
45 const double mid = 0.5 * (lo + hi);
46 const double fm = theta_beta_M_residual(mid, M, gamma, theta);
47 if (theta_beta_M_residual(lo, M, gamma, theta) * fm <= 0.0)
48 hi = mid;
49 else
50 lo = mid;
51 }
52 return 0.5 * (lo + hi);
53}

57void fill_oblique_postshock(Config &cfg)
58{
59 const double rho1 = std::max(cfg.freestream_rho, 1.0e-14);
60 const double p1 = std::max(cfg.freestream_p, 1.0e-14);
61 const double a1 = std::sqrt(kGamma * p1 / rho1);
62 const double M1 = std::fabs(cfg.freestream_u) / std::max(a1, 1.0e-14);
63 const double theta = cfg.shock_theta_deg * 3.141592653589793 / 180.0;
64
65 cfg.post_rho = rho1;
66 cfg.post_u = cfg.freestream_u;
67 cfg.post_v = cfg.freestream_v;
68 cfg.post_w = cfg.freestream_w;
69 cfg.post_p = p1;
70 if (theta < 1.0e-8 || M1 <= 1.01)
71 return;
72
73 const double beta = solve_shock_angle(M1, kGamma, theta);
74 const double Mn1 = M1 * std::sin(beta);
75 const double p2 = p1 * (1.0 + 2.0 * kGamma / (kGamma + 1.0) * (Mn1 * Mn1 - 1.0));
76 const double rho2 = rho1 * ((kGamma + 1.0) * Mn1 * Mn1) /
77 ((kGamma - 1.0) * Mn1 * Mn1 + 2.0);
78 const double Mn2 = std::sqrt((1.0 + 0.5 * (kGamma - 1.0) * Mn1 * Mn1) /
79 (kGamma * Mn1 * Mn1 - 0.5 * (kGamma - 1.0)));
80 const double M2 = Mn2 / std::sin(std::max(beta - theta, 1.0e-8));
81 const double a2 = std::sqrt(kGamma * p2 / rho2);
82 const double V2 = M2 * a2;
83 /* Shock from the top wall: flow is turned downward. */
84 cfg.post_rho = rho2;
85 cfg.post_p = p2;
86 cfg.post_u = V2 * std::cos(theta);
87 cfg.post_v = -V2 * std::sin(theta);
88 cfg.post_w = 0.0;
89
90 const double x_hit = cfg.shock_x_top + cfg.Ly / std::tan(beta);
91 std::cout << "SWBLI RH: M1=" << M1 << " theta=" << cfg.shock_theta_deg
92 << " deg beta=" << beta * 180.0 / 3.141592653589793
93 << " deg impinge x≈" << x_hit << " p2/p1=" << p2 / p1 << "\n";
94}

96Mesh make_swbli_plate_mesh(const Config &cfg)
97{
98 Mesh m;
99 m.nx = cfg.nx;
100 m.ny = cfg.ny;
101 m.nz = cfg.nz;
102 m.Lx = cfg.Lx;
103 m.Ly = cfg.Ly;
104 m.Lz = cfg.Lz;
105 m.n_cells = m.nx * m.ny * m.nz;
106 m.dx = cfg.Lx / std::max(m.nx, 1);
107 m.dz = cfg.Lz / std::max(m.nz, 1);
108 m.dy = cfg.Ly / std::max(m.ny, 1);
109 m.has_nodes = true;
110
111 const int npx = m.nx + 1, npy = m.ny + 1, npz = m.nz + 1;
112 m.nodes.assign(static_cast<std::size_t>(npx) * npy * npz * 3, 0.0);
113
114 const double stretch = std::max(cfg.y_stretch, 1.0);
115 std::vector<double> ynode(static_cast<std::size_t>(npy), 0.0);
116 if (stretch <= 1.0 + 1.0e-12)
117 {
118 for (int j = 0; j < npy; ++j)
119 ynode[j] = cfg.Ly * j / m.ny;
120 }
121 else
122 {
123 const double denom = std::pow(stretch, m.ny) - 1.0;
124 for (int j = 0; j < npy; ++j)
125 ynode[j] = cfg.Ly * (std::pow(stretch, j) - 1.0) / denom;
126 }
127
128 for (int k = 0; k < npz; ++k)
129 for (int j = 0; j < npy; ++j)
130 for (int i = 0; i < npx; ++i)
131 {
132 const int id = nid(i, j, k, npx, npy);
133 m.nodes[id * 3 + 0] = cfg.Lx * i / m.nx;
134 m.nodes[id * 3 + 1] = ynode[j];
135 m.nodes[id * 3 + 2] = cfg.Lz * k / m.nz;
136 }
137
138 m.xc.resize(m.n_cells);
139 m.yc.resize(m.n_cells);
140 m.zc.resize(m.n_cells);
141 m.volume.assign(m.n_cells, 0.0);
142 m.neigh.assign(m.n_cells * kFaces, -1);
143 m.nxyz.assign(m.n_cells * kFaces * 3, 0.0);
144 m.area.assign(m.n_cells * kFaces, 0.0);
145 m.wall_face.assign(m.n_cells * kFaces, 0);
146 m.bc_type.assign(m.n_cells * kFaces, BC_INTERIOR);
147
148 auto parse_local = [](const std::string &s) {
149 if (s == "noslip" || s == "no_slip" || s == "wall_viscous")
150 return BC_NOSLIP;
151 if (s == "extrapolate" || s == "outlet" || s == "outflow")
152 return BC_EXTRAP;
153 if (s == "freestream" || s == "inlet" || s == "farfield")
154 return BC_FREESTREAM;
155 if (s == "postshock" || s == "shock" || s == "rh")
156 return BC_POSTSHOCK;
157 return BC_SLIP;
158 };
159 const int bc_face[kFaces] = {
160 parse_local(cfg.bc_xmin), parse_local(cfg.bc_xmax), parse_local(cfg.bc_ymin),
161 parse_local(cfg.bc_ymax), parse_local(cfg.bc_zmin), parse_local(cfg.bc_zmax)};
162
163 const double nrm[kFaces][3] = {{-1, 0, 0}, {1, 0, 0}, {0, -1, 0},
164 {0, 1, 0}, {0, 0, -1}, {0, 0, 1}};
165
166 for (int k = 0; k < m.nz; ++k)
167 for (int j = 0; j < m.ny; ++j)
168 for (int i = 0; i < m.nx; ++i)
169 {
170 const int c = cid(i, j, k, m.nx, m.ny);
171 const double x0 = cfg.Lx * i / m.nx;
172 const double x1 = cfg.Lx * (i + 1) / m.nx;
173 const double y0 = ynode[j];
174 const double y1 = ynode[j + 1];
175 const double z0 = cfg.Lz * k / m.nz;
176 const double z1 = cfg.Lz * (k + 1) / m.nz;
177 const double dx = x1 - x0;
178 const double dy = y1 - y0;
179 const double dz = z1 - z0;
180 m.xc[c] = 0.5 * (x0 + x1);
181 m.yc[c] = 0.5 * (y0 + y1);
182 m.zc[c] = 0.5 * (z0 + z1);
183 m.volume[c] = dx * dy * dz;
184
185 const double face_area[kFaces] = {dy * dz, dy * dz, dx * dz,
186 dx * dz, dx * dy, dx * dy};
187 const int ni[kFaces][3] = {{i - 1, j, k}, {i + 1, j, k}, {i, j - 1, k},
188 {i, j + 1, k}, {i, j, k - 1}, {i, j, k + 1}};
189 for (int f = 0; f < kFaces; ++f)
190 {
191 m.area[c * kFaces + f] = face_area[f];
192 m.nxyz[(c * kFaces + f) * 3 + 0] = nrm[f][0];
193 m.nxyz[(c * kFaces + f) * 3 + 1] = nrm[f][1];
194 m.nxyz[(c * kFaces + f) * 3 + 2] = nrm[f][2];
195 const int ii = ni[f][0], jj = ni[f][1], kk = ni[f][2];
196 if (ii >= 0 && ii < m.nx && jj >= 0 && jj < m.ny && kk >= 0 && kk < m.nz)
197 m.neigh[c * kFaces + f] = cid(ii, jj, kk, m.nx, m.ny);
198 else
199 {
200 m.wall_face[c * kFaces + f] = 1;
201 int bc = bc_face[f];
202 /* Incident shock generator: post-shock RH on ymax for x >= Shock_X_Top. */
203 if (f == 3 && m.xc[c] >= cfg.shock_x_top)
204 bc = BC_POSTSHOCK;
205 m.bc_type[c * kFaces + f] = bc;
206 }
207 }
208 }
209
210 return m;
211}
