6void write_grid_vtk(const string &filename, Grid &G, int nx, int ny)
7{

35 for (int i = 0; i < nx; i++)

152/* ======================================================================
153 Test 3: Boundary layer grid (prescribed wall-normal clustering).
154
155 For a rectangle with horizontal south/north and matching east/west point
156 distributions, TFI gives y(i,j) = Y[j] for all i — the interior inherits the
157 wall-normal spacing from the side boundaries. The elliptic Poisson step would
158 move interior points and destroy that spacing, so this test uses TFI only.
159
160 Reference spacing is taken from the east boundary line (same as west at each j).
161 ====================================================================== */
162void test_boundary_layer_grid()
163{
164 cout << "\n========== TEST 3: Boundary Layer Grid (BL Clustering) ==========\n";
165 int nx = 61, ny = 41;
166
167 Point p1, p2;
168 Line south, east, north, west;
169
170 p1(0.0, 0.0, 0.0); p2(2.0, 0.0, 0.0);
171 south.generate(p1, p2, nx);
172
173 double first_height = 0.001;
174 double growth = 1.15;
175 p1(2.0, 0.0, 0.0); p2(2.0, 0.5, 0.0);
176 east.generate_boundary_layer(p1, p2, ny, first_height, growth);
177 p1(0.0, 0.5, 0.0); p2(0.0, 0.0, 0.0);
178 Line west_bl;
179 west_bl.generate_boundary_layer(p2, p1, ny, first_height, growth);
180 west_bl.Reverse_Points();
181 west = west_bl;
182
183 p1(2.0, 0.5, 0.0); p2(0.0, 0.5, 0.0);
184 north.generate(p1, p2, nx);
185
186 vector<Point> s_pts = south.Get_Point_list();
187 vector<Point> e_pts = east.Get_Point_list();
188 vector<Point> n_pts = north.Get_Point_list();
189 vector<Point> w_pts = west.Get_Point_list();
190
191 double dy0_ref = e_pts[1].Get_y() - e_pts[0].Get_y();
192 double dy1_ref = e_pts[2].Get_y() - e_pts[1].Get_y();
193 double growth_ref = (dy0_ref > 1e-30) ? (dy1_ref / dy0_ref) : 0.0;
194
195 cout << " Reference (east line): first dy=" << dy0_ref << ", second dy=" << dy1_ref
196 << ", growth=" << growth_ref << "\n";
197 cout << " Mode: TFI only (elliptic OFF — Poisson would violate prescribed BL spacing).\n";
198
199 Grid G;
200 G(s_pts, e_pts, n_pts, w_pts);
201
202 bool periodic = false, elliptic = false;
203 int iters = 0;
204 G.Generate_Grid(periodic, elliptic, iters, 1.5, 1e-10);
205
206 write_grid_vtk("test3_boundary_layer.vtk", G, nx, ny);
207
208 int i_left = nx / 4, i_mid = nx / 2, i_right = (3 * nx) / 4;
209 double dy0_left = G.Get_y(i_left, 1) - G.Get_y(i_left, 0);
210 double dy0_mid = G.Get_y(i_mid, 1) - G.Get_y(i_mid, 0);
211 double dy0_right = G.Get_y(i_right, 1) - G.Get_y(i_right, 0);
212 double dy1_mid = G.Get_y(i_mid, 2) - G.Get_y(i_mid, 1);
213 cout << " First cell dy at left (i=" << i_left << "): " << dy0_left << "\n"
214 << " First cell dy at mid (i=" << i_mid << "): " << dy0_mid << "\n"
215 << " First cell dy at right (i=" << i_right << "): " << dy0_right << "\n"
216 << " Second cell dy at mid: " << dy1_mid << "\n"
217 << " Growth ratio at mid: " << dy1_mid / dy0_mid << "\n";
218
219 double min_jac, max_ar, max_skew, min_orth;
220 G.ComputeGridQuality(min_jac, max_ar, max_skew, min_orth);
221 assert(min_jac > 0 && "Negative Jacobian in BL grid!");
222
223 const double tol = 1e-9;
224 auto near_ref = [&](double a, double b) {
225 return fabs(a - b) <= tol * (1.0 + fabs(b));
226 };
227 assert(near_ref(dy0_left, dy0_ref) && near_ref(dy0_mid, dy0_ref) && near_ref(dy0_right, dy0_ref)
228 && "TFI interior first cell must match east-boundary BL spacing");
229 assert(near_ref(dy1_mid, dy1_ref) && "TFI interior second cell must match east boundary");
230 double gr_mid = dy1_mid / dy0_mid;
231 assert(near_ref(gr_mid, growth_ref) && "Growth ratio should match line-generated boundary");
232
233 cout << "TEST 3 PASSED: min_jac=" << min_jac << " first_dy=" << dy0_mid
234 << " (matches boundary line)\n";
235}
