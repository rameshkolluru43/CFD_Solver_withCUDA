145inline void cudaCheckKernel(const char *kernel_name)
146{
147 cudaError_t err = cudaGetLastError();
148 if (err != cudaSuccess)
149 {
150 printf("CUDA Kernel Error in %s: %s\n", kernel_name, cudaGetErrorString(err));
151 exit(1);
