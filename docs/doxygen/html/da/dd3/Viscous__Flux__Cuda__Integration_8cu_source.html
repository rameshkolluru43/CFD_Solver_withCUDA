29__device__ double d_sutherland_mu(double T)
30{
31 const double term1 = 110.4 / 288.15;
32 if (!(T > 1e-12) || !isfinite(T))
33 return 1.0;
34 const double term2 = T + term1;
35 return pow(T, 1.5) * ((1.0 + term1) / term2);
36}

38__device__ void green_gauss_grad_cell(
39 int cell,
40 int grad_comp,
41 const int *num_faces,
42 const int *neighbours,
43 const double *face_normals,
44 const double *face_areas,
45 double inv_area,
46 const double *prim,
47 int n_cells,
48 double &gx,
49 double &gy)
50{
51 gx = 0.0;
52 gy = 0.0;
53 if (inv_area <= 0.0)
54 return;
55
56 const double phi_c = prim_at(prim, cell, grad_comp);
57 const int nf = num_faces[cell];
58
59 for (int f = 0; f < nf && f < kMaxFaces; ++f)
60 {
61 const int neigh = neighbours[cell * kMaxFaces + f];
62 double phi_n = phi_c;
63 if (neigh >= 0 && neigh < n_cells)
64 phi_n = prim_at(prim, neigh, grad_comp);
65
66 const int nb = (cell * kMaxFaces + f) * 2;
67 const double nx = face_normals[nb + 0];
68 const double ny = face_normals[nb + 1];
69 const double dl = face_areas[cell * kMaxFaces + f];
70 const double phi_face = 0.5 * (phi_c + phi_n);
71 gx += phi_face * nx * dl;
72 gy += phi_face * ny * dl;
73 }
74 gx *= inv_area;
75 gy *= inv_area;
76}

78__device__ void face_grad_corrected(
79 int cell,
80 int neigh,
81 int grad_comp,
82 const int *num_faces,
83 const int *neighbours,
84 const double *face_normals,
85 const double *face_areas,
86 const double *inv_area,
87 const double *prim,
88 const double *centers,
89 int n_cells,
90 double alpha,
91 double &gx,
92 double &gy)
93{
94 double gxc = 0.0, gyc = 0.0, gxn = 0.0, gyn = 0.0;
95 green_gauss_grad_cell(cell, grad_comp, num_faces, neighbours, face_normals, face_areas,
96 inv_area[cell], prim, n_cells, gxc, gyc);
97 if (neigh >= 0 && neigh < n_cells)
98 green_gauss_grad_cell(neigh, grad_comp, num_faces, neighbours, face_normals, face_areas,
99 inv_area[neigh], prim, n_cells, gxn, gyn);
100 else
101 {
102 gxn = gxc;
103 gyn = gyc;
104 }
105 gx = 0.5 * (gxc + gxn);
106 gy = 0.5 * (gyc + gyn);
107
108 if (neigh < 0 || neigh >= n_cells || centers == nullptr)
109 return;
110
111 const double dx = centers[neigh * 2 + 0] - centers[cell * 2 + 0];
112 const double dy = centers[neigh * 2 + 1] - centers[cell * 2 + 1];
113 const double ds2 = dx * dx + dy * dy;
114 if (!(ds2 > 1e-30))
115 return;
116 const double ds = sqrt(ds2);
117 const double ex = dx / ds;
118 const double ey = dy / ds;
119 const double dphi_exact = (prim_at(prim, neigh, grad_comp) - prim_at(prim, cell, grad_comp)) / ds;
120 const double dphi_approx = gx * ex + gy * ey;
121 const double corr = alpha * (dphi_exact - dphi_approx);
122 gx += corr * ex;
123 gy += corr * ey;
124}

126__device__ void viscous_flux_on_face(
127 int cell,
128 int neigh,
129 int face,
130 const int *num_faces,
131 const int *neighbours,
132 const double *face_normals,
133 const double *face_areas,
134 const double *inv_area,
135 const double *prim,
136 const double *centers,
137 int n_cells,
138 double inv_re,
139 double k1_base,
140 double alpha,
141 double &f0,
142 double &f1,
143 double &f2,
144 double &f3)
145{
146 f0 = f1 = f2 = f3 = 0.0;
147 if (neigh < 0 || neigh >= n_cells)
148 return;
149
150 const double nx = face_normals[(cell * kMaxFaces + face) * 2 + 0];
151 const double ny = face_normals[(cell * kMaxFaces + face) * 2 + 1];
152 const double dl = face_areas[cell * kMaxFaces + face];
153 if (dl <= 0.0)
154 return;
155
156 const double u_L = prim_at(prim, cell, 1);
157 const double v_L = prim_at(prim, cell, 2);
158 const double u_R = prim_at(prim, neigh, 1);
159 const double v_R = prim_at(prim, neigh, 2);
160 double mu = 0.5 * (prim_at(prim, cell, 8) + prim_at(prim, neigh, 8));
161 if (!(mu > 0.0) || !isfinite(mu))
162 {
163 const double T_L = prim_at(prim, cell, 3);
164 const double T_R = prim_at(prim, neigh, 3);
165 mu = 0.5 * (d_sutherland_mu(T_L) + d_sutherland_mu(T_R));
166 }
167
168 const double v1 = 0.5 * (u_L + u_R);
169 const double v2 = 0.5 * (v_L + v_R);
170
171 double u11, u12, u21, u22, tgx, tgy;
172 face_grad_corrected(cell, neigh, 1, num_faces, neighbours, face_normals, face_areas, inv_area,
173 prim, centers, n_cells, alpha, u11, u12);
174 face_grad_corrected(cell, neigh, 2, num_faces, neighbours, face_normals, face_areas, inv_area,
175 prim, centers, n_cells, alpha, u21, u22);
176 face_grad_corrected(cell, neigh, 3, num_faces, neighbours, face_normals, face_areas, inv_area,
177 prim, centers, n_cells, alpha, tgx, tgy);
178
179 const double T11 = (2.0 / 3.0) * mu * inv_re * (2.0 * u11 - u22);
180 const double T12 = mu * inv_re * (u12 + u21);
181 const double T21 = T12;
182 const double T22 = (2.0 / 3.0) * mu * inv_re * (2.0 * u22 - u11);
183 const double k1 = k1_base * mu;
184 const double Qx = k1 * tgx;
185 const double Qy = k1 * tgy;
186
187 f0 = 0.0;
188 f1 = (T11 * nx + T21 * ny) * dl;
189 f2 = (T12 * nx + T22 * ny) * dl;
190 f3 = ((T11 * v1 + T12 * v2 + Qx) * nx + (T21 * v1 + T22 * v2 + Qy) * ny) * dl;
191}
