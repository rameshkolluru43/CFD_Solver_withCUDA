25__device__ void estimate_wave_speeds_device(
26 double rho_L, double u_L, double v_L, double P_L, double a_L,
27 double rho_R, double u_R, double v_R, double P_R, double a_R,
28 double nx, double ny,
29 double& S_L, double& S_R, double& S_star
30) {
31 // Normal velocities
32 double un_L = nx * u_L + ny * v_L;
33 double un_R = nx * u_R + ny * v_R;
34
35 // Roe-averaged sound speed
36 double sqrt_rho_L = sqrt(rho_L);
37 double sqrt_rho_R = sqrt(rho_R);
38 double a_roe = (a_L * sqrt_rho_L + a_R * sqrt_rho_R) / (sqrt_rho_L + sqrt_rho_R);
39 double u_roe = (u_L * sqrt_rho_L + u_R * sqrt_rho_R) / (sqrt_rho_L + sqrt_rho_R);
40 double v_roe = (v_L * sqrt_rho_L + v_R * sqrt_rho_R) / (sqrt_rho_L + sqrt_rho_R);
41 double un_roe = nx * u_roe + ny * v_roe;
42
43 // Wave speed estimates (Davis)
44 S_L = fmin(un_L - a_L, un_roe - a_roe);
45 S_R = fmax(un_R + a_R, un_roe + a_roe);
46
47 // Contact wave speed (middle wave)
48 double numer = P_R - P_L + rho_L * un_L * (S_L - un_L) - rho_R * un_R * (S_R - un_R);
49 double denom = rho_L * (S_L - un_L) - rho_R * (S_R - un_R);
50
51 if (fabs(denom) > 1e-14) {
52 S_star = numer / denom;
53 } else {
54 S_star = 0.5 * (un_L + un_R);
55 }
56}

61__device__ void compute_star_state_device(
62 double rho, double u, double v, double E, double P,
63 double nx, double ny, double S_K, double S_star,
64 double U_star[4]
65) {
66 double un = nx * u + ny * v;
67
68 double factor = rho * (S_K - un) / (S_K - S_star);
69
70 U_star[0] = factor;
71 U_star[1] = factor * (u + (S_star - un) * nx);
72 U_star[2] = factor * (v + (S_star - un) * ny);
73 U_star[3] = factor * (E / rho + (S_star - un) * (S_star + P / (rho * (S_K - un))));
74}

154 flux[0] = rho_L * un_L;
155 flux[1] = rho_L * u_L * un_L + P_L * nx;
156 flux[2] = rho_L * v_L * un_L + P_L * ny;
157 flux[3] = (U_L[3] + P_L) * un_L;
158
159 } else if (S_R <= 0.0) {

164 flux[2] = rho_R * v_R * un_R + P_R * ny;

170 double F_L[4];
171 F_L[0] = rho_L * un_L;
172 F_L[1] = rho_L * u_L * un_L + P_L * nx;
173 F_L[2] = rho_L * v_L * un_L + P_L * ny;
174 F_L[3] = (U_L[3] + P_L) * un_L;

178 nx, ny, S_L, S_star, U_star_L);
179
180 for (int i = 0; i < 4; i++) {
181 flux[i] = F_L[i] + S_L * (U_star_L[i] - U_L[i]);
182 }
183
184 } else {
185 // Right star region
186 double un_R = nx * u_R + ny * v_R;
187 double F_R[4];
188 F_R[0] = rho_R * un_R;
189 F_R[1] = rho_R * u_R * un_R + P_R * nx;
190 F_R[2] = rho_R * v_R * un_R + P_R * ny;
191 F_R[3] = (U_R[3] + P_R) * un_R;
192

195 nx, ny, S_R, S_star, U_star_R);
196

198 flux[i] = F_R[i] + S_R * (U_star_R[i] - U_R[i]);
199 }
200 }
201
202 // Store flux
203 int flux_offset = (cell_idx * 4 + face) * 4;
204 for (int i = 0; i < 4; i++) {
205 flux_output[flux_offset + i] = flux[i] * area;
206 }
