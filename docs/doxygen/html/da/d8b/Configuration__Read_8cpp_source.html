149// Function to read parameters from a configuration file
150map<string, string> ReadConfigFile(const string &filename)
151{
152 map<string, string> config;
153 ifstream file = CFD_OpenInputFileOrThrow(filename, "ReadConfigFile");
154
155 string line;
156 while (getline(file, line))
157 {
158 if (line.empty() || line[0] == '#')
159 continue; // Skip empty lines and comments
160 size_t delimiter = line.find('=');
161 if (delimiter == string::npos)
162 continue;
163
164 string key = line.substr(0, delimiter);
165 string value = line.substr(delimiter + 1);
166
167 // Trim whitespace from key and value
168 key.erase(key.find_last_not_of(" \t") + 1);
169 value.erase(0, value.find_first_not_of(" \t"));
170 value.erase(value.find_last_not_of(" \t") + 1);
171
172 config[key] = value;
173 }
174 file.close();
175 return config;
176}
