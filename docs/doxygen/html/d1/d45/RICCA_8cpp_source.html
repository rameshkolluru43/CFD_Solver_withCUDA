31void Condition_For_RICCA(double &d_U, double &d_F, double &Vn_L, double &Vn_R, double &dP, double &Rho_I, double &P_I, double &Alpha)
32{
33 double epsilon = 1e-8, Max = 0.0, ita = 1.0;
34 //***************************************************************************************************************
35 Alpha = 0.0;
36 if (dP < epsilon)
37 dP = 0.0;
38
39 Max = max(fabs(Vn_L), fabs(Vn_R));
40 if ((fabs(d_F) < epsilon and fabs(d_U) < epsilon))
41 {
42 Alpha = 0.5 * (fabs(Vn_L) + fabs(Vn_R));
43 }
44 else
