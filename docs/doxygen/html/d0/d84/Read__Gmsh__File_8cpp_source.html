320void mapFacesToCells(vector<Cell> &cells, map<pair<int, int>, set<int>> &faceToCells)
321{
322 set<pair<int, int>> nonManifoldFaces;
323
324 for (size_t cellID = 0; cellID < cells.size(); ++cellID)
325 {
326 const auto &cellNodes = cells[cellID].nodeIndices;
327 for (size_t i = 0; i < cellNodes.size(); ++i)
328 {
329 int node1 = cellNodes[i];
330 int node2 = cellNodes[(i + 1) % cellNodes.size()];
331 if (node1 > node2)
332 swap(node1, node2);
333 pair<int, int> face = {node1, node2};
334
335 auto it = faceToCells.find(face);
336 if (it != faceToCells.end())
337 {
338 set<int> &sharedCells = it->second;
339 sharedCells.insert(static_cast<int>(cellID));
340 if (sharedCells.size() > 2)
341 nonManifoldFaces.insert(face);
342 }
343 else
344 faceToCells[face] = {static_cast<int>(cellID)};
345 }
346 }
347
348 if (!nonManifoldFaces.empty())
349 {
350 cout << "Warning: Non-manifold faces detected (shared by >2 cells).\n";
351 }
352}

354// Fill Neighbours in face order: Neighbours[f] = cell across face f, or ghost index for boundary.
355void fillFaceOrderedNeighbours(vector<Cell> &cells, const map<pair<int, int>, set<int>> &faceToCells, const int No_Physical_Cells, int &GhostCellIndex)
356{
357 const int nCells = static_cast<int>(cells.size());
358 for (int cellID = 0; cellID < nCells; cellID++)
359 {
360 const auto &cellNodes = cells[cellID].nodeIndices;
361 const int nF = static_cast<int>(cellNodes.size());
362 cells[cellID].Neighbours.assign(nF, -1);
363
364 for (int f = 0; f < nF; f++)
365 {
366 int node1 = cellNodes[f];
367 int node2 = cellNodes[(f + 1) % nF];
368 if (node1 > node2)
369 swap(node1, node2);
370 pair<int, int> edge = {node1, node2};
371
372 auto it = faceToCells.find(edge);
373 if (it != faceToCells.end() && it->second.size() == 2)
374 {
375 for (int other : it->second)
376 {
377 if (other != cellID)
378 {
379 cells[cellID].Neighbours[f] = other;
380 break;
381 }
382 }
383 }
384 // else: boundary face, leave -1; will assign ghost below
385 }
386
387 for (int f = 0; f < nF; f++)
388 {
389 if (cells[cellID].Neighbours[f] < 0)
390 {
391 cells[cellID].Neighbours[f] = GhostCellIndex++;
392 cells[cellID].hasBoundaryface = true;
393 cells[cellID].NoBoundaryFaces++;
394 }
395 }
396 }

480std::string Get_Boundary_Type(V_D &p1, V_D &p2, double &minX, double &maxX, double &minY, double &maxY, bool isBoundaryFace)

504{

507 for (size_t i = 0; i < Cells.size(); i++)

569void Create_Boundary_Cells_Lists(vector<Cell> &Cells, vector<int> &Inlet_Cells_List, vector<int> &Exit_Cells_List, vector<int> &Wall_Cells_List)
570{
571 // Per-face boundary lists: (cell, face_index, ghost_cell_index) for each boundary face.
572 cout << "Creating Boundary Cells Lists" << endl;
573 cout << "Size of Cells is : " << Cells.size() << endl;
574 const int nPhys = No_Physical_Cells;
575
576 for (size_t i = 0; i < Cells.size(); i++)
577 {
578 if (static_cast<int>(i) >= nPhys)
579 break;
580 const int nF = static_cast<int>(Cells[i].Face_Boundary_Kind.size());
581 for (int f = 0; f < nF; f++)
582 {
583 int neigh = Cells[i].Neighbours[f];

590 Inlet_Cells_List.push_back(static_cast<int>(i));
591 Inlet_Cells_List.push_back(f);

595 {
596 Exit_Cells_List.push_back(static_cast<int>(i));
597 Exit_Cells_List.push_back(f);
598 Exit_Cells_List.push_back(neigh);
599 }
600 else if (kind == FACE_KIND_TOP || kind == FACE_KIND_BOTTOM || kind == FACE_KIND_WALL)
601 {
602 Wall_Cells_List.push_back(static_cast<int>(i));
603 Wall_Cells_List.push_back(f);
604 Wall_Cells_List.push_back(neigh);

613// Helper: Given a face's normal (nx, ny), returns a priority integer:
614// 1 -> left, 2 -> bottom, 3 -> right, 4 -> top.
615// The assignment is made by comparing the normal's angle with the target angles:
616// left: π (or –π), bottom: -π/2, right: 0, top: π/2.
617int compute_face_priority(double nx, double ny)
618{
619 // Compute the angle in radians; range is (-PI, PI]
620 double angle = std::atan2(ny, nx);
621
622 // For the left face, we want to consider angles near π and -π as equivalent.
623 double diff_left = fabs(angle - M_PI);
624 if (angle < 0)
625 diff_left = std::min(diff_left, fabs(angle + M_PI));
626
627 double diff_bottom = fabs(angle - (-M_PI / 2));
628 double diff_right = fabs(angle - 0);
629 double diff_top = fabs(angle - (M_PI / 2));
630
631 // Start by assuming left has the smallest difference.
632 int priority = 1;
633 double minDiff = diff_left;
634 if (diff_bottom < minDiff)
635 {
636 minDiff = diff_bottom;
637 priority = 2;
638 }
639 if (diff_right < minDiff)
640 {
641 minDiff = diff_right;
642 priority = 3;
643 }
644 if (diff_top < minDiff)
645 { /* minDiff = diff_top; */
646 priority = 4;
647 }
648 return priority;
649}

651// Sort_Neighbours reorders each cell's Neighbours vector so that the
652// indices (or ghost marker -1) are arranged in the following order:
653// left face, bottom face, right face, top face.
654// It uses the cell's Face_Normals (assumed to be a vector of 2D vectors)
655// to decide each face's orientation.
656void Sort_Neighbours(vector<Cell> &Cells)
657{
658 for (size_t i = 0; i < Cells.size(); i++)
659 {
660 Cell &cell = Cells[i];
661 // Assume that the number of faces is stored in cell.numFaces
662 // (or, equivalently, cell.nodeIndices.size() for many cell types).
663 int nFaces = cell.numFaces;
664
665 // Build a vector of (faceIndex, priority) pairs.
666 vector<pair<int, int>> facePriority;
667 for (int f = 0; f < nFaces; f++)
668 {
669 // Get the face normal. We assume that Face_Normals[f]
670 // is a vector<double> of at least 2 values: [nx, ny].
671 double nx = cell.Face_Normals[f * 2 + 0];
672 double ny = cell.Face_Normals[f * 2 + 1];
673 int priority = compute_face_priority(nx, ny);
674 facePriority.push_back(make_pair(f, priority));
675 }
676
677 // Sort faces based on the computed priority.
678 std::sort(facePriority.begin(), facePriority.end(),
679 [](const pair<int, int> &a, const pair<int, int> &b)
680 {
681 return a.second < b.second;
682 });
683
684 // Rebuild a sorted neighbour vector. The ghost cell value (-1)
685 // stays attached to its face.
686 vector<int> sortedNeighbours(nFaces, -1);
687 for (size_t idx = 0; idx < facePriority.size(); idx++)
688 {
689 int origFaceIndex = facePriority[idx].first;
690 sortedNeighbours[idx] = cell.Neighbours[origFaceIndex];
691 }
692
693 // Update the cell's neighbour vector.
694 cell.Neighbours = sortedNeighbours;
695 }
696}
