169// Function to calculate conservative variables from given inputs
170// var: Either pressure or density
171// V: Velocity vector
172// var1: Either temperature or density
173// i: Flag to identify the type of input (1: Pressure & Temperature, 2: Pressure & Density, default: Density & Temperature)
174void Calculate_Computational_Variables(const double &var, const V_D &V, const double &var1, const int &i)
175{
176 double temp_rho = 0.0, vmag = 0.0, v1 = 0.0, v2 = 0.0, T = 0.0, P = 0.0;
177
178 Global_U[0] = 0.0;
179 Global_U[1] = 0.0;
180 Global_U[2] = 0.0;
181 Global_U[3] = 0.0;
182
183 switch (i)
184 {
185 case 1:
186 T = var1;
187 temp_rho = var / (R_GC * T); // If Pressure and Temperature are given
188 break;
189 case 2:
190 P = var;
191 temp_rho = var1; // If Pressure and Density are given
192 break;
193 default:
194 temp_rho = var;
195 T = var1;
196 P = temp_rho * R_GC * T; // If Density and Temperature are given
197 break;
198 }
199
200 v1 = V[0];
201 v2 = V[1];
202 vmag = 0.5 * (v1 * v1 + v2 * v2);
203
204 Global_U[0] = temp_rho;
205 Global_U[1] = temp_rho * v1;
206 Global_U[2] = temp_rho * v2;
207 Global_U[3] = ((P / gamma_M_1) + temp_rho * vmag);
