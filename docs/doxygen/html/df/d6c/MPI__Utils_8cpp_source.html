33void flatten_owned_state(const vector<V_D> &source, const int components, vector<double> &buffer)
34{
35 const int begin = owned_begin();
36 const int end = owned_end();
37 buffer.assign(static_cast<size_t>(end - begin) * components, 0.0);
38
39 for (int c = begin; c < end; ++c)
40 {
41 if (c >= static_cast<int>(source.size()))
42 continue;
43 for (int v = 0; v < components && v < static_cast<int>(source[c].size()); ++v)
44 buffer[static_cast<size_t>(c - begin) * components + v] = source[c][v];
45 }

48void scatter_physical_state(vector<V_D> &target, const vector<double> &buffer, const int components)
49{
50 if (target.size() < static_cast<size_t>(No_Physical_Cells))
51 return;
52
53 for (int c = 0; c < No_Physical_Cells; ++c)
54 {
55 if (target[c].size() < static_cast<size_t>(components))
56 target[c].resize(components, 0.0);
57 for (int v = 0; v < components; ++v)
58 target[c][v] = buffer[static_cast<size_t>(c) * components + v];

63void allgather_state(vector<V_D> &state, const int components)
64{
65 if (mpi_size <= 1 || No_Physical_Cells <= 0)
66 return;
67
68 vector<double> sendbuf;
69 flatten_owned_state(state, components, sendbuf);
70
71 vector<int> recvcounts(mpi_size, 0);
72 vector<int> displs(mpi_size, 0);
73 for (int r = 0; r < mpi_size; ++r)
74 {
75 const int begin = (No_Physical_Cells * r) / mpi_size;
76 const int end = (No_Physical_Cells * (r + 1)) / mpi_size;
77 recvcounts[r] = (end - begin) * components;
78 displs[r] = begin * components;
79 }
80
81 vector<double> recvbuf(static_cast<size_t>(No_Physical_Cells) * components, 0.0);
82 MPI_Allgatherv(sendbuf.data(), static_cast<int>(sendbuf.size()), MPI_DOUBLE,
83 recvbuf.data(), recvcounts.data(), displs.data(), MPI_DOUBLE,
84 MPI_COMM_WORLD);
