281Mesh make_cartesian_mesh(const Config &cfg)
282{
283 Mesh m;
284 m.nx = cfg.nx;
285 m.ny = cfg.ny;
286 m.nz = cfg.nz;
287 m.Lx = cfg.Lx;
288 m.Ly = cfg.Ly;
289 m.Lz = cfg.Lz;
290 m.n_cells = m.nx * m.ny * m.nz;
291 m.dx = cfg.Lx / m.nx;
292 m.dy = cfg.Ly / m.ny;
293 m.dz = cfg.Lz / m.nz;
294
295 m.xc.resize(m.n_cells);
296 m.yc.resize(m.n_cells);
297 m.zc.resize(m.n_cells);
298 m.volume.assign(m.n_cells, m.dx * m.dy * m.dz);
299 m.neigh.assign(m.n_cells * kFaces, -1);
300 m.nxyz.assign(m.n_cells * kFaces * 3, 0.0);
301 m.area.assign(m.n_cells * kFaces, 0.0);
302 m.wall_face.assign(m.n_cells * kFaces, 0);
303 m.bc_type.assign(m.n_cells * kFaces, BC_INTERIOR);
304
305 const int bc_face[kFaces] = {parse_bc(cfg.bc_xmin), parse_bc(cfg.bc_xmax),
306 parse_bc(cfg.bc_ymin), parse_bc(cfg.bc_ymax),
307 parse_bc(cfg.bc_zmin), parse_bc(cfg.bc_zmax)};
308
309 const double face_area[kFaces] = {m.dy * m.dz, m.dy * m.dz, m.dx * m.dz,
310 m.dx * m.dz, m.dx * m.dy, m.dx * m.dy};
311 const double nrm[kFaces][3] = {{-1, 0, 0}, {1, 0, 0}, {0, -1, 0},
312 {0, 1, 0}, {0, 0, -1}, {0, 0, 1}};
313
314 for (int k = 0; k < m.nz; ++k)
315 for (int j = 0; j < m.ny; ++j)
316 for (int i = 0; i < m.nx; ++i)
317 {
318 const int c = cid(i, j, k, m.nx, m.ny);
319 m.xc[c] = (i + 0.5) * m.dx;
320 m.yc[c] = (j + 0.5) * m.dy;
321 m.zc[c] = (k + 0.5) * m.dz;
322
323 const int ni[kFaces][3] = {{i - 1, j, k}, {i + 1, j, k}, {i, j - 1, k},
324 {i, j + 1, k}, {i, j, k - 1}, {i, j, k + 1}};
325
326 for (int f = 0; f < kFaces; ++f)
327 {
328 m.area[c * kFaces + f] = face_area[f];
329 m.nxyz[(c * kFaces + f) * 3 + 0] = nrm[f][0];
330 m.nxyz[(c * kFaces + f) * 3 + 1] = nrm[f][1];
331 m.nxyz[(c * kFaces + f) * 3 + 2] = nrm[f][2];
332
333 const int ii = ni[f][0], jj = ni[f][1], kk = ni[f][2];
334 if (ii >= 0 && ii < m.nx && jj >= 0 && jj < m.ny && kk >= 0 && kk < m.nz)
335 m.neigh[c * kFaces + f] = cid(ii, jj, kk, m.nx, m.ny);
336 else
337 {
338 m.wall_face[c * kFaces + f] = 1;
339 m.bc_type[c * kFaces + f] = bc_face[f];
340 }
341 }
342 }
343 return m;
344}

346Mesh make_mesh_for_case(const Config &cfg)
347{
348 Mesh m;
349 if (cfg.case_name == "ramp_15" || cfg.case_name == "ramp15" ||
350 cfg.case_name == "Ramp_15_Degree")
351 m = make_ramp15_mesh(cfg);
352 else if (cfg.case_name == "swbli_plate" || cfg.case_name == "SWBLI" ||
353 cfg.case_name == "flat_plate_swbli")
354 m = make_swbli_plate_mesh(cfg);
355 else if (cfg.case_name == "cavity" || cfg.case_name == "open_cavity" ||
356 cfg.case_name == "supersonic_cavity")
357 m = make_cavity_mesh(cfg);
358 else
359 m = make_cartesian_mesh(cfg);
360 m.wall_dist = m.yc;
361 if (cfg.case_name == "cavity" || cfg.case_name == "open_cavity" ||
362 cfg.case_name == "supersonic_cavity")
363 {
364 const double xs = std::min(cfg.cavity_x_start, cfg.cavity_x_end);
365 const double xe = std::max(cfg.cavity_x_start, cfg.cavity_x_end);
366 const double D = std::max(cfg.cavity_depth, 1.0e-8);
367 for (int c = 0; c < m.n_cells; ++c)
368 {
369 if (!cell_active(m, c))
370 {
371 m.wall_dist[c] = 0.0;
372 continue;
373 }
374 const double x = m.xc[c];
375 const double y = m.yc[c];
376 if (y >= 0.0 && (x < xs || x > xe))
377 m.wall_dist[c] = y; /* plate */
378 else
379 {
380 double d = y + D; /* floor */
381 d = std::min(d, std::max(x - xs, 0.0));
382 d = std::min(d, std::max(xe - x, 0.0));
383 if (y >= 0.0)
384 d = std::min(d, y); /* mouth */
385 m.wall_dist[c] = std::max(d, 0.0);
386 }
387 }
388 }
389 return m;
390}

449double compute_dt(const State &st, const Mesh &m, const Config &cfg)
450{
451 double dt_min = 1e300;
452#ifdef _OPENMP
453#pragma omp parallel for reduction(min : dt_min) schedule(static)
454#endif
455 for (int c = 0; c < m.n_cells; ++c)
456 {
457 if (!cell_active(m, c))
458 continue;
459 const double *p = &st.prim[c * 6];
460 double denom = 0.0;
461 for (int f = 0; f < kFaces; ++f)
462 {
463 const double nx = m.nxyz[(c * kFaces + f) * 3 + 0];
464 const double ny = m.nxyz[(c * kFaces + f) * 3 + 1];
465 const double nz = m.nxyz[(c * kFaces + f) * 3 + 2];
466 const double a = m.area[c * kFaces + f];
467 const double un = p[1] * nx + p[2] * ny + p[3] * nz;
468 denom += (std::fabs(un) + p[5]) * a;
469 }
470 double dt = (denom > 1e-30) ? cfg.cfl * m.volume[c] / denom : 1e300;
471 if (cfg.viscous && cfg.mu > 0.0)
472 {
473 double hx = m.dx, hy = m.dy, hz = m.dz;
474 const int i = c % m.nx;
475 const int k = c / (m.nx * m.ny);
476 const int j = (c / m.nx) % m.ny;
477 if (i + 1 < m.nx)
478 hx = std::fabs(m.xc[c + 1] - m.xc[c]);
479 else if (i > 0)
480 hx = std::fabs(m.xc[c] - m.xc[c - 1]);
481 if (j + 1 < m.ny)
482 hy = std::fabs(m.yc[c + m.nx] - m.yc[c]);
483 else if (j > 0)
484 hy = std::fabs(m.yc[c] - m.yc[c - m.nx]);
485 if (k + 1 < m.nz)
486 hz = std::fabs(m.zc[c + m.nx * m.ny] - m.zc[c]);
487 else if (k > 0)
488 hz = std::fabs(m.zc[c] - m.zc[c - m.nx * m.ny]);
489 const double h2 = std::min({hx * hx, hy * hy, hz * hz});
490 const double nu = cfg.mu / std::max(p[0], 1e-14);
491 dt = std::min(dt, 0.25 * cfg.cfl * h2 / std::max(nu, 1e-30));
492 }
493 dt_min = std::min(dt_min, dt);
494 }
495 return dt_min;
496}

498void update_explicit(State &st, const Mesh &m, const std::vector<double> &R, double dt,
499 double err[5])
500{
501 double err_acc[5] = {0, 0, 0, 0, 0};
502#ifdef _OPENMP
503#pragma omp parallel
504#endif
505 {
506 double local_err[5] = {0, 0, 0, 0, 0};
507#ifdef _OPENMP
508#pragma omp for schedule(static)
509#endif
510 for (int c = 0; c < m.n_cells; ++c)
511 {
512 if (!cell_active(m, c))
513 continue;
514 double Uold[kNv];
515 for (int v = 0; v < kNv; ++v)
516 Uold[v] = st.U[c * kNv + v];
517 for (int v = 0; v < kNv; ++v)
518 {
519 const double dU = -dt / m.volume[c] * R[c * kNv + v];
520 const double denom = std::fabs(Uold[v]) < 1e-12 ? 1.0 : std::fabs(Uold[v]);
521 local_err[v] += (dU / denom) * (dU / denom);
522 st.U[c * kNv + v] = Uold[v] + dU;
523 }
524 double prim[6];
525 fill_prim_cell(&st.U[c * kNv], prim);
526 if (!(prim[0] > 1e-12) || !(prim[4] > 1e-12))
527 {
528 for (int v = 0; v < kNv; ++v)
529 st.U[c * kNv + v] = Uold[v];
530 }
531 }
532#ifdef _OPENMP
533#pragma omp critical
534#endif
535 {
536 for (int v = 0; v < kNv; ++v)
537 err_acc[v] += local_err[v];
538 }
539 }
540 for (int v = 0; v < kNv; ++v)
541 err[v] = std::sqrt(err_acc[v]);
542 prim_from_U(st, m);
543}

595bool read_vtk_restart(const std::string &path, const Mesh &m, State &st)
596{
597 std::ifstream in(path);
598 if (!in)
599 return false;
600
601 std::string line;
602 int n_cells = -1;
603 while (std::getline(in, line))
604 {
605 std::istringstream hs(line);
606 std::string tag;
607 hs >> tag;
608 if (tag == "CELL_DATA")
609 {
610 hs >> n_cells;
611 break;
612 }
613 }
614 if (n_cells <= 0)
615 return false;
616 const int nxy = m.nx * m.ny;
617 if (nxy <= 0 || n_cells % nxy != 0 || m.n_cells % nxy != 0)
618 return false;
619 const int nz_src = n_cells / nxy;
620 const int nz_dst = m.n_cells / nxy;
621 if (nz_dst != m.nz)
622 return false;
623
624 st.U.assign(static_cast<std::size_t>(m.n_cells) * kNv, 0.0);
625 st.prim.assign(static_cast<std::size_t>(m.n_cells) * 6, 0.0);
626
627 auto skip_to = [&](const std::string &key) {
628 while (std::getline(in, line))
629 {
630 if (line.compare(0, key.size(), key) == 0)
631 return true;
632 }
633 return false;
634 };
635 auto read_n = [&](std::vector<double> &dst, int n) {
636 dst.assign(static_cast<std::size_t>(n), 0.0);
637 int got = 0;
638 while (got < n && in >> dst[static_cast<std::size_t>(got)])
639 ++got;
640 return got == n;
641 };
642
643 if (!skip_to("SCALARS Density"))
644 return false;
645 std::getline(in, line);
646 std::vector<double> rho, pressure, u, v, w;
647 if (!read_n(rho, n_cells))
648 return false;
649 if (!skip_to("SCALARS Pressure"))
650 return false;
651 std::getline(in, line);
652 if (!read_n(pressure, n_cells))
653 return false;
654 if (!skip_to("VECTORS Velocity"))
655 return false;
656 u.assign(n_cells, 0.0);
657 v.assign(n_cells, 0.0);
658 w.assign(n_cells, 0.0);
659 for (int c = 0; c < n_cells; ++c)
660 {
661 if (!(in >> u[c] >> v[c] >> w[c]))
662 return false;
663 }
664 for (int c = 0; c < m.n_cells; ++c)
665 {
666 const int k = c / nxy;
667 const int ij = c - k * nxy;
668 const int k_src = k % nz_src;
669 const int src = ij + k_src * nxy;
670 U_from_prim_vals(rho[src], u[src], v[src], w[src], pressure[src], &st.U[c * kNv]);
671 }
672 prim_from_U(st, m);
673 if (nz_src != nz_dst)
674 std::cout << "Restart: tiled z " << nz_src << " -> " << nz_dst << "\n";
675 return true;
676}

678void apply_les_trip(State &st, const Mesh &m, const Config &cfg)
679{
680 if (cfg.turb_model < 3 || cfg.les_perturb <= 0.0)
681 return;
682 std::mt19937 rng(12345);
683 std::uniform_real_distribution<double> dist(-1.0, 1.0);
684 const double amp = cfg.les_perturb * std::max(std::fabs(cfg.freestream_u), 1.0e-6);
685 for (int c = 0; c < m.n_cells; ++c)
686 {
687 if (!cell_active(m, c))
688 continue;
689 double q[6];
690 fill_prim_cell(&st.U[c * kNv], q);
691 const double wall = (c < static_cast<int>(m.wall_dist.size())) ? m.wall_dist[c] : m.yc[c];
692 const double damp = std::min(1.0, wall / 0.02);
693 const double du = amp * damp * dist(rng);
694 const double dv = amp * damp * dist(rng);
695 const double dw = amp * damp * dist(rng);
696 U_from_prim_vals(q[0], q[1] + du, q[2] + dv, q[3] + dw, q[4], &st.U[c * kNv]);
697 }
698 prim_from_U(st, m);
699 std::cout << "LES trip: " << cfg.les_perturb << " |Uinf| (damped at wall)\n";
700}

702int run_host(const Config &cfg)
703{
704 std::cout << "=== 3D HOST " << scheme_name(cfg.scheme) << "/" << order_name(cfg.order)
705 << (cfg.order == 2 ? (cfg.weno_char ? "+Char" : "+Comp") : "")
706 << (cfg.order == 2 ? (cfg.weno_z ? "+WENOZ" : "+WENOJS") : "")
707 << (cfg.order == 2
708 ? (std::string(" hybrid=") + weno_hybrid_name(cfg.weno_hybrid))
709 : "")
710 << (cfg.scheme == 1
711 ? (std::string(" ricca=") + ricca_sensor_name(cfg.ricca_sensor))
712 : "")
713 << (cfg.viscous ? " +NS" : "") << " turb=" << turb_name(cfg.turb_model)
714 << " " << cfg.nx << "x" << cfg.ny << "x" << cfg.nz
715 << " case=" << cfg.case_name << " ===\n";
716 Mesh m = make_mesh_for_case(cfg);
717 State st;
718 init_case(st, m, cfg);
719 if (!cfg.restart.empty())
720 {
721 if (!read_vtk_restart(cfg.restart, m, st))
722 throw std::runtime_error("Failed to restart from " + cfg.restart);
723 std::cout << "Restart from " << cfg.restart << "\n";
724 }
725 apply_les_trip(st, m, cfg);
726
727 std::vector<double> R;
728 double err[5];
729 for (int it = 1; it <= cfg.max_iter; ++it)
730 {
731 const double dt = compute_dt(st, m, cfg);
732 net_flux_host(st, m, cfg, R);
733 add_viscous_flux_host(st, m, cfg, R);
734 update_explicit(st, m, R, dt, err);
735 if (it % cfg.print_every == 0 || it == 1 || it == cfg.max_iter)
736 {
737 double pmin = 1e300, pmax = -1e300, mmin = 1e300, mmax = -1e300;
738 for (int c = 0; c < m.n_cells; ++c)
739 {
740 pmin = std::min(pmin, st.prim[c * 6 + 4]);
741 pmax = std::max(pmax, st.prim[c * 6 + 4]);
742 const double spd = std::sqrt(st.prim[c * 6 + 1] * st.prim[c * 6 + 1] +
743 st.prim[c * 6 + 2] * st.prim[c * 6 + 2] +
744 st.prim[c * 6 + 3] * st.prim[c * 6 + 3]);
745 const double mach = spd / std::max(st.prim[c * 6 + 5], 1e-14);
746 mmin = std::min(mmin, mach);
747 mmax = std::max(mmax, mach);
748 }
749 std::printf("%8d dt=%.3e err(rho)=%.3e p[%.3e,%.3e] M[%.3f,%.3f]\n", it, dt,
750 err[0], pmin, pmax, mmin, mmax);
751 std::fflush(stdout);
752 }
753 }
754 if (!write_vtk(cfg.vtk_out, m, st))
755 std::cerr << "Warning: failed to write VTK " << cfg.vtk_out << "\n";
756 else
757 std::cout << "Wrote " << cfg.vtk_out << "\n";
758 return 0;
759}
