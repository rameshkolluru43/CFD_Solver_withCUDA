59// Fast inverse square root approximation (Quake III algorithm)
60inline float fast_inv_sqrt(float x)
61{
62 float xhalf = 0.5f * x;
63 int i = *(int *)&x;
64 i = 0x5f3759df - (i >> 1);
65 x = *(float *)&i;
66 x = x * (1.5f - xhalf * x * x);
67 return x;
68}

80#ifdef __AVX2__
81inline void simd_cross_product(const double *a, const double *b, double *result)
82{
83 // Load vectors into AVX registers
84 __m256d va = _mm256_loadu_pd(a);
85 __m256d vb = _mm256_loadu_pd(b);
86
87 // Perform cross product using AVX operations
88 // result[0] = a[1]*b[2] - a[2]*b[1]
89 // result[1] = a[2]*b[0] - a[0]*b[2]
90 // result[2] = a[0]*b[1] - a[1]*b[0]
91
92 __m256d temp1 = _mm256_permute4x64_pd(va, 0xC9); // [a1, a2, a0, a3]
93 __m256d temp2 = _mm256_permute4x64_pd(vb, 0xD2); // [b2, b0, b1, b3]
94 __m256d temp3 = _mm256_permute4x64_pd(va, 0xD2); // [a2, a0, a1, a3]
95 __m256d temp4 = _mm256_permute4x64_pd(vb, 0xC9); // [b1, b2, b0, b3]
96
97 __m256d prod1 = _mm256_mul_pd(temp1, temp2);
98 __m256d prod2 = _mm256_mul_pd(temp3, temp4);
99 __m256d cross = _mm256_sub_pd(prod1, prod2);
100
101 _mm256_storeu_pd(result, cross);
102}
103#endif

124// Optimized face construction
125void Construct_Face_Optimized(Cell &Grid_Cell)
126{
127 const auto &vertices = Grid_Cell.Cell_Vertices;
128
129 // Extract points efficiently
130 const double *p0 = &vertices[0]; // [x0, y0, z0]
131 const double *p1 = &vertices[3]; // [x1, y1, z1]
132 const double *p2 = &vertices[6]; // [x2, y2, z2]
133 const double *p3 = &vertices[9]; // [x3, y3, z3]
134
135 // Construct faces with SIMD operations where available
136 auto construct_face_segment = [&](const double *a, const double *b)
137 {
138 double dx = b[0] - a[0];
139 double dy = b[1] - a[1];
140 double length = std::sqrt(dx * dx + dy * dy);
141
142 Grid_Cell.Face_Areas.push_back(length);
143 Grid_Cell.Face_Normals.push_back(dy / length); // nx = dy/dl
144 Grid_Cell.Face_Normals.push_back(-dx / length); // ny = -dx/dl
145 };
146
147 // Face segments: (p3,p0), (p0,p1), (p1,p2), (p2,p3)
148 construct_face_segment(p3, p0);
149 construct_face_segment(p0, p1);
150 construct_face_segment(p1, p2);
151 construct_face_segment(p2, p3);
152}

154// Optimized cell construction with move semantics
155void Construct_Cell_Optimized(Cell &&Grid_Cell)
156{
157 // Use references to avoid copies
158 auto &vertices = Grid_Cell.Cell_Vertices;
159
160 // Pre-allocate vectors
161 Grid_Cell.Cell_Center.resize(3);
162 Grid_Cell.Face_Areas.reserve(4);
163 Grid_Cell.Face_Normals.reserve(8);
164
165 // Compute centroid efficiently
166 double cx = 0.0, cy = 0.0, cz = 0.0;
167 for (size_t i = 0; i < vertices.size(); i += 3)
168 {
169 cx += vertices[i];
170 cy += vertices[i + 1];
171 cz += vertices[i + 2];
172 }
173

175 const double inv_vertices = 1.0 / num_vertices;
176 Grid_Cell.Cell_Center[0] = cx * inv_vertices;
177 Grid_Cell.Cell_Center[1] = cy * inv_vertices;
178 Grid_Cell.Cell_Center[2] = cz * inv_vertices;
179
180 // Construct faces with optimized calculations
181 Construct_Face_Optimized(Grid_Cell);
182
183 // Calculate area using optimized cross product
184 if (vertices.size() == 12)
185 { // Quadrilateral
186 double diag1_x = vertices[6] - vertices[0];
187 double diag1_y = vertices[7] - vertices[1];
188 double diag2_x = vertices[9] - vertices[3];
189 double diag2_y = vertices[10] - vertices[4];

191 // 2D cross product magnitude
192 Grid_Cell.Area = 0.5 * std::abs(diag1_x * diag2_y - diag1_y * diag2_x);

266// Fast angle calculation using atan2 approximation
267inline double fast_atan2(double y, double x)
268{
269 // Fast atan2 approximation using polynomial approximation
270 const double abs_y = std::abs(y);
271 const double abs_x = std::abs(x);
272
273 if (abs_x < 1e-10 && abs_y < 1e-10)
274 return 0.0;
275
276 double a = std::min(abs_x, abs_y) / std::max(abs_x, abs_y);
277 double s = a * a;
278 double r = ((-0.0464964749 * s + 0.15931422) * s - 0.327622764) * s * a + a;
279
280 if (abs_y > abs_x)
281 r = 1.57079637 - r;
282 if (x < 0)
283 r = 3.14159274 - r;
284 if (y < 0)
285 r = -r;
286
287 return r;
288}

290// Optimized anti-clockwise sorting using faster angle calculation
291void Sort_Points_AntiClockWise_Optimized(V_D &points, V_I &indices)
292{
293 if (points.empty() || points.size() % 3 != 0)
294 return;
295
296 GridPerformance::start_timer();
297
298 // Compute centroid
299 V_D centroid(3, 0.0);
300 Compute_Centroid_Optimized(points, centroid);
301
302 // Create point-index pairs
303 std::vector<std::pair<std::array<double, 3>, int>> point_pairs;
304 point_pairs.reserve(points.size() / 3);
305
306 for (size_t i = 0; i < points.size(); i += 3)
307 {
308 std::array<double, 3> point = {points[i], points[i + 1], points[i + 2]};
309 point_pairs.emplace_back(std::move(point), indices[i / 3]);
310 }
311
312 // Sort using optimized angle calculation
313 std::sort(point_pairs.begin(), point_pairs.end(),
314 [&centroid](const auto &a, const auto &b)
315 {
316 double angle_a = fast_atan2(a.first[1] - centroid[1], a.first[0] - centroid[0]);
317 double angle_b = fast_atan2(b.first[1] - centroid[1], b.first[0] - centroid[0]);
318 return angle_a < angle_b;
319 });
320
321 // Rebuild vectors
322 points.clear();
323 indices.clear();
324 points.reserve(point_pairs.size() * 3);
325 indices.reserve(point_pairs.size());
326
327 for (const auto &pair : point_pairs)
328 {
329 points.insert(points.end(), pair.first.begin(), pair.first.end());
330 indices.push_back(pair.second);
331 }
332
333 GridPerformance::end_timer("Optimized Anti-clockwise Sorting");
334}
