18inline void fill_prim(const double U[5], double prim[6])
19{
20 const double rho = std::max(U[0], kTiny);
21 const double u = U[1] / rho;
22 const double v = U[2] / rho;
23 const double w = U[3] / rho;
24 const double kinetic = 0.5 * rho * (u * u + v * v + w * w);
25 const double p = std::max((kGamma - 1.0) * (U[4] - kinetic), kTiny);
26 prim[0] = rho;
27 prim[1] = u;
28 prim[2] = v;
29 prim[3] = w;
30 prim[4] = p;
31 prim[5] = std::sqrt(kGamma * p / rho);
32}
33
34inline void U_from_prim(double rho, double u, double v, double w, double p, double U[5])
35{
36 rho = std::max(rho, kTiny);
37 p = std::max(p, kTiny);
38 U[0] = rho;
39 U[1] = rho * u;
40 U[2] = rho * v;
41 U[3] = rho * w;
42 U[4] = p / (kGamma - 1.0) + 0.5 * rho * (u * u + v * v + w * w);
43}

45inline void physical_flux(const double U[5], const double p[6],
46 double nx, double ny, double nz, double F[5])
47{
48 const double un = p[1] * nx + p[2] * ny + p[3] * nz;
49 F[0] = U[0] * un;
50 F[1] = U[1] * un + p[4] * nx;
51 F[2] = U[2] * un + p[4] * ny;
52 F[3] = U[3] * un + p[4] * nz;
53 F[4] = (U[4] + p[4]) * un;
54}

56inline void slip_reflect(const double U_in[5], double nx, double ny, double nz,
57 double U_out[5])
58{
59 double q[6];
60 fill_prim(U_in, q);

62 U_from_prim(q[0], q[1] - 2.0 * un * nx, q[2] - 2.0 * un * ny,

171inline void cell_ijk(int c, const Mesh &m, int &i, int &j, int &k)
172{
173 const int plane = m.nx * m.ny;
174 k = c / plane;
175 const int r = c - k * plane;
176 j = r / m.nx;
177 i = r - j * m.nx;

180inline int line_cell(int c, int f, int offset, const Mesh &m)
181{
182 int i, j, k;
183 cell_ijk(c, m, i, j, k);
184 const int sign = (f & 1) ? 1 : -1;
185 if (f < 2)
186 i += sign * offset;
187 else if (f < 4)
188 j += sign * offset;
189 else
190 k += sign * offset;
191 const int id = cell_id(i, j, k, m);
192 if (id >= 0 && !cell_active(m, id))
193 return -1;
194 return id;
195}

212inline bool admissible(const double U[5])
213{
214 if (!(U[0] > kTiny))
215 return false;
216 const double kinetic = 0.5 * (U[1] * U[1] + U[2] * U[2] + U[3] * U[3]) / U[0];
217 return (kGamma - 1.0) * (U[4] - kinetic) > kTiny;

220inline double weno5_js(double v0, double v1, double v2, double v3, double v4)
221{
222 /* Jiang–Shu WENO5 for u_{i+1/2}^- from stencil {i-2..i+2}. */
223 const double d0 = 0.1, d1 = 0.6, d2 = 0.3;
224 const double p0 = (2.0 * v0 - 7.0 * v1 + 11.0 * v2) / 6.0;
225 const double p1 = (-v1 + 5.0 * v2 + 2.0 * v3) / 6.0;
226 const double p2 = (2.0 * v2 + 5.0 * v3 - v4) / 6.0;
227 const double b0 = (13.0 / 12.0) * std::pow(v0 - 2.0 * v1 + v2, 2) +
228 0.25 * std::pow(v0 - 4.0 * v1 + 3.0 * v2, 2);
229 const double b1 = (13.0 / 12.0) * std::pow(v1 - 2.0 * v2 + v3, 2) +
230 0.25 * std::pow(v1 - v3, 2);
231 const double b2 = (13.0 / 12.0) * std::pow(v2 - 2.0 * v3 + v4, 2) +
232 0.25 * std::pow(3.0 * v2 - 4.0 * v3 + v4, 2);
233 const double eps = 1.0e-6;
234 const double a0 = d0 / ((eps + b0) * (eps + b0));
235 const double a1 = d1 / ((eps + b1) * (eps + b1));
236 const double a2 = d2 / ((eps + b2) * (eps + b2));
237 const double sum = a0 + a1 + a2;
238 if (!(sum > eps))
239 return v2;
240 return (a0 * p0 + a1 * p1 + a2 * p2) / sum;
241}

243inline double weno5_z(double v0, double v1, double v2, double v3, double v4)
244{
245 /* Borges WENO-Z for u_{i+1/2}^- from stencil {i-2..i+2}. */
246 const double d0 = 0.1, d1 = 0.6, d2 = 0.3;
247 const double p0 = (2.0 * v0 - 7.0 * v1 + 11.0 * v2) / 6.0;
248 const double p1 = (-v1 + 5.0 * v2 + 2.0 * v3) / 6.0;
249 const double p2 = (2.0 * v2 + 5.0 * v3 - v4) / 6.0;
250 const double b0 = (13.0 / 12.0) * std::pow(v0 - 2.0 * v1 + v2, 2) +
251 0.25 * std::pow(v0 - 4.0 * v1 + 3.0 * v2, 2);
252 const double b1 = (13.0 / 12.0) * std::pow(v1 - 2.0 * v2 + v3, 2) +
253 0.25 * std::pow(v1 - v3, 2);
254 const double b2 = (13.0 / 12.0) * std::pow(v2 - 2.0 * v3 + v4, 2) +
255 0.25 * std::pow(3.0 * v2 - 4.0 * v3 + v4, 2);
256 const double eps = 1.0e-40;
257 const double tau5 = std::fabs(b0 - b2);
258 const double a0 = d0 * (1.0 + std::pow(tau5 / (eps + b0), 2));
259 const double a1 = d1 * (1.0 + std::pow(tau5 / (eps + b1), 2));
260 const double a2 = d2 * (1.0 + std::pow(tau5 / (eps + b2), 2));
261 const double sum = a0 + a1 + a2;
262 if (!(sum > 0.0) || !std::isfinite(sum))
263 return v2;
264 const double out = (a0 * p0 + a1 * p1 + a2 * p2) / sum;
265 return std::isfinite(out) ? out : v2;
266}

274inline void build_char_matrices(const double qL[6], const double qR[6], double nx, double ny,
275 double nz, double L[25], double R[25])
276{
277 const double rL = std::sqrt(std::max(qL[0], kTiny));
278 const double rR = std::sqrt(std::max(qR[0], kTiny));
279 const double den = std::max(rL + rR, kTiny);
280 const double u = (rL * qL[1] + rR * qR[1]) / den;
281 const double v = (rL * qL[2] + rR * qR[2]) / den;
282 const double w = (rL * qL[3] + rR * qR[3]) / den;
283 const double a = std::max((rL * qL[5] + rR * qR[5]) / den, kTiny);
284 const double un = u * nx + v * ny + w * nz;
285 const double vel2 = u * u + v * v + w * w;
286 const double ek = 0.5 * vel2;
287 const double H = a * a / (kGamma - 1.0) + ek;
288 const double gm1 = kGamma - 1.0;
289 const double a2 = a * a;
290 const double t1 = 0.5 / a2;
291 const double t2 = gm1 * t1;
292
293 double tx, ty, tz;
294 if (std::fabs(nx) > std::fabs(nz))
295 {
296 const double inv = 1.0 / std::sqrt(nx * nx + ny * ny + 1e-30);
297 tx = -ny * inv;
298 ty = nx * inv;
299 tz = 0.0;
300 }
301 else
302 {
303 const double inv = 1.0 / std::sqrt(ny * ny + nz * nz + 1e-30);
304 tx = 0.0;
305 ty = -nz * inv;
306 tz = ny * inv;
307 }
308 const double sx = ny * tz - nz * ty;
309 const double sy = nz * tx - nx * tz;
310 const double sz = nx * ty - ny * tx;
311 const double ut = u * tx + v * ty + w * tz;
312 const double us = u * sx + v * sy + w * sz;
313
314 /* Columns of R: acoustic-, entropy, shear-t, shear-s, acoustic+ */
315 const double cols[5][5] = {
316 {1.0, u - a * nx, v - a * ny, w - a * nz, H - a * un},
317 {1.0, u, v, w, ek},
318 {0.0, tx, ty, tz, ut},
319 {0.0, sx, sy, sz, us},
320 {1.0, u + a * nx, v + a * ny, w + a * nz, H + a * un}};
321 for (int col = 0; col < 5; ++col)
322 for (int row = 0; row < 5; ++row)
323 R[row * 5 + col] = cols[col][row];
324
325 /* Rows of L */
326 L[0 * 5 + 0] = t2 * ek + t1 * a * un;
327 L[0 * 5 + 1] = -t2 * u - t1 * a * nx;
328 L[0 * 5 + 2] = -t2 * v - t1 * a * ny;
329 L[0 * 5 + 3] = -t2 * w - t1 * a * nz;
330 L[0 * 5 + 4] = t2;
331
332 L[1 * 5 + 0] = 1.0 - 2.0 * t2 * ek;
333 L[1 * 5 + 1] = 2.0 * t2 * u;
334 L[1 * 5 + 2] = 2.0 * t2 * v;
335 L[1 * 5 + 3] = 2.0 * t2 * w;
336 L[1 * 5 + 4] = -2.0 * t2;
337
338 L[2 * 5 + 0] = -ut;
339 L[2 * 5 + 1] = tx;
340 L[2 * 5 + 2] = ty;
341 L[2 * 5 + 3] = tz;
342 L[2 * 5 + 4] = 0.0;
343
344 L[3 * 5 + 0] = -us;
345 L[3 * 5 + 1] = sx;
346 L[3 * 5 + 2] = sy;
347 L[3 * 5 + 3] = sz;
348 L[3 * 5 + 4] = 0.0;
349
350 L[4 * 5 + 0] = t2 * ek - t1 * a * un;
351 L[4 * 5 + 1] = -t2 * u + t1 * a * nx;
352 L[4 * 5 + 2] = -t2 * v + t1 * a * ny;
353 L[4 * 5 + 3] = -t2 * w + t1 * a * nz;
354 L[4 * 5 + 4] = t2;
355}

367inline void weno5_face_states(const State &st, const Mesh &m, const Config &cfg, int c, int f,
368 int nb, double UL[5], double UR[5])
369{
370 int ids[6];
371 for (int s = -2; s <= 3; ++s)
372 {
373 ids[s + 2] = line_cell(c, f, s, m);
374 if (ids[s + 2] < 0)
375 return;
376 }
377
378 double Usten[6][5];
379 for (int s = 0; s < 6; ++s)
380 for (int v = 0; v < 5; ++v)
381 Usten[s][v] = st.U[ids[s] * 5 + v];
382
383 const bool use_z = cfg.weno_z;
384 auto recon_LR = [&](const double sten[6][5], double outL[5], double outR[5]) {
385 for (int v = 0; v < 5; ++v)
386 {
387 outL[v] = weno5_scalar(sten[0][v], sten[1][v], sten[2][v], sten[3][v], sten[4][v],
388 use_z);
389 /* Right state: mirror stencil into the same left-biased operator. */
390 outR[v] = weno5_scalar(sten[5][v], sten[4][v], sten[3][v], sten[2][v], sten[1][v],
391 use_z);
392 }
393 };
394
395 if (!cfg.weno_char)
396 {
397 recon_LR(Usten, UL, UR);
398 }
399 else
400 {
401 double qL[6], qR[6], Lmat[25], Rmat[25];
402 fill_prim(Usten[2], qL);
403 fill_prim(Usten[3], qR);
404 const int ni = (c * kFaces + f) * 3;
405 build_char_matrices(qL, qR, m.nxyz[ni], m.nxyz[ni + 1], m.nxyz[ni + 2], Lmat, Rmat);
406
407 double Wsten[6][5];
408 for (int s = 0; s < 6; ++s)
409 mat5_vec(Lmat, Usten[s], Wsten[s]);
410
411 double WL[5], WR[5];
412 recon_LR(Wsten, WL, WR);
413 mat5_vec(Rmat, WL, UL);
414 mat5_vec(Rmat, WR, UR);
415 }
416
417 if (!admissible(UL) || !admissible(UR) || !std::isfinite(UL[0]) || !std::isfinite(UR[0]))
418 {
419 std::copy(&st.U[c * 5], &st.U[c * 5 + 5], UL);
420 std::copy(&st.U[nb * 5], &st.U[nb * 5 + 5], UR);
421 return;
422 }

424 /* Paper WENO hybrid: blend toward FO near discontinuities.
425 Indicators are face-based (reconstructed L/R). Combo uses shock
426 (p, u_n) always; density/entropy only when φ_P is small (contacts),
427 so post-shock entropy noise does not freeze the whole field to FO. */
428 if (cfg.weno_hybrid > 0)
429 {
430 const double *UfoL = Usten[2];
431 const double *UfoR = Usten[3];
432 double qL[6], qR[6];
433 fill_prim(UL, qL);
434 fill_prim(UR, qR);
435 const int ni = (c * kFaces + f) * 3;
436 const double nx = m.nxyz[ni], ny = m.nxyz[ni + 1], nz = m.nxyz[ni + 2];
437 const double unL = qL[1] * nx + qL[2] * ny + qL[3] * nz;
438 const double unR = qR[1] * nx + qR[2] * ny + qR[3] * nz;
439 const double a_ref = 0.5 * (qL[5] + qR[5]) + 1.0;
440 const double phi_P =
441 std::fabs(qR[4] - qL[4]) / (0.5 * (std::fabs(qL[4]) + std::fabs(qR[4])) + 1.0);
442 const double phi_rho =
443 std::fabs(qR[0] - qL[0]) / (0.5 * (std::fabs(qL[0]) + std::fabs(qR[0])) + 1.0);
444 const double sL = qL[4] / std::pow(std::max(qL[0], kTiny), kGamma);
445 const double sR = qR[4] / std::pow(std::max(qR[0], kTiny), kGamma);
446 const double phi_S =
447 std::fabs(sR - sL) / (0.5 * (std::fabs(sL) + std::fabs(sR)) + 1.0);
448 const double phi_u = std::fabs(unR - unL) / a_ref;
449 double phi = phi_P;
450 if (cfg.weno_hybrid == 2)
451 {
452 phi = std::max(phi_P, phi_u);
453 if (phi_P < 0.05)
454 phi = std::max({phi, phi_rho, phi_S});
455 }
456
457 for (int v = 0; v < 5; ++v)
458 {
459 const double C = (v >= 1 && v <= 3) ? 120.0 : 50.0;
460 const double theta = 1.0 / (1.0 + C * phi);
461 UL[v] = theta * UL[v] + (1.0 - theta) * UfoL[v];
462 UR[v] = theta * UR[v] + (1.0 - theta) * UfoR[v];
463 const double lo = std::min({Usten[1][v], Usten[2][v], Usten[3][v], Usten[4][v]});
464 const double hi = std::max({Usten[1][v], Usten[2][v], Usten[3][v], Usten[4][v]});
465 UL[v] = std::min(hi, std::max(lo, UL[v]));
466 UR[v] = std::min(hi, std::max(lo, UR[v]));
467 }
468 if (!admissible(UL) || !admissible(UR))
469 {
470 std::copy(UfoL, UfoL + 5, UL);
471 std::copy(UfoR, UfoR + 5, UR);
472 }
473 }
474}

476void face_flux_llf(const double UL[5], const double UR[5],
477 double nx, double ny, double nz, double area, double flux[5])
478{

481 fill_prim(UR, qR);
482 physical_flux(UL, qL, nx, ny, nz, FL);
483 physical_flux(UR, qR, nx, ny, nz, FR);
484 const double unL = qL[1] * nx + qL[2] * ny + qL[3] * nz;
485 const double unR = qR[1] * nx + qR[2] * ny + qR[3] * nz;
486 const double smax = std::max(std::fabs(unL) + qL[5], std::fabs(unR) + qR[5]);
487 for (int v = 0; v < 5; ++v)
488 flux[v] = 0.5 * (FL[v] + FR[v] - smax * (UR[v] - UL[v])) * area;
489}

492inline double ricca_phi_rh(const double qL[6], const double qR[6], double unL, double unR)
493{
494 const double massL = qL[0] * unL;
495 const double massR = qR[0] * unR;
496 const double FnL = massL * unL + qL[4];
497 const double FnR = massR * unR + qR[4];
498 const double dr = qR[0] - qL[0];
499 const double s = (std::fabs(dr) > kTiny) ? (massR - massL) / dr : 0.5 * (unL + unR);
500 const double resid = std::fabs(FnR - FnL - s * (massR - massL));
501 return resid / (0.5 * (std::fabs(FnL) + std::fabs(FnR)) + 1.0);
502}

513void face_flux_ricca(const double UL[5], const double UR[5],
514 double nx, double ny, double nz, double area, double flux[5],
515 int sensor_type, double rh_threshold)
516{
517 double qL[6], qR[6], FL[5], FR[5];
518 fill_prim(UL, qL);
519 fill_prim(UR, qR);
520 physical_flux(UL, qL, nx, ny, nz, FL);
521 physical_flux(UR, qR, nx, ny, nz, FR);
522 const double unL = qL[1] * nx + qL[2] * ny + qL[3] * nz;
523 const double unR = qR[1] * nx + qR[2] * ny + qR[3] * nz;
524 const double a_contact = std::max(std::fabs(unL), std::fabs(unR));
525 const double a_acoustic =
526 std::max(std::fabs(unL) + qL[5], std::fabs(unR) + qR[5]);
527 const double phi_P =
528 std::fabs(qR[4] - qL[4]) / (0.5 * (std::fabs(qL[4]) + std::fabs(qR[4])) + 1.0);
529
530 bool use_acoustic = false;
531 if (sensor_type == 1)
532 {
533 const double phi_RH = ricca_phi_rh(qL, qR, unL, unR);
534 use_acoustic = (phi_P > 1.0e-8) || (phi_RH > rh_threshold);
535 }
536 else
537 {
538 use_acoustic = (phi_P > 1.0e-8);
539 }
540
541 const double alpha = use_acoustic ? a_acoustic : a_contact;
542 for (int v = 0; v < 5; ++v)
543 flux[v] = 0.5 * (FL[v] + FR[v]) * area - 0.5 * alpha * area * (UR[v] - UL[v]);
544}
545
546inline double entropy_abs(double lambda, double delta)
547{
548 const double a = std::fabs(lambda);
549 return a < delta ? 0.5 * (a * a / delta + delta) : a;
550}

552void face_flux_roe(const double UL[5], const double UR[5],
553 double nx, double ny, double nz, double area, double flux[5])
554{
555 double qL[6], qR[6], FL[5], FR[5];
556 fill_prim(UL, qL);
557 fill_prim(UR, qR);
558 physical_flux(UL, qL, nx, ny, nz, FL);
559 physical_flux(UR, qR, nx, ny, nz, FR);
560
561 double tx, ty, tz;
562 if (std::fabs(nx) > std::fabs(nz))
563 {
564 const double inv = 1.0 / std::sqrt(nx * nx + ny * ny);
565 tx = -ny * inv;
566 ty = nx * inv;
567 tz = 0.0;
568 }
569 else
570 {
571 const double inv = 1.0 / std::sqrt(ny * ny + nz * nz);
572 tx = 0.0;
573 ty = -nz * inv;
574 tz = ny * inv;
575 }
576 const double sx = ny * tz - nz * ty;
577 const double sy = nz * tx - nx * tz;
578 const double sz = nx * ty - ny * tx;
579
580 const double rL = std::sqrt(qL[0]), rR = std::sqrt(qR[0]);
581 const double den = rL + rR;
582 const double u = (rL * qL[1] + rR * qR[1]) / den;
583 const double v = (rL * qL[2] + rR * qR[2]) / den;
584 const double w = (rL * qL[3] + rR * qR[3]) / den;
585 const double HL = (UL[4] + qL[4]) / qL[0];
586 const double HR = (UR[4] + qR[4]) / qR[0];
587 const double H = (rL * HL + rR * HR) / den;
588 const double vel2 = u * u + v * v + w * w;
589 const double a = std::sqrt(std::max((kGamma - 1.0) * (H - 0.5 * vel2), kTiny));
590 const double rho = rL * rR;
591 const double un = u * nx + v * ny + w * nz;
592 const double ut = u * tx + v * ty + w * tz;
593 const double us = u * sx + v * sy + w * sz;
594
595 const double dr = qR[0] - qL[0];
596 const double dun = (qR[1] - qL[1]) * nx + (qR[2] - qL[2]) * ny +
597 (qR[3] - qL[3]) * nz;
598 const double dut = (qR[1] - qL[1]) * tx + (qR[2] - qL[2]) * ty +
599 (qR[3] - qL[3]) * tz;
600 const double dus = (qR[1] - qL[1]) * sx + (qR[2] - qL[2]) * sy +
601 (qR[3] - qL[3]) * sz;
602 const double dp = qR[4] - qL[4];
603 const double a2 = a * a;
604 const double alpha[5] = {
605 (dp - rho * a * dun) / (2.0 * a2),
606 dr - dp / a2,
607 rho * dut,
608 rho * dus,
609 (dp + rho * a * dun) / (2.0 * a2)};
610 const double delta = std::max(0.1 * a, kTiny);
611 const double lambda[5] = {
612 entropy_abs(un - a, delta), entropy_abs(un, delta), entropy_abs(un, delta),
613 entropy_abs(un, delta), entropy_abs(un + a, delta)};
614 const double wave[5][5] = {
615 {1.0, u - a * nx, v - a * ny, w - a * nz, H - a * un},
616 {1.0, u, v, w, 0.5 * vel2},
617 {0.0, tx, ty, tz, ut},
618 {0.0, sx, sy, sz, us},
619 {1.0, u + a * nx, v + a * ny, w + a * nz, H + a * un}};
620 for (int n = 0; n < 5; ++n)
621 {
622 double diss = 0.0;
623 for (int k = 0; k < 5; ++k)
624 diss += lambda[k] * alpha[k] * wave[k][n];
625 flux[n] = (0.5 * (FL[n] + FR[n]) - 0.5 * diss) * area;
626 }
627}

629void vanleer_split(const double U[5], const double q[6],
630 double nx, double ny, double nz, bool plus, double F[5])
631{
632 const double un = q[1] * nx + q[2] * ny + q[3] * nz;
633 const double M = un / q[5];
634 const bool all = plus ? (M >= 1.0) : (M <= -1.0);
635 const bool none = plus ? (M <= -1.0) : (M >= 1.0);
636 if (all)
637 {
638 physical_flux(U, q, nx, ny, nz, F);
639 return;
640 }
641 if (none)
642 {
643 std::fill(F, F + 5, 0.0);
644 return;
645 }
646 const double s = plus ? 1.0 : -1.0;
647 const double mass = s * 0.25 * q[0] * q[5] * (M + s) * (M + s);
648 const double ps = 0.25 * q[4] * (M + s) * (M + s) * (2.0 - s * M);
649 const double vt2 = q[1] * q[1] + q[2] * q[2] + q[3] * q[3] - un * un;
650 const double hs = std::pow((kGamma - 1.0) * un + s * 2.0 * q[5], 2) /
651 (2.0 * (kGamma * kGamma - 1.0)) +
652 0.5 * vt2;
653 F[0] = mass;
654 F[1] = mass * q[1] + ps * nx;
655 F[2] = mass * q[2] + ps * ny;
656 F[3] = mass * q[3] + ps * nz;
657 F[4] = mass * hs;
658}

660void face_flux_vanleer(const double UL[5], const double UR[5],
661 double nx, double ny, double nz, double area, double flux[5])
662{
663 double qL[6], qR[6], Fp[5], Fm[5];
664 fill_prim(UL, qL);
665 fill_prim(UR, qR);
666 vanleer_split(UL, qL, nx, ny, nz, true, Fp);
667 vanleer_split(UR, qR, nx, ny, nz, false, Fm);
668 for (int v = 0; v < 5; ++v)
669 flux[v] = (Fp[v] + Fm[v]) * area;
670}

684void get_LR_states(const State &st, const Mesh &m, const Config &cfg,
685 int c, int f, double UL[5], double UR[5])
686{
687 std::copy(&st.U[c * 5], &st.U[c * 5 + 5], UL);
688 const int nb = m.neigh[c * kFaces + f];
689 const int ni = (c * kFaces + f) * 3;
690 const double nx = m.nxyz[ni], ny = m.nxyz[ni + 1], nz = m.nxyz[ni + 2];
691 if (nb < 0)
692 {
693 boundary_ghost(UL, m, cfg, c, f, nx, ny, nz, UR);
694 return;
695 }
696 std::copy(&st.U[nb * 5], &st.U[nb * 5 + 5], UR);
697
698 const int order = cfg_order(cfg, 0);
699 if (order == 1)
700 {
701 const int cm1 = line_cell(c, f, -1, m);
702 const int cp1 = line_cell(c, f, 1, m);
703 const int cp2 = line_cell(c, f, 2, m);
704 if (cm1 < 0 || cp1 < 0 || cp2 < 0)
705 return;
706 double qm1[6], q0[6], q1[6], q2[6], qL[5], qR[5];
707 primitive_at(st, cm1, qm1);
708 primitive_at(st, c, q0);
709 primitive_at(st, cp1, q1);
710 primitive_at(st, cp2, q2);
711 for (int v = 0; v < 5; ++v)
712 {
713 qL[v] = q0[v] + 0.5 * minmod(q0[v] - qm1[v], q1[v] - q0[v]);
714 qR[v] = q1[v] - 0.5 * minmod(q1[v] - q0[v], q2[v] - q1[v]);
715 }
716 if (qL[0] <= kTiny || qL[4] <= kTiny || qR[0] <= kTiny || qR[4] <= kTiny)
717 return;
718 U_from_prim(qL[0], qL[1], qL[2], qL[3], qL[4], UL);
719 U_from_prim(qR[0], qR[1], qR[2], qR[3], qR[4], UR);
720 }
721 else if (order == 2)
722 {
723 weno5_face_states(st, m, cfg, c, f, nb, UL, UR);
724 }
725}

749void net_flux_host(const State &st, const Mesh &m, const Config &cfg, std::vector<double> &R)
750{
751 R.assign(m.n_cells * kNv, 0.0);
752#ifdef _OPENMP

767 for (int v = 0; v < kNv; ++v)
768 R[c * kNv + v] += flux[v];
769 }
770 }
