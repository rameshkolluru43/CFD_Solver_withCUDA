23inline double primitive_value(const State &st, int cell, int variable)
24{
25 if (variable < 3)
26 return st.prim[static_cast<std::size_t>(cell) * 6 + variable + 1];
27
28 const double rho =
29 std::max(st.prim[static_cast<std::size_t>(cell) * 6], 1.0e-14);
30 const double pressure =
31 st.prim[static_cast<std::size_t>(cell) * 6 + 4];
32 return pressure / rho; // R_gas = 1
33}

35inline double wall_value(int variable, double cell_value)
36{
37 // Stationary no-slip wall for velocity; adiabatic wall for temperature.
38 return variable < 3 ? 0.0 : cell_value;

41inline double neighbour_distance(const Mesh &m, int cell, int neighbour, int face,
42 double fallback)
43{
44 if (neighbour < 0)
45 return fallback;
46 if (face < 2)
47 return std::fabs(m.xc[neighbour] - m.xc[cell]);
48 if (face < 4)
49 return std::fabs(m.yc[neighbour] - m.yc[cell]);
50 return std::fabs(m.zc[neighbour] - m.zc[cell]);

53inline double directional_gradient(const State &st, const Mesh &m, int cell,
54 int variable, int negative_face,
55 int positive_face, double spacing)
56{
57 const int negative = m.neigh[cell * kFaces + negative_face];
58 const int positive = m.neigh[cell * kFaces + positive_face];
59
60 auto bc_of = [&](int face) {
61 const int idx = cell * kFaces + face;
62 return idx < static_cast<int>(m.bc_type.size()) ? m.bc_type[idx] : BC_EXTRAP;
63 };
64
65 if (negative >= 0 && positive >= 0)
66 {
67 const double dn = neighbour_distance(m, cell, negative, negative_face, spacing);
68 const double dp = neighbour_distance(m, cell, positive, positive_face, spacing);
69 return (primitive_value(st, positive, variable) -
70 primitive_value(st, negative, variable)) /
71 std::max(dn + dp, 1.0e-14);
72 }
73
74 const double center = primitive_value(st, cell, variable);
75 if (negative < 0)
76 {
77 if (bc_of(negative_face) == BC_NOSLIP)
78 {
79 const double boundary = wall_value(variable, center);
80 const double h = (positive >= 0)
81 ? neighbour_distance(m, cell, positive, positive_face, spacing)
82 : spacing;
83 return (center - boundary) / std::max(0.5 * h, 1.0e-14);
84 }
85 if (positive >= 0)
86 return (primitive_value(st, positive, variable) - center) /
87 std::max(neighbour_distance(m, cell, positive, positive_face, spacing),
88 1.0e-14);
89 return 0.0;
90 }
91
92 if (bc_of(positive_face) == BC_NOSLIP)
93 {
94 const double boundary = wall_value(variable, center);
95 const double h = neighbour_distance(m, cell, negative, negative_face, spacing);
96 return (boundary - center) / std::max(0.5 * h, 1.0e-14);
97 }
98 return (center - primitive_value(st, negative, variable)) /
99 std::max(neighbour_distance(m, cell, negative, negative_face, spacing), 1.0e-14);
100}

102std::vector<Gradient> compute_cell_gradients(const State &st, const Mesh &m)
103{
104 std::vector<Gradient> gradients(static_cast<std::size_t>(m.n_cells));
105
106 for (int cell = 0; cell < m.n_cells; ++cell)
107 {
108 if (!cell_active(m, cell))
109 continue;
110 for (int variable = 0; variable < kNGrad; ++variable)
111 {
112 gradients[cell][variable * kNDim] =
113 directional_gradient(st, m, cell, variable, 0, 1, m.dx);
114 gradients[cell][variable * kNDim + 1] =
115 directional_gradient(st, m, cell, variable, 2, 3, m.dy);
116 gradients[cell][variable * kNDim + 2] =
117 directional_gradient(st, m, cell, variable, 4, 5, m.dz);
118 }
119 }
120
121 return gradients;
122}

126void add_viscous_flux_host(const State &st, const Mesh &m, const Config &cfg,
127 std::vector<double> &R)
128{
129 if (!cfg.viscous)
130 return;
131
132 const std::size_t residual_size =
133 static_cast<std::size_t>(m.n_cells) * kNv;
134 if (R.size() < residual_size)
135 R.resize(residual_size, 0.0);
136
137 const std::vector<Gradient> gradients = compute_cell_gradients(st, m);
138 const double mu_lam = cfg.mu;
139 const double cp = kGamma / (kGamma - 1.0);
140 std::vector<double> mu_sgs(static_cast<std::size_t>(m.n_cells), 0.0);
141 if (cfg.turb_model >= 2)
142 {
143 for (int cell = 0; cell < m.n_cells; ++cell)
144 {
145 if (!cell_active(m, cell))
146 continue;
147 double g[12];
148 for (int i = 0; i < 12; ++i)
149 g[i] = gradients[cell][static_cast<std::size_t>(i)];
150 const double rho = std::max(st.prim[static_cast<std::size_t>(cell) * 6], 1.0e-14);
151 const double delta = std::cbrt(std::max(m.volume[cell], 1.0e-30));
152 const double yw =
153 cell < static_cast<int>(m.wall_dist.size()) ? m.wall_dist[cell] : m.yc[cell];
154 mu_sgs[cell] = mu_sgs_from_grad(cfg.turb_model, rho, g, delta, yw, mu_lam, cfg);
155 }
156 }
157
158 for (int cell = 0; cell < m.n_cells; ++cell)
159 {
160 if (!cell_active(m, cell))
161 continue;
162 for (int face = 0; face < kFaces; ++face)
163 {
164 const int face_index = cell * kFaces + face;
165 const int neighbour = m.neigh[face_index];
166 const bool wall = neighbour < 0;
167
168 Gradient face_gradient = gradients[cell];
169 if (!wall)
170 {
171 for (int component = 0; component < kNGrad * kNDim; ++component)
172 face_gradient[component] =
173 0.5 * (gradients[cell][component] +
174 gradients[neighbour][component]);
175 }
176
177 const double ux = face_gradient[0], uy = face_gradient[1];
178 const double uz = face_gradient[2];
179 const double vx = face_gradient[3], vy = face_gradient[4];
180 const double vz = face_gradient[5];
181 const double wx = face_gradient[6], wy = face_gradient[7];
182 const double wz = face_gradient[8];
183
184 const double mu_t_f =
185 wall ? mu_sgs[cell]
186 : 0.5 * (mu_sgs[cell] + mu_sgs[static_cast<std::size_t>(neighbour)]);
187 const double mu = mu_lam + mu_t_f;
188 const double thermal_conductivity =
189 (cfg.Pr > 0.0 ? mu_lam * cp / cfg.Pr : 0.0) +
190 (cfg.Pr_t > 0.0 ? mu_t_f * cp / cfg.Pr_t : 0.0);
191 const double two_thirds_mu = (2.0 / 3.0) * mu;
192 const double tau_xx = two_thirds_mu * (2.0 * ux - vy - wz);
193 const double tau_yy = two_thirds_mu * (2.0 * vy - ux - wz);
194 const double tau_zz = two_thirds_mu * (2.0 * wz - ux - vy);
195 const double tau_xy = mu * (uy + vx);
196 const double tau_xz = mu * (uz + wx);
197 const double tau_yz = mu * (vz + wy);
198
199 const double qx = -thermal_conductivity * face_gradient[9];
200 const double qy = -thermal_conductivity * face_gradient[10];
201 const double qz = -thermal_conductivity * face_gradient[11];
202
203 double u = primitive_value(st, cell, 0);
204 double v = primitive_value(st, cell, 1);
205 double w = primitive_value(st, cell, 2);
206 if (wall)
207 {
208 u = 0.0;
209 v = 0.0;
210 w = 0.0;
211 }
212 else
213 {
214 u = 0.5 * (u + primitive_value(st, neighbour, 0));
215 v = 0.5 * (v + primitive_value(st, neighbour, 1));
216 w = 0.5 * (w + primitive_value(st, neighbour, 2));
217 }
218
219 const double nx = m.nxyz[face_index * 3];
220 const double ny = m.nxyz[face_index * 3 + 1];
221 const double nz = m.nxyz[face_index * 3 + 2];
222 const double area = m.area[face_index];
223
224 const double traction_x =
225 tau_xx * nx + tau_xy * ny + tau_xz * nz;
226 const double traction_y =
227 tau_xy * nx + tau_yy * ny + tau_yz * nz;
228 const double traction_z =
229 tau_xz * nx + tau_yz * ny + tau_zz * nz;
230 const double energy_flux =
231 u * traction_x + v * traction_y + w * traction_z -
232 (qx * nx + qy * ny + qz * nz);
233
234 // R stores outward inviscid flux. Navier--Stokes contributes
235 // minus the outward viscous flux.
236 R[static_cast<std::size_t>(cell) * kNv + 1] -= traction_x * area;
237 R[static_cast<std::size_t>(cell) * kNv + 2] -= traction_y * area;
238 R[static_cast<std::size_t>(cell) * kNv + 3] -= traction_z * area;
239 R[static_cast<std::size_t>(cell) * kNv + 4] -= energy_flux * area;
240 }
241 }
