19void ensure_parent_dirs(const std::string &path)
20{
21 const std::string::size_type pos = path.find_last_of('/');
22 if (pos == std::string::npos)
23 return;
24
25 const std::string dir = path.substr(0, pos);
26 std::string current;
27 for (std::size_t i = 0; i < dir.size(); ++i)
28 {
29 current.push_back(dir[i]);
30 if (dir[i] == '/' || i + 1 == dir.size())
31 {
32 if (!current.empty() && current != "/")
33 mkdir(current.c_str(), 0755);
34 }
35 }
36}

38void print_progress(int iteration, double dt, const euler3d::State &state,
39 const euler3d::Mesh &mesh)
40{
41 double pmin = 1.0e300;
42 double pmax = -1.0e300;
43 for (int c = 0; c < mesh.n_cells; ++c)
44 {
45 if (!euler3d::cell_active(mesh, c))
46 continue;
47 const double p = state.prim[c * 6 + 4];
48 pmin = std::min(pmin, p);
49 pmax = std::max(pmax, p);
50 }
51 std::printf("%8d dt=%.3e p[min,max]=[%.3e,%.3e]\n",
52 iteration, dt, pmin, pmax);
53 std::fflush(stdout);
54}

61int run_cuda(const Config &cfg)
62{
63 std::cout << "=== 3D CUDA " << scheme_name(cfg.scheme) << "/" << order_name(cfg.order)
64 << (cfg.order == 2 ? (cfg.weno_char ? "+Char" : "+Comp") : "")
65 << (cfg.order == 2 ? (cfg.weno_z ? "+WENOZ" : "+WENOJS") : "")
66 << (cfg.order == 2
67 ? (std::string(" hybrid=") + weno_hybrid_name(cfg.weno_hybrid))
68 : "")
69 << (cfg.scheme == 1
70 ? (std::string(" ricca=") + ricca_sensor_name(cfg.ricca_sensor))
71 : "")
72 << (cfg.viscous ? " +NS" : "") << " turb=" << turb_name(cfg.turb_model)
73 << " " << cfg.nx << "x" << cfg.ny << "x"
74 << cfg.nz << " (resident GPU) ===\n";
75 std::fflush(stdout);
76
77 Mesh mesh = make_mesh_for_case(cfg);
78 State state;
79 init_case(state, mesh, cfg);
80 if (!cfg.restart.empty())
81 {
82 if (!read_vtk_restart(cfg.restart, mesh, state))
83 throw std::runtime_error("Failed to restart from " + cfg.restart);
84 std::cout << "Restart from " << cfg.restart << "\n";
85 std::fflush(stdout);
86 }
87 apply_les_trip(state, mesh, cfg);
88
89 euler3d_cuda::DeviceBundle device;
90 euler3d_cuda::allocate(device, mesh.n_cells);
91 euler3d_cuda::set_grid_size(device, mesh.nx, mesh.ny, mesh.nz);
92 euler3d_cuda::upload_mesh(device, mesh.volume, mesh.neigh, mesh.nxyz, mesh.area,
93 mesh.bc_type, mesh.xc, mesh.yc, mesh.zc, mesh.wall_dist,
94 mesh.active, mesh.dx, mesh.dy, mesh.dz);
95 euler3d_cuda::set_freestream(device, cfg.freestream_rho, cfg.freestream_u,
96 cfg.freestream_v, cfg.freestream_w, cfg.freestream_p);
97 euler3d_cuda::set_postshock(device, cfg.post_rho, cfg.post_u, cfg.post_v,
98 cfg.post_w, cfg.post_p);
99 euler3d_cuda::upload_U(device, state.U);
100
101 for (int iteration = 1; iteration <= cfg.max_iter; ++iteration)
102 {
103 const double dt = euler3d_cuda::launch_min_dt(device, cfg.cfl, cfg.viscous, cfg.mu);
104 euler3d_cuda::launch_net_flux(device, cfg.scheme, cfg.order,
105 cfg.weno_char, cfg.weno_z,
106 cfg.ricca_sensor, cfg.ricca_rh_threshold,
107 cfg.weno_hybrid);
108 if (cfg.viscous)
109 euler3d_cuda::launch_viscous_flux(device, cfg.mu, cfg.Pr, cfg.turb_model,
110 cfg.Cs, cfg.Cw, cfg.Cv, cfg.Pr_t,
111 cfg.kappa_von, cfg.A_plus,
112 cfg.freestream_u);
113
114 euler3d_cuda::launch_update(device, dt);
115 const bool report = iteration == 1 || iteration == cfg.max_iter ||
116 (cfg.print_every > 0 && iteration % cfg.print_every == 0);
117 if (report)
118 {
119 euler3d_cuda::download_U(device, state.U);
120 prim_from_U(state, mesh);
121 print_progress(iteration, dt, state, mesh);
122 }
123 }
124
125 euler3d_cuda::download_U(device, state.U);
126 prim_from_U(state, mesh);
127 euler3d_cuda::release(device);
128
129 if (!write_vtk(cfg.vtk_out, mesh, state))
130 std::cerr << "Warning: failed to write VTK " << cfg.vtk_out << "\n";
131 else
132 std::cout << "Wrote " << cfg.vtk_out << "\n";
133 return 0;
134}
