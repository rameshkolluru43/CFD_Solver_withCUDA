22__global__ void viscous_wall_ghost_kernel(
23 double *U,
24 const int *wall_list,
25 int n_triplets,
26 double Tw,
27 double rgas,
28 double gamma)
29{
30 const int idx = blockIdx.x * blockDim.x + threadIdx.x;
31 if (idx >= n_triplets)
32 return;
33
34 const int cell = wall_list[idx * 3 + 0];
35 const int ghost = wall_list[idx * 3 + 2];
36
37 const int cb = cell * kNv;
38 const int gb = ghost * kNv;
39
40 const double rho_i = fmax(U[cb + 0], 1e-14);
41 const double u_i = U[cb + 1] / rho_i;
42 const double v_i = U[cb + 2] / rho_i;
43 double p_i = (gamma - 1.0) * (U[cb + 3] - 0.5 * rho_i * (u_i * u_i + v_i * v_i));
44 if (!(p_i > 1e-12) || !isfinite(p_i))
45 p_i = 1e-12;
46 const double T_i = p_i / (rho_i * rgas);
47
48 double rho_g = rho_i;
49 double p_g = p_i;
50 if (Tw > 0.0)
51 {
52 double T_g = 2.0 * Tw - T_i;
53 if (!(T_g > 1e-12) || !isfinite(T_g))
54 T_g = Tw;
55 p_g = p_i;
56 rho_g = p_g / (rgas * T_g);
57 if (!(rho_g > 1e-14) || !isfinite(rho_g))
58 rho_g = rho_i;
59 }
60
61 U[gb + 0] = rho_g;
62 U[gb + 1] = -rho_g * u_i;
63 U[gb + 2] = -rho_g * v_i;
64 U[gb + 3] = p_g / (gamma - 1.0) + 0.5 * rho_g * (u_i * u_i + v_i * v_i);
65}
