28__device__ void compute_roe_averages_device(
29 double rho_L, double u_L, double v_L, double H_L,
30 double rho_R, double u_R, double v_R, double H_R,
31 double gamma,
32 double& roe_u, double& roe_v, double& roe_H, double& roe_a
33) {
34 double sqrt_rho_L = sqrt(rho_L);
35 double sqrt_rho_R = sqrt(rho_R);
36 double term = 1.0 / (sqrt_rho_L + sqrt_rho_R);
37
38 // Density-weighted averaging
39 roe_u = term * (u_L * sqrt_rho_L + u_R * sqrt_rho_R);
40 roe_v = term * (v_L * sqrt_rho_L + v_R * sqrt_rho_R);
41 roe_H = term * (H_L * sqrt_rho_L + H_R * sqrt_rho_R);
42
43 // Sound speed from enthalpy
44 double roe_Vmag = 0.5 * (roe_u * roe_u + roe_v * roe_v);
45 roe_a = sqrt(fmax((gamma - 1.0) * (roe_H - roe_Vmag), 1e-14));
46}

51__device__ double apply_entropy_fix_device(double lambda, double a) {
52 double delta = 0.1 * a;
53 if (fabs(lambda) < delta) {
54 return 0.5 * (lambda * lambda / delta + delta);
55 }
56 return fabs(lambda);
57}

77__global__ void roe_flux_kernel(
78 const double* U_cells,
79 const double* P_cells,
80 const int* face_neighbors,
81 const double* face_normals,
82 const double* face_areas,
83 double* flux_output,
84 int num_cells,
85 double gamma
86) {
87 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
88 if (cell_idx >= num_cells) return;
89
90 // Process each face
91 for (int face = 0; face < 4; face++) {
92 int neighbor_idx = face_neighbors[cell_idx * 4 + face];
93
94 // Skip invalid neighbors
95 if (neighbor_idx < 0 || neighbor_idx >= num_cells) continue;
96
97 // Extract face geometry
98 int normal_offset = cell_idx * 8 + face * 2;
99 double nx = face_normals[normal_offset + 0];
100 double ny = face_normals[normal_offset + 1];
101 double area = face_areas[cell_idx * 4 + face];
102
103 // Normalize face normal
104 double normal_mag = sqrt(nx * nx + ny * ny);
105 if (normal_mag < 1e-14) continue;
106 nx /= normal_mag;
107 ny /= normal_mag;
108
109 // Extract left state (current cell)
110 int L_offset = cell_idx * 6;
111 double rho_L = P_cells[L_offset + 0];
112 double u_L = P_cells[L_offset + 1];
113 double v_L = P_cells[L_offset + 2];
114 double P_L = P_cells[L_offset + 3];
115 double a_L = P_cells[L_offset + 5];
116
117 // Validate left state
118 if (rho_L <= 0.0 || P_L <= 0.0 || a_L <= 0.0) continue;
119
120 double Vmag_L = 0.5 * (u_L * u_L + v_L * v_L);
121 double H_L = (a_L * a_L) / (gamma - 1.0) + Vmag_L;
122
123 // Extract right state (neighbor cell)
124 int R_offset = neighbor_idx * 6;
125 double rho_R = P_cells[R_offset + 0];
126 double u_R = P_cells[R_offset + 1];
127 double v_R = P_cells[R_offset + 2];
128 double P_R = P_cells[R_offset + 3];
129 double a_R = P_cells[R_offset + 5];
130
131 // Validate right state
132 if (rho_R <= 0.0 || P_R <= 0.0 || a_R <= 0.0) continue;
133
134 double Vmag_R = 0.5 * (u_R * u_R + v_R * v_R);
135 double H_R = (a_R * a_R) / (gamma - 1.0) + Vmag_R;
136
137 // Compute Roe averages
138 double roe_u, roe_v, roe_H, roe_a;
139 compute_roe_averages_device(rho_L, u_L, v_L, H_L,
140 rho_R, u_R, v_R, H_R,
141 gamma, roe_u, roe_v, roe_H, roe_a);
142
143 double sqrt_rho_L = sqrt(rho_L);
144 double sqrt_rho_R = sqrt(rho_R);
145 double roe_rho = sqrt(rho_L * rho_R);
146
147 // Normal and tangential velocities
148 double Un = nx * roe_u + ny * roe_v;
149 double Ut = -ny * roe_u + nx * roe_v;
150
151 // State differences
152 double du = u_R - u_L;
153 double dv = v_R - v_L;
154 double dP = P_R - P_L;
155 double drho = rho_R - rho_L;
156 double dUn = nx * du + ny * dv;
157 double dUt = -ny * du + nx * dv;
158
159 // Wave strengths (characteristic variable coefficients)
160 double a2_inv = 1.0 / (roe_a * roe_a);
161 double alpha_1 = 0.5 * a2_inv * (dP - roe_rho * roe_a * dUn);
162 double alpha_2 = drho - a2_inv * dP;
163 double alpha_3 = roe_rho * dUt;
164 double alpha_4 = 0.5 * a2_inv * (dP + roe_rho * roe_a * dUn);
165
166 // Eigenvalues with entropy fix
167 double lambda_1 = apply_entropy_fix_device(Un - roe_a, roe_a);
168 double lambda_2 = fabs(Un);
169 double lambda_3 = fabs(Un);
170 double lambda_4 = apply_entropy_fix_device(Un + roe_a, roe_a);
171
172 // Right eigenvectors
173 // R1: lambda = u - a
174 double R11 = 1.0;
175 double R12 = roe_u - nx * roe_a;
176 double R13 = roe_v - ny * roe_a;
177 double R14 = roe_H - roe_a * Un;
178
179 // R2: lambda = u (entropy)
180 double R21 = 1.0;
181 double R22 = roe_u;
182 double R23 = roe_v;
183 double R24 = 0.5 * (roe_u * roe_u + roe_v * roe_v);
184
185 // R3: lambda = u (shear)
186 double R31 = 0.0;
187 double R32 = -ny;
188 double R33 = nx;
189 double R34 = Ut;
190
191 // R4: lambda = u + a
192 double R41 = 1.0;
193 double R42 = roe_u + nx * roe_a;
194 double R43 = roe_v + ny * roe_a;
195 double R44 = roe_H + roe_a * Un;
196
197 // Compute dissipative flux: D = 0.5 * sum(|lambda_k| * alpha_k * R_k)
198 double D1 = 0.5 * (lambda_1 * alpha_1 * R11 +
199 lambda_2 * alpha_2 * R21 +
200 lambda_3 * alpha_3 * R31 +
201 lambda_4 * alpha_4 * R41);
202
203 double D2 = 0.5 * (lambda_1 * alpha_1 * R12 +
204 lambda_2 * alpha_2 * R22 +
205 lambda_3 * alpha_3 * R32 +
206 lambda_4 * alpha_4 * R42);
207
208 double D3 = 0.5 * (lambda_1 * alpha_1 * R13 +
209 lambda_2 * alpha_2 * R23 +
210 lambda_3 * alpha_3 * R33 +
211 lambda_4 * alpha_4 * R43);
212
213 double D4 = 0.5 * (lambda_1 * alpha_1 * R14 +
214 lambda_2 * alpha_2 * R24 +
215 lambda_3 * alpha_3 * R34 +
216 lambda_4 * alpha_4 * R44);
217
218 // Compute convective fluxes
219 double Un_L = nx * u_L + ny * v_L;
220 double Un_R = nx * u_R + ny * v_R;
221
222 double F_L[4], F_R[4];
223 F_L[0] = rho_L * Un_L;
224 F_L[1] = rho_L * u_L * Un_L + P_L * nx;
225 F_L[2] = rho_L * v_L * Un_L + P_L * ny;
226 F_L[3] = (U_cells[cell_idx * 4 + 3] + P_L) * Un_L;
227
228 F_R[0] = rho_R * Un_R;
229 F_R[1] = rho_R * u_R * Un_R + P_R * nx;
230 F_R[2] = rho_R * v_R * Un_R + P_R * ny;
231 F_R[3] = (U_cells[neighbor_idx * 4 + 3] + P_R) * Un_R;
232
233 // Final Roe flux: F_roe = 0.5*(F_L + F_R) - D
234 int flux_offset = (cell_idx * 4 + face) * 4;
235 flux_output[flux_offset + 0] = 0.5 * (F_L[0] + F_R[0] - D1) * area;
236 flux_output[flux_offset + 1] = 0.5 * (F_L[1] + F_R[1] - D2) * area;
237 flux_output[flux_offset + 2] = 0.5 * (F_L[2] + F_R[2] - D3) * area;
238 flux_output[flux_offset + 3] = 0.5 * (F_L[3] + F_R[3] - D4) * area;
239 }

262 const double* U_cells,
263 const double* P_cells,
264 const int* face_neighbors,
265 const double* face_normals,
266 const double* face_areas,
267 const double* limited_gradients,
268 const double* cell_centers,
269 const double* face_centers,
270 double* flux_output,
271 int num_cells,
272 double gamma
273) {
274 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
275 if (cell_idx >= num_cells) return;
276
277 // Get cell center
278 double xc = cell_centers[cell_idx * 2 + 0];
279 double yc = cell_centers[cell_idx * 2 + 1];
280
281 // Process each face
282 for (int face = 0; face < 4; face++) {
283 int neighbor_idx = face_neighbors[cell_idx * 4 + face];
284 if (neighbor_idx < 0 || neighbor_idx >= num_cells) continue;
285
286 // Face geometry
287 int normal_offset = cell_idx * 8 + face * 2;
288 double nx = face_normals[normal_offset + 0];
289 double ny = face_normals[normal_offset + 1];
290 double area = face_areas[cell_idx * 4 + face];
291
292 double normal_mag = sqrt(nx * nx + ny * ny);
293 if (normal_mag < 1e-14) continue;
294 nx /= normal_mag;
295 ny /= normal_mag;
296
297 // Get face center
298 int face_center_offset = (cell_idx * 4 + face) * 2;
299 double xf = face_centers[face_center_offset + 0];
300 double yf = face_centers[face_center_offset + 1];
301
302 // Distance from cell center to face center
303 double dx_L = xf - xc;
304 double dy_L = yf - yc;
305
306 // Get neighbor cell center
307 double xc_R = cell_centers[neighbor_idx * 2 + 0];
308 double yc_R = cell_centers[neighbor_idx * 2 + 1];
309 double dx_R = xf - xc_R;
310 double dy_R = yf - yc_R;
311
312 // Reconstruct left state using gradients
313 int grad_offset_L = (cell_idx * 4 + face) * 4;
314 double U_L[4];
315 for (int var = 0; var < 4; var++) {
316 double gradient = limited_gradients[grad_offset_L + var];
317 U_L[var] = U_cells[cell_idx * 4 + var] + gradient * dx_L;
318 }
319
320 // Reconstruct right state
321 int grad_offset_R = (neighbor_idx * 4 + face) * 4;
322 double U_R[4];
323 for (int var = 0; var < 4; var++) {
324 double gradient = limited_gradients[grad_offset_R + var];
325 U_R[var] = U_cells[neighbor_idx * 4 + var] - gradient * dx_R;
326 }
327
328 // Convert to primitive variables
329 double rho_L = fmax(U_L[0], 1e-14);
330 double u_L = U_L[1] / rho_L;
331 double v_L = U_L[2] / rho_L;
332 double E_L = U_L[3];
333 double P_L = fmax((gamma - 1.0) * (E_L - 0.5 * rho_L * (u_L * u_L + v_L * v_L)), 1e-14);
334 double a_L = sqrt(gamma * P_L / rho_L);
335 double H_L = (E_L + P_L) / rho_L;
336
337 double rho_R = fmax(U_R[0], 1e-14);
338 double u_R = U_R[1] / rho_R;
339 double v_R = U_R[2] / rho_R;
340 double E_R = U_R[3];
341 double P_R = fmax((gamma - 1.0) * (E_R - 0.5 * rho_R * (u_R * u_R + v_R * v_R)), 1e-14);
342 double a_R = sqrt(gamma * P_R / rho_R);
343 double H_R = (E_R + P_R) / rho_R;
344
345 // Compute Roe averages
346 double roe_u, roe_v, roe_H, roe_a;
347 compute_roe_averages_device(rho_L, u_L, v_L, H_L,
348 rho_R, u_R, v_R, H_R,
349 gamma, roe_u, roe_v, roe_H, roe_a);
350
351 double roe_rho = sqrt(rho_L * rho_R);
352 double Un = nx * roe_u + ny * roe_v;
353 double Ut = -ny * roe_u + nx * roe_v;
354
355 // State differences
356 double du = u_R - u_L;
357 double dv = v_R - v_L;
358 double dP = P_R - P_L;
359 double drho = rho_R - rho_L;
360 double dUn = nx * du + ny * dv;
361 double dUt = -ny * du + nx * dv;
362
363 // Wave strengths
364 double a2_inv = 1.0 / (roe_a * roe_a);
365 double alpha_1 = 0.5 * a2_inv * (dP - roe_rho * roe_a * dUn);
366 double alpha_2 = drho - a2_inv * dP;
367 double alpha_3 = roe_rho * dUt;
368 double alpha_4 = 0.5 * a2_inv * (dP + roe_rho * roe_a * dUn);
369
370 // Eigenvalues with entropy fix
371 double lambda_1 = apply_entropy_fix_device(Un - roe_a, roe_a);
372 double lambda_2 = fabs(Un);
373 double lambda_3 = fabs(Un);
374 double lambda_4 = apply_entropy_fix_device(Un + roe_a, roe_a);
375
376 // Right eigenvectors (same as first order)
377 double R11 = 1.0, R12 = roe_u - nx * roe_a, R13 = roe_v - ny * roe_a, R14 = roe_H - roe_a * Un;
378 double R21 = 1.0, R22 = roe_u, R23 = roe_v, R24 = 0.5 * (roe_u * roe_u + roe_v * roe_v);
379 double R31 = 0.0, R32 = -ny, R33 = nx, R34 = Ut;
380 double R41 = 1.0, R42 = roe_u + nx * roe_a, R43 = roe_v + ny * roe_a, R44 = roe_H + roe_a * Un;
381
382 // Dissipative flux
383 double D1 = 0.5 * (lambda_1 * alpha_1 * R11 + lambda_2 * alpha_2 * R21 + lambda_3 * alpha_3 * R31 + lambda_4 * alpha_4 * R41);
384 double D2 = 0.5 * (lambda_1 * alpha_1 * R12 + lambda_2 * alpha_2 * R22 + lambda_3 * alpha_3 * R32 + lambda_4 * alpha_4 * R42);
385 double D3 = 0.5 * (lambda_1 * alpha_1 * R13 + lambda_2 * alpha_2 * R23 + lambda_3 * alpha_3 * R33 + lambda_4 * alpha_4 * R43);
386 double D4 = 0.5 * (lambda_1 * alpha_1 * R14 + lambda_2 * alpha_2 * R24 + lambda_3 * alpha_3 * R34 + lambda_4 * alpha_4 * R44);
387
388 // Convective fluxes with reconstructed states
389 double Un_L = nx * u_L + ny * v_L;
390 double Un_R = nx * u_R + ny * v_R;
391
392 double F_L[4], F_R[4];
393 F_L[0] = rho_L * Un_L;
394 F_L[1] = rho_L * u_L * Un_L + P_L * nx;
395 F_L[2] = rho_L * v_L * Un_L + P_L * ny;
396 F_L[3] = (E_L + P_L) * Un_L;
397
398 F_R[0] = rho_R * Un_R;
399 F_R[1] = rho_R * u_R * Un_R + P_R * nx;
400 F_R[2] = rho_R * v_R * Un_R + P_R * ny;
401 F_R[3] = (E_R + P_R) * Un_R;
402
403 // Final flux
404 int flux_offset = (cell_idx * 4 + face) * 4;
405 flux_output[flux_offset + 0] = 0.5 * (F_L[0] + F_R[0] - D1) * area;
406 flux_output[flux_offset + 1] = 0.5 * (F_L[1] + F_R[1] - D2) * area;
407 flux_output[flux_offset + 2] = 0.5 * (F_L[2] + F_R[2] - D3) * area;
408 flux_output[flux_offset + 3] = 0.5 * (F_L[3] + F_R[3] - D4) * area;
409 }
