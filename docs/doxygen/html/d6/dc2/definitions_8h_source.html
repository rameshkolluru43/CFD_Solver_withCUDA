75inline void CFD_EnsureParentDirectoryForOutput(const std::string &path, const std::string &context)
76{
77 if (path.empty())
78 throw std::runtime_error(context + ": empty output file path");
79
80 const std::filesystem::path outputPath(path);
81 const std::filesystem::path parent = outputPath.parent_path();
82 if (parent.empty())
83 return;
84
85 std::error_code ec;
86 if (!std::filesystem::exists(parent, ec) || ec)
87 throw std::runtime_error(context + ": output directory not found: " + parent.string());
88 if (!std::filesystem::is_directory(parent, ec) || ec)
89 throw std::runtime_error(context + ": output parent is not a directory: " + parent.string());
90}
