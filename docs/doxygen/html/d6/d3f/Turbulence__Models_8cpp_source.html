148void Update_Effective_Viscosity()
149{
150 if (current_turbulence_model == TurbulenceModel::LAMINAR)
151 return;
152
153 for (int i = 0; i < Total_No_Cells; i++)
154 {
155 if (Cells[i].cellType != -1)
156 {
157 double mu_laminar = Calculate_Laminar_Viscosity(i);
158 double mu_turbulent = turbulence_vars[i].mut;
159
160 // Store effective viscosity (to be used in viscous flux calculations)
161 turbulence_vars[i].mu_effective = mu_laminar + mu_turbulent;
162
163 // Update thermal conductivity if needed
164 double Pr_laminar = 0.72; // Prandtl number for air
165 double Pr_turbulent = 0.9; // Turbulent Prandtl number
166
167 turbulence_vars[i].k_effective = mu_laminar * cp / Pr_laminar +
168 mu_turbulent * cp / Pr_turbulent;
169 }
170 }

246void Calculate_Wall_Distance(int cell_index)
247{
248 if (Cells[cell_index].cellType == -1)
249 return;
250
251 double min_distance = 1e10;
252
253 // Find minimum distance to wall boundaries
254 for (int i = 0; i < Total_No_Cells; i++)
255 {
256 if (Cells[i].has_Wall_Face || Cells[i].hasBoundaryface)
257 {
258 // Check if this is actually a wall boundary
259 bool is_wall = false;
260 // This should be implemented based on your boundary condition marking
261 // For now, assume cells with hasBoundaryface are walls
262 if (Cells[i].hasBoundaryface)
263 {
264 is_wall = true;
265 }
266
267 if (is_wall)
268 {
269 double dx = Cells[cell_index].Cell_Center[0] - Cells[i].Cell_Center[0];
270 double dy = Cells[cell_index].Cell_Center[1] - Cells[i].Cell_Center[1];
271 double distance = sqrt(dx * dx + dy * dy);
272
273 min_distance = min(min_distance, distance);
274 }
275 }
276 }
277
278 turbulence_vars[cell_index].y_wall = min_distance;
279}

285void Calculate_Wall_Shear_Stress(int cell_index)
286{
287 if (Cells[cell_index].cellType == -1 || !Cells[cell_index].hasBoundaryface)
288 return;
289
290 double rho = U_Cells[cell_index][0];
291 double u_vel = U_Cells[cell_index][1] / rho;
292 double v_vel = U_Cells[cell_index][2] / rho;
293
294 double U_magnitude = sqrt(u_vel * u_vel + v_vel * v_vel);
295 double mu = Calculate_Laminar_Viscosity(cell_index);
296 double y_wall = turbulence_vars[cell_index].y_wall;
297
298 // Simple wall shear stress calculation
299 if (y_wall > 0)
300 {
301 turbulence_vars[cell_index].tau_wall = mu * U_magnitude / y_wall;
302 }
303 else
304 {
305 turbulence_vars[cell_index].tau_wall = 0.0;
306 }
307}

313void Calculate_Y_Plus(int cell_index)
314{
315 if (Cells[cell_index].cellType == -1)
316 return;
317
318 double rho = U_Cells[cell_index][0];
319 double tau_wall = turbulence_vars[cell_index].tau_wall;
320 double mu = Calculate_Laminar_Viscosity(cell_index);
321 double y_wall = turbulence_vars[cell_index].y_wall;
322
323 if (tau_wall > 0)
324 {
325 turbulence_vars[cell_index].u_tau = sqrt(tau_wall / rho);
326 turbulence_vars[cell_index].y_plus = rho * turbulence_vars[cell_index].u_tau * y_wall / mu;
327 }
328 else
329 {
330 turbulence_vars[cell_index].u_tau = 0.0;
331 turbulence_vars[cell_index].y_plus = 0.0;
332 }

520void Calculate_Turbulence_Statistics()
521{
522 if (current_turbulence_model == TurbulenceModel::LAMINAR)
523 return;
524
525 double k_max = 0.0, k_min = 1e10, k_avg = 0.0;
526 double mut_max = 0.0, mut_min = 1e10, mut_avg = 0.0;
527 double y_plus_max = 0.0, y_plus_min = 1e10, y_plus_avg = 0.0;
528 int count = 0;
529
530 for (int i = 0; i < Total_No_Cells; i++)
531 {
532 if (Cells[i].cellType != -1)
533 {
534 k_max = max(k_max, turbulence_vars[i].k);
535 k_min = min(k_min, turbulence_vars[i].k);
536 k_avg += turbulence_vars[i].k;
537
538 mut_max = max(mut_max, turbulence_vars[i].mut);
539 mut_min = min(mut_min, turbulence_vars[i].mut);
540 mut_avg += turbulence_vars[i].mut;
541
542 if (turbulence_vars[i].y_plus > 0)
543 {
544 y_plus_max = max(y_plus_max, turbulence_vars[i].y_plus);
545 y_plus_min = min(y_plus_min, turbulence_vars[i].y_plus);
546 y_plus_avg += turbulence_vars[i].y_plus;
547 }
548
549 count++;
550 }
551 }
552
553 if (count > 0)
554 {
555 k_avg /= count;
556 mut_avg /= count;

567 cout << " Max: " << mut_max << " kg/(m·s)" << endl;

644void Apply_Low_Re_Wall_Treatment(int cell_index)
645{
646 // Set k and epsilon/omega to zero at the wall for low-Re treatment
647 turbulence_vars[cell_index].k = 0.0;
648
649 if (current_turbulence_model == TurbulenceModel::K_EPSILON)
650 {
651 double rho = U_Cells[cell_index][0];
652 double mu = Calculate_Laminar_Viscosity(cell_index);
653 double y_wall = turbulence_vars[cell_index].y_wall;
654
655 // Near-wall epsilon treatment
656 turbulence_vars[cell_index].epsilon = 2.0 * mu * turbulence_vars[cell_index].k /
657 (rho * y_wall * y_wall);
658 }
659 else
660 {
661 double rho = U_Cells[cell_index][0];
662 double mu = Calculate_Laminar_Viscosity(cell_index);
663 double y_wall = turbulence_vars[cell_index].y_wall;
664
665 // Near-wall omega treatment
666 turbulence_vars[cell_index].omega = 60.0 * mu / (rho * KOmega::beta * y_wall * y_wall);
667 }
