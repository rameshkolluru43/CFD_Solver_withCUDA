15inline void cfd_primitive_from_U(float4 U, thread float& rho, thread float& u, thread float& v,
16 thread float& P, thread float& c)
17{
18 rho = U.x;
19 float invR = 1.0f / max(rho, 1e-20f);
20 u = U.y * invR;
21 v = U.z * invR;
22 float ke = 0.5f * (u * u + v * v);
23 P = kGammaM1 * (U.w - rho * ke);
24 P = max(P, 1e-20f);
25 c = sqrt(kGamma * P * invR);
26}

54kernel void cfd_rusanov_face_flux_2d(device const float* U_L [[buffer(0)]],
55 device const float* U_R [[buffer(1)]], device const float* face_n [[buffer(2)]],
56 device const float* face_dl [[buffer(3)]], device float* flux_out [[buffer(4)]],
57 constant uint& n_faces [[buffer(5)]], uint gid [[thread_position_in_grid]])
58{
59 if (gid >= n_faces)
60 return;
61 size_t b = static_cast<size_t>(gid) * 4;
62 float4 UL(U_L[b], U_L[b + 1], U_L[b + 2], U_L[b + 3]);
63 float4 UR(U_R[b], U_R[b + 1], U_R[b + 2], U_R[b + 3]);
64
65 float rL, uL, vL, pL, cL;
66 float rR, uR, vR, pR, cR;
67 cfd_primitive_from_U(UL, rL, uL, vL, pL, cL);
68 cfd_primitive_from_U(UR, rR, uR, vR, pR, cR);
69
70 float nx = face_n[gid * 2];
71 float ny = face_n[gid * 2 + 1];
72 float dl = face_dl[gid];
73
74 float vnL = uL * nx + vL * ny;
75 float vnR = uR * nx + vR * ny;
76
77 float FL0 = rL * vnL * dl;
78 float FL1 = (rL * uL * vnL + pL * nx) * dl;
79 float FL2 = (rL * vL * vnL + pL * ny) * dl;
80 float FL3 = vnL * (UL.w + pL) * dl;
81
82 float FR0 = rR * vnR * dl;
83 float FR1 = (rR * uR * vnR + pR * nx) * dl;
84 float FR2 = (rR * vR * vnR + pR * ny) * dl;
85 float FR3 = vnR * (UR.w + pR) * dl;
86
87 float lam = max(fabs(vnL) + cL, fabs(vnR) + cR);
88 float lamdl = lam * dl;
89
90 flux_out[b] = 0.5f * (FL0 + FR0) - 0.5f * lamdl * (UR.x - UL.x);
91 flux_out[b + 1] = 0.5f * (FL1 + FR1) - 0.5f * lamdl * (U_R[b + 1] - U_L[b + 1]);
92 flux_out[b + 2] = 0.5f * (FL2 + FR2) - 0.5f * lamdl * (U_R[b + 2] - U_L[b + 2]);
93 flux_out[b + 3] = 0.5f * (FL3 + FR3) - 0.5f * lamdl * (U_R[b + 3] - U_L[b + 3]);
94}

96kernel void cfd_inviscid_dt_denominator_quad(device const float* face_pack [[buffer(0)]],
97 device const float* rho [[buffer(1)]], device const float* u [[buffer(2)]],
98 device const float* v [[buffer(3)]], device const float* P [[buffer(4)]],
99 device float* denom_out [[buffer(5)]], constant uint& n_cells [[buffer(6)]],
100 uint gid [[thread_position_in_grid]])
101{
102 if (gid >= n_cells)
103 return;
104 float r = max(rho[gid], 1e-20f);
105 float ui = u[gid];
106 float vi = v[gid];
107 float pi = max(P[gid], 1e-20f);
108 float c = sqrt(kGamma * pi / r);
109 size_t o = static_cast<size_t>(gid) * 12;
110 float s = 0.0f;
111 for (int f = 0; f < 4; f++)
112 {
113 float nx = face_pack[o + f * 3];
114 float ny = face_pack[o + f * 3 + 1];
115 float dl = face_pack[o + f * 3 + 2];
116 float vn = ui * nx + vi * ny;
117 s += (fabs(vn) + c) * dl;
118 }
119 denom_out[gid] = s;
120}
