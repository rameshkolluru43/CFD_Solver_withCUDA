208void Runge_Kutta_Method_With_Turbulence()
209{
210 // Store initial state
211 vector<vector<double>> U_initial(Total_No_Cells, vector<double>(5));
212 vector<TurbulenceVariables> turb_initial(Total_No_Cells);
213
214 for (int i = 0; i < Total_No_Cells; i++)
215 {
216 if (Cells[i].cellType != -1)
217 {
218 U_initial[i] = U_Cells[i];
219 turb_initial[i] = turbulence_vars[i];
220 }
221 }
222
223 // RK4 stages
224 for (int stage = 1; stage <= 4; stage++)
225 {
226 // Calculate time step
227 Calculate_Time_Step();
228
229 // Calculate flow residuals
230 Calculate_Flow_Residuals();
231
232 // Calculate turbulence residuals
233 Calculate_Turbulence_Residuals();
234
235 // Update variables based on RK4 coefficients
236 double alpha = Get_RK4_Coefficient(stage);
237 double beta = Get_RK4_Beta_Coefficient(stage);
238
239 Update_Flow_Variables_RK4(U_initial, alpha, beta);
240 Update_Turbulence_Variables_RK4(turb_initial, alpha, beta);
241
242 // Apply boundary conditions after each stage
243 Apply_Boundary_Conditions();
244 Apply_All_Turbulence_Boundary_Conditions();
245 }

291void Calculate_Turbulence_Residuals()
292{
293 if (current_turbulence_model == TurbulenceModel::LAMINAR)
294 return;
295
296 // Calculate production terms for all cells
297 for (int i = 0; i < Total_No_Cells; i++)
298 {
299 if (Cells[i].cellType != -1)
300 {
301 switch (current_turbulence_model)
302 {
303 case TurbulenceModel::K_EPSILON:
304 Calculate_KEpsilon_Production_Terms(i);
305 break;
306 case TurbulenceModel::K_OMEGA_WILCOX:
307 case TurbulenceModel::K_OMEGA_SST:
308 Calculate_KOmega_Production_Terms(i);
309 break;
310 default:
311 break;
312 }
313 }
314 }
315
316 // Calculate diffusion and source terms
317 for (int i = 0; i < Total_No_Cells; i++)
318 {
319 if (Cells[i].cellType != -1)
320 {
321 Calculate_Turbulence_Cell_Residual(i);
322 }
323 }
324}

330void Calculate_Turbulence_Cell_Residual(int cell_index)
331{
332 if (Cells[cell_index].cellType == -1)
333 return;
334
335 double rho = U_Cells[cell_index][0];
336 double k = turbulence_vars[cell_index].k;
337 double Pk = turbulence_vars[cell_index].Pk;
338
339 switch (current_turbulence_model)
340 {
341 case TurbulenceModel::K_EPSILON:
342 {
343 double epsilon = turbulence_vars[cell_index].epsilon;
344
345 // K-equation residual
346 double dk_diff = Calculate_KEpsilon_K_Diffusion(cell_index);
347 double Rk = dk_diff + Pk - rho * epsilon;
348
349 // Epsilon-equation residual
350 double depsilon_diff = Calculate_KEpsilon_Epsilon_Diffusion(cell_index);
351 double Repsilon = depsilon_diff + KEpsilon::C1_eps * (epsilon / k) * Pk -
352 KEpsilon::C_2 * rho * (epsilon * epsilon / k);
353
354 // Store residuals (these would be used in implicit solver)
355 turbulence_vars[cell_index].k_residual = Rk;
356 turbulence_vars[cell_index].epsilon_residual = Repsilon;
357 break;
358 }
359
360 case TurbulenceModel::K_OMEGA_WILCOX:
361 case TurbulenceModel::K_OMEGA_SST:
362 {
363 double omega = turbulence_vars[cell_index].omega;
364
365 // K-equation residual
366 double dk_diff = Calculate_KOmega_K_Diffusion(cell_index);
367 double Rk = dk_diff + Pk - KOmega::beta_star * rho * k * omega;
368
369 // Omega-equation residual
370 double domega_diff = Calculate_KOmega_Omega_Diffusion(cell_index);
371 double Romega = domega_diff + KOmega::alpha * (omega / k) * Pk -
372 KOmega::beta * rho * omega * omega;
373
374 // Store residuals
375 turbulence_vars[cell_index].k_residual = Rk;
376 turbulence_vars[cell_index].omega_residual = Romega;
377 break;
378 }
379
380 default:
381 break;
382 }
383}

395void Update_Turbulence_Variables_RK4(const vector<TurbulenceVariables> &turb_initial,
396 double alpha, double beta)
397{
398 if (current_turbulence_model == TurbulenceModel::LAMINAR)
399 return;
400
401 for (int i = 0; i < Total_No_Cells; i++)
402 {
403 if (Cells[i].cellType != -1)
404 {
405 double rho = U_Cells[i][0];
406 double dt = Min_dt;
407
408 // Update k
409 double k_old = turb_initial[i].k;
410 double k_residual = turbulence_vars[i].k_residual;
411 double k_new = k_old + alpha * dt * k_residual / rho;
412 turbulence_vars[i].k = max(k_new, 1e-10);
413
414 // Update epsilon or omega
415 if (current_turbulence_model == TurbulenceModel::K_EPSILON)
416 {
417 double eps_old = turb_initial[i].epsilon;
418 double eps_residual = turbulence_vars[i].epsilon_residual;
419 double eps_new = eps_old + alpha * dt * eps_residual / rho;
420 turbulence_vars[i].epsilon = max(eps_new, 1e-10);
421
422 Calculate_KEpsilon_Turbulent_Viscosity(i);
423 }
424 else
425 {
426 double omega_old = turb_initial[i].omega;
427 double omega_residual = turbulence_vars[i].omega_residual;
428 double omega_new = omega_old + alpha * dt * omega_residual / rho;
429 turbulence_vars[i].omega = max(omega_new, 1e-6);
430
431 Calculate_KOmega_Turbulent_Viscosity(i);
432 }
433 }
434 }

446double Calculate_Turbulence_Error(const string &variable)
447{
448 if (current_turbulence_model == TurbulenceModel::LAMINAR)
449 return 0.0;
450
451 double error_sum = 0.0;
452 int count = 0;
453
454 for (int i = 0; i < Total_No_Cells; i++)
455 {
456 if (Cells[i].cellType != -1)
457 {
458 double error = 0.0;
459
460 if (variable == "k")
461 {
462 error = abs(turbulence_vars[i].k_residual);
463 }
464 else if (variable == "eps_omega")
465 {
466 if (current_turbulence_model == TurbulenceModel::K_EPSILON)
467 {
468 error = abs(turbulence_vars[i].epsilon_residual);
469 }
470 else
471 {
472 error = abs(turbulence_vars[i].omega_residual);
473 }
474 }
475
476 error_sum += error * error;
477 count++;
478 }
479 }
480
481 return (count > 0) ? sqrt(error_sum / count) : 0.0;
482}

526void Apply_Turbulence_Limiters(int cell_index)
527{
528 if (Cells[cell_index].cellType == -1)
529 return;
530
531 // Apply positivity constraints
532 turbulence_vars[cell_index].k = max(turbulence_vars[cell_index].k, 1e-10);
533
534 if (current_turbulence_model == TurbulenceModel::K_EPSILON)
535 {
536 turbulence_vars[cell_index].epsilon = max(turbulence_vars[cell_index].epsilon, 1e-10);
537 }
538 else
539 {
540 turbulence_vars[cell_index].omega = max(turbulence_vars[cell_index].omega, 1e-6);
541 }
542
543 // Apply realizability constraints
544 Apply_Realizability_Constraints(cell_index);
545
546 // Limit turbulent viscosity ratio
547 double mu_laminar = Calculate_Laminar_Viscosity(cell_index);
548 double max_turbulent_viscosity = 1000.0 * mu_laminar;
549
550 if (turbulence_vars[cell_index].mut > max_turbulent_viscosity)
551 {
552 turbulence_vars[cell_index].mut = max_turbulent_viscosity;
553 turbulence_vars[cell_index].nut = turbulence_vars[cell_index].mut /
554 U_Cells[cell_index][0];
555 }
556}
