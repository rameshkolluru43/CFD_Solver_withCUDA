61/* ======================================================================
62 Transfinite Interpolation (TFI) for algebraic grid generation.
63 Generates interior points from four boundary curves using bilinear
64 blending.
65 ====================================================================== */
66void Grid::TFI()
67{
68 cout << "Generating Grid using Algebraic Transfinite Interpolation\n";
69 cout << "Grid dimensions: " << nx << " x " << ny << endl;
70 cout << "Corner (0,0): (" << x[0][0] << ", " << y[0][0] << ")\n";
71 cout << "Corner (nx-1,0): (" << x[nx - 1][0] << ", " << y[nx - 1][0] << ")\n";
72 cout << "Corner (nx-1,ny-1): (" << x[nx - 1][ny - 1] << ", " << y[nx - 1][ny - 1] << ")\n";
73 cout << "Corner (0,ny-1): (" << x[0][ny - 1] << ", " << y[0][ny - 1] << ")\n";
74
75 /* Step 2: TFI — interior from boundaries only; boundaries are NOT modified here. */
76
77 for (int j = 1; j < ny - 1; j++)
78 {
79 double eta_j = static_cast<double>(j) / (ny - 1);
80 for (int i = 1; i < nx - 1; i++)
81 {
82 double xi_i = static_cast<double>(i) / (nx - 1);
83
84 double x_proj_eta = (1.0 - eta_j) * x[i][0] + eta_j * x[i][ny - 1];
85 double x_proj_xi = (1.0 - xi_i) * x[0][j] + xi_i * x[nx - 1][j];
86 double x_corners = (1.0 - xi_i) * (1.0 - eta_j) * x[0][0]
87 + xi_i * (1.0 - eta_j) * x[nx - 1][0]
88 + (1.0 - xi_i) * eta_j * x[0][ny - 1]
89 + xi_i * eta_j * x[nx - 1][ny - 1];
90
91 double y_proj_eta = (1.0 - eta_j) * y[i][0] + eta_j * y[i][ny - 1];
92 double y_proj_xi = (1.0 - xi_i) * y[0][j] + xi_i * y[nx - 1][j];
93 double y_corners = (1.0 - xi_i) * (1.0 - eta_j) * y[0][0]
94 + xi_i * (1.0 - eta_j) * y[nx - 1][0]
95 + (1.0 - xi_i) * eta_j * y[0][ny - 1]
96 + xi_i * eta_j * y[nx - 1][ny - 1];
97
98 x[i][j] = x_proj_eta + x_proj_xi - x_corners;
99 y[i][j] = y_proj_eta + y_proj_xi - y_corners;
100 }
101 }
102
103 xtemp = x;
104 ytemp = y;
105 cout << "TFI complete. Ready for elliptic smoothing (Step 3; boundaries remain fixed).\n";
106}

108/* ======================================================================
109 Original Laplace elliptic smoother (P=Q=0).
110 Kept for backward compatibility.
111 ====================================================================== */
112void Grid::Elliptic()
113{
114 double dxi = 1.0 / (nx - 1);
115 double deta = 1.0 / (ny - 1);
116 double dxi2 = dxi * dxi;
117 double deta2 = deta * deta;
118 double dxi_deta_4 = 4.0 * dxi * deta;
119
120 for (int j = 1; j < ny - 1; j++)
121 {
122 for (int i = 1; i < nx - 1; i++)
123 {
124 double x_xi = (x[i + 1][j] - xtemp[i - 1][j]) / (2.0 * dxi);
125 double x_eta = (x[i][j + 1] - xtemp[i][j - 1]) / (2.0 * deta);
126 double y_xi = (y[i + 1][j] - ytemp[i - 1][j]) / (2.0 * dxi);
127 double y_eta = (y[i][j + 1] - ytemp[i][j - 1]) / (2.0 * deta);
128
129 double alp = x_eta * x_eta + y_eta * y_eta;
130 double gam = x_xi * x_xi + y_xi * y_xi;
131 double bet = x_xi * x_eta + y_xi * y_eta;
132
133 double coeff_B = alp / dxi2;
134 double coeff_D = gam / deta2;
135 double coeff_C = -bet / dxi_deta_4;
136 double coeff_A = coeff_B + coeff_D;
137
138 double cross_x = coeff_C * (x[i + 1][j + 1] - xtemp[i + 1][j - 1]
139 - x[i - 1][j + 1] + xtemp[i - 1][j - 1]);
140 double cross_y = coeff_C * (y[i + 1][j + 1] - ytemp[i + 1][j - 1]
141 - y[i - 1][j + 1] + ytemp[i - 1][j - 1]);
142
143 xtemp[i][j] = (1.0 / (2.0 * coeff_A)) *
144 (coeff_B * (x[i + 1][j] + xtemp[i - 1][j])
145 + cross_x
146 + coeff_D * (x[i][j + 1] + xtemp[i][j - 1]));
147
148 ytemp[i][j] = (1.0 / (2.0 * coeff_A)) *
149 (coeff_B * (y[i + 1][j] + ytemp[i - 1][j])
150 + cross_y
151 + coeff_D * (y[i][j + 1] + ytemp[i][j - 1]));
152 }
153 }
154}

156/* ======================================================================
157 Poisson elliptic grid generator with source terms P(xi,eta), Q(xi,eta).
158
159 Solves the transformed Poisson equations:
160 alpha * x_xixi - 2*beta * x_xieta + gamma * x_etaeta = -J^2*(P*x_xi + Q*x_eta)
161 alpha * y_xixi - 2*beta * y_xieta + gamma * y_etaeta = -J^2*(P*y_xi + Q*y_eta)
162
163 where:
164 alpha = x_eta^2 + y_eta^2
165 beta = x_xi*x_eta + y_xi*y_eta
166 gamma = x_xi^2 + y_xi^2
167 J = x_xi*y_eta - x_eta*y_xi (Jacobian)
168
169 Source terms P, Q control grid clustering and orthogonality:
170 - Thomas-Middlecoff sources: enforce boundary point distribution
171 - Boundary attraction sources: force orthogonality at walls
172 ====================================================================== */
173void Grid::Elliptic_Poisson()
174{
175 double dxi = 1.0 / (nx - 1);
176 double deta = 1.0 / (ny - 1);
177 double dxi2 = dxi * dxi;
178 double deta2 = deta * deta;
179
180 /* Step 3: Elliptic — update interior only; boundaries x[i][0], x[i][ny-1], x[0][j], x[nx-1][j] are never written. */
181 for (int j = 1; j < ny - 1; j++)
182 {
183 for (int i = 1; i < nx - 1; i++)
184 {
185 double x_xi = (x[i + 1][j] - xtemp[i - 1][j]) / (2.0 * dxi);
186 double x_eta = (x[i][j + 1] - xtemp[i][j - 1]) / (2.0 * deta);
187 double y_xi = (y[i + 1][j] - ytemp[i - 1][j]) / (2.0 * dxi);
188 double y_eta = (y[i][j + 1] - ytemp[i][j - 1]) / (2.0 * deta);
189
190 double alp = x_eta * x_eta + y_eta * y_eta;
191 double gam = x_xi * x_xi + y_xi * y_xi;
192 double bet = x_xi * x_eta + y_xi * y_eta;
193
194 double J = x_xi * y_eta - x_eta * y_xi;
195 double J2 = J * J;
196
197 double Pval = P_source[i][j];
198 double Qval = Q_source[i][j];
199
200 double rhs_x = -J2 * (Pval * x_xi + Qval * x_eta);
201 double rhs_y = -J2 * (Pval * y_xi + Qval * y_eta);
202
203 double coeff_B = alp / dxi2;
204 double coeff_D = gam / deta2;
205 double coeff_C = -bet / (4.0 * dxi * deta);
206 double coeff_A = coeff_B + coeff_D;
207
208 double cross_x = coeff_C * (x[i + 1][j + 1] - xtemp[i + 1][j - 1]
209 - x[i - 1][j + 1] + xtemp[i - 1][j - 1]);
210 double cross_y = coeff_C * (y[i + 1][j + 1] - ytemp[i + 1][j - 1]
211 - y[i - 1][j + 1] + ytemp[i - 1][j - 1]);
212
213 xtemp[i][j] = (1.0 / (2.0 * coeff_A)) *
214 (coeff_B * (x[i + 1][j] + xtemp[i - 1][j])
215 + cross_x
216 + coeff_D * (x[i][j + 1] + xtemp[i][j - 1])
217 + rhs_x);
218
219 ytemp[i][j] = (1.0 / (2.0 * coeff_A)) *
220 (coeff_B * (y[i + 1][j] + ytemp[i - 1][j])
221 + cross_y
222 + coeff_D * (y[i][j + 1] + ytemp[i][j - 1])
223 + rhs_y);
224 }
225 }
226}

228/* ======================================================================
229 Thomas-Middlecoff source term computation.
230 Computes P(xi,eta) and Q(xi,eta) from the boundary point distribution
231 so the interior grid inherits the boundary clustering while decaying
232 smoothly toward the interior.
233
234 On each boundary, the source is computed from the second derivative
235 of arc-length with respect to the computational coordinate.
236 Interior values are interpolated with exponential decay.
237 ====================================================================== */
238void Grid::ComputeThomasMiddlecoffSources()
239{
240 P_source.assign(nx, vector<double>(ny, 0.0));
241 Q_source.assign(nx, vector<double>(ny, 0.0));
242
243 double dxi = 1.0 / (nx - 1);
244 double deta = 1.0 / (ny - 1);
245 double decay = source_params.decay_rate;
246
247 vector<double> P_south(nx, 0.0), P_north(nx, 0.0);
248 vector<double> Q_west(ny, 0.0), Q_east(ny, 0.0);
249
250 // P on j=0 (south) boundary from boundary point spacing
251 for (int i = 1; i < nx - 1; i++)
252 {
253 double x_xi = (x[i + 1][0] - x[i - 1][0]) / (2.0 * dxi);
254 double y_xi = (y[i + 1][0] - y[i - 1][0]) / (2.0 * dxi);
255 double x_xixi = (x[i + 1][0] - 2.0 * x[i][0] + x[i - 1][0]) / (dxi * dxi);
256 double y_xixi = (y[i + 1][0] - 2.0 * y[i][0] + y[i - 1][0]) / (dxi * dxi);
257 double denom = x_xi * x_xi + y_xi * y_xi;
258 if (fabs(denom) > 1e-30)
259 P_south[i] = -(x_xi * x_xixi + y_xi * y_xixi) / denom;
260 }
261
262 // P on j=ny-1 (north) boundary
263 for (int i = 1; i < nx - 1; i++)
264 {
265 int jb = ny - 1;
266 double x_xi = (x[i + 1][jb] - x[i - 1][jb]) / (2.0 * dxi);
267 double y_xi = (y[i + 1][jb] - y[i - 1][jb]) / (2.0 * dxi);
268 double x_xixi = (x[i + 1][jb] - 2.0 * x[i][jb] + x[i - 1][jb]) / (dxi * dxi);
269 double y_xixi = (y[i + 1][jb] - 2.0 * y[i][jb] + y[i - 1][jb]) / (dxi * dxi);
270 double denom = x_xi * x_xi + y_xi * y_xi;
271 if (fabs(denom) > 1e-30)
272 P_north[i] = -(x_xi * x_xixi + y_xi * y_xixi) / denom;
273 }
274
275 // Q on i=0 (west) boundary
276 for (int j = 1; j < ny - 1; j++)
277 {
278 double x_eta = (x[0][j + 1] - x[0][j - 1]) / (2.0 * deta);
279 double y_eta = (y[0][j + 1] - y[0][j - 1]) / (2.0 * deta);
280 double x_etaeta = (x[0][j + 1] - 2.0 * x[0][j] + x[0][j - 1]) / (deta * deta);
281 double y_etaeta = (y[0][j + 1] - 2.0 * y[0][j] + y[0][j - 1]) / (deta * deta);
282 double denom = x_eta * x_eta + y_eta * y_eta;
283 if (fabs(denom) > 1e-30)
284 Q_west[j] = -(x_eta * x_etaeta + y_eta * y_etaeta) / denom;
285 }
286
287 // Q on i=nx-1 (east) boundary
288 for (int j = 1; j < ny - 1; j++)
289 {
290 int ib = nx - 1;
291 double x_eta = (x[ib][j + 1] - x[ib][j - 1]) / (2.0 * deta);
292 double y_eta = (y[ib][j + 1] - y[ib][j - 1]) / (2.0 * deta);
293 double x_etaeta = (x[ib][j + 1] - 2.0 * x[ib][j] + x[ib][j - 1]) / (deta * deta);
294 double y_etaeta = (y[ib][j + 1] - 2.0 * y[ib][j] + y[ib][j - 1]) / (deta * deta);
295 double denom = x_eta * x_eta + y_eta * y_eta;
296 if (fabs(denom) > 1e-30)
297 Q_east[j] = -(x_eta * x_etaeta + y_eta * y_etaeta) / denom;
298 }
299
300 // Interpolate into interior with exponential decay
301 for (int j = 1; j < ny - 1; j++)
302 {
303 double eta_j = static_cast<double>(j) / (ny - 1);
304 double decay_south = exp(-decay * eta_j * (ny - 1));
305 double decay_north = exp(-decay * (1.0 - eta_j) * (ny - 1));
306
307 for (int i = 1; i < nx - 1; i++)
308 {
309 double xi_i = static_cast<double>(i) / (nx - 1);
310 double decay_west = exp(-decay * xi_i * (nx - 1));
311 double decay_east = exp(-decay * (1.0 - xi_i) * (nx - 1));
312
313 P_source[i][j] = P_south[i] * decay_south
314 + P_north[i] * decay_north;
315
316 Q_source[i][j] = Q_west[j] * decay_west
317 + Q_east[j] * decay_east;
318 }
319 }
320}

322/* ======================================================================
323 Boundary attraction source terms.
324 Attract grid lines toward walls so boundary layer clustering occurs
325 along the FULL extent of each boundary (every i for south/north,
326 every j for east/west), not at a single point.
327 When south or north BL is enabled, we set Q_source from wall attraction
328 only (zero T-M eta contribution first) so clustering is uniform in xi
329 along the wall instead of only near corners.
330 ====================================================================== */
331void Grid::ComputeBoundaryAttractionSources()
332{
333 double amp = source_params.attraction_strength;
334
335 // Zero eta (Q) source so south/north attraction alone sets it — uniform in i along wall
336 if (bl_south.enabled || bl_north.enabled)
337 for (int i = 1; i < nx - 1; i++)
338 for (int j = 1; j < ny - 1; j++)
339 Q_source[i][j] = 0.0;
340
341 // South (j=0): Q source — same strength at every i so BL along full wall
342 if (bl_south.enabled)
343 {
344 double wall_decay = -log(0.01) / bl_south.num_layers;
345 double y_range = fabs(y[nx/2][ny-1] - y[nx/2][0]);
346 if (y_range < 1e-14) y_range = 1.0;
347 double typical_dy = y_range / (ny - 1);
348 // Scale attraction so interior tends toward first_cell_height (cap for stability)
349 double scale = typical_dy / (bl_south.first_cell_height + 1e-30);
350 if (scale > 5.0) scale = 5.0; // cap for stability
351 if (scale < 1.0) scale = 1.0;
352 for (int j = 1; j < ny - 1; j++)
353 {
354 double dist = static_cast<double>(j);
355 double attraction = amp * scale * exp(-wall_decay * dist);
356 for (int i = 1; i < nx - 1; i++)
357 Q_source[i][j] -= attraction;
358 }
359 }
360
361 if (bl_north.enabled)
362 {
363 double wall_decay = -log(0.01) / bl_north.num_layers;
364 for (int j = 1; j < ny - 1; j++)
365 {
366 double dist = static_cast<double>(ny - 1 - j);
367 double attraction = amp * exp(-wall_decay * dist);
368 for (int i = 1; i < nx - 1; i++)
369 Q_source[i][j] += attraction;
370 }
371 }
372
373 // Zero xi (P) source so east/west attraction alone sets it — uniform in j
374 if (bl_west.enabled || bl_east.enabled)
375 for (int i = 1; i < nx - 1; i++)
376 for (int j = 1; j < ny - 1; j++)
377 P_source[i][j] = 0.0;
378
379 if (bl_west.enabled)
380 {
381 double wall_decay = -log(0.01) / bl_west.num_layers;
382 for (int i = 1; i < nx - 1; i++)
383 {
384 double dist = static_cast<double>(i);
385 double attraction = amp * exp(-wall_decay * dist);
386 for (int j = 1; j < ny - 1; j++)
387 P_source[i][j] -= attraction;
388 }
389 }
390
391 if (bl_east.enabled)
392 {
393 double wall_decay = -log(0.01) / bl_east.num_layers;
394 for (int i = 1; i < nx - 1; i++)
395 {
396 double dist = static_cast<double>(nx - 1 - i);
397 double attraction = amp * exp(-wall_decay * dist);
398 for (int j = 1; j < ny - 1; j++)
399 P_source[i][j] += attraction;
400 }
401 }

423void Grid::Periodic_Boundary_Condition()
424{
425 double dxi = 1.0 / (nx - 1);
426 double deta = 1.0 / (ny - 1);
427
428 int i = nx - 1;
429 for (int j = 1; j < ny - 1; j++)
430 {
431 double x_xi = (x[1][j] - x[i - 1][j]) / (2.0 * dxi);
432 double x_eta = (x[i][j + 1] - x[i][j - 1]) / (2.0 * deta);
433 double y_xi = (y[1][j] - y[i - 1][j]) / (2.0 * dxi);
434 double y_eta = (y[i][j + 1] - y[i][j - 1]) / (2.0 * deta);
435
436 double alp = x_eta * x_eta + y_eta * y_eta;
437 double gam = x_xi * x_xi + y_xi * y_xi;
438 double bet = x_xi * x_eta + y_xi * y_eta;
439
440 double coeff_B = alp / (dxi * dxi);
441 double coeff_D = gam / (deta * deta);
442 double coeff_C = -bet / (4.0 * dxi * deta);
443 double coeff_A = coeff_B + coeff_D;
444
445 xtemp[i][j] = (1.0 / (2.0 * coeff_A)) *
446 (coeff_B * (x[1][j] + x[i - 1][j])
447 + coeff_C * (x[1][j + 1] - x[1][j - 1] - x[i - 1][j + 1] + x[i - 1][j - 1])
448 + coeff_D * (x[i][j + 1] + x[i][j - 1]));
449
450 ytemp[i][j] = (1.0 / (2.0 * coeff_A)) *
451 (coeff_B * (y[1][j] + y[i - 1][j])
452 + coeff_C * (y[1][j + 1] - y[1][j - 1] - y[i - 1][j + 1] + y[i - 1][j - 1])
453 + coeff_D * (y[i][j + 1] + y[i][j - 1]));
454
455 x[0][j] = xtemp[i][j];
456 y[0][j] = ytemp[i][j];
457 }
458}

460/* ======================================================================
461 Error estimation and grid update (standard Gauss-Seidel).
462 Uses L2 norm of absolute change to avoid division-by-zero when
463 grid coordinates pass through zero.
464 ====================================================================== */
465void Grid::Estimate_Error_and_Update()
466{
467 ertot = 0.0;
468 for (int j = 1; j < ny - 1; j++)
469 {
470 for (int i = 1; i < nx - 1; i++)
471 {
472 double dx = xtemp[i][j] - x[i][j];
473 double dy = ytemp[i][j] - y[i][j];
474 ertot += dx * dx + dy * dy;
475
476 x[i][j] = xtemp[i][j];
477 y[i][j] = ytemp[i][j];
478 }
479 }
480 ertot = sqrt(ertot / ((nx - 2) * (ny - 2)));
481}

483/* ======================================================================
484 Error estimation with Successive Over-Relaxation (SOR).
485 omega in (1,2) for over-relaxation; omega=1 is standard Gauss-Seidel.
486 Typical optimal values are 1.4-1.8 for elliptic grid generation.
487 ====================================================================== */
488void Grid::Estimate_Error_and_Update_SOR(double omega)
489{
490 ertot = 0.0;
491 for (int j = 1; j < ny - 1; j++)
492 {
493 for (int i = 1; i < nx - 1; i++)
494 {
495 double dx = xtemp[i][j] - x[i][j];
496 double dy = ytemp[i][j] - y[i][j];
497 ertot += dx * dx + dy * dy;
498
499 x[i][j] = x[i][j] + omega * dx;
500 y[i][j] = y[i][j] + omega * dy;
501 xtemp[i][j] = x[i][j];
502 ytemp[i][j] = y[i][j];
503 }
504 }
505 ertot = sqrt(ertot / ((nx - 2) * (ny - 2)));
506}

508/* ======================================================================
509 Original grid generation (backward compatible).

511void Grid::Generate_Grid(bool &With_Periodic_Domain)

653/* ======================================================================
654 Grid quality metrics computation.
655 Evaluates Jacobian, aspect ratio, skewness, and orthogonality
656 for every interior cell in the structured grid.
657 ====================================================================== */
658void Grid::ComputeGridQuality(double &min_jac, double &max_ar,
659 double &max_skew, double &min_orth) const
660{
661 min_jac = 1e30;
662 max_ar = 0.0;
663 max_skew = 0.0;
664 min_orth = 180.0;
665
666 double dxi = 1.0 / (nx - 1);
667 double deta = 1.0 / (ny - 1);
668
669 for (int j = 1; j < ny - 1; j++)
670 {
671 for (int i = 1; i < nx - 1; i++)
672 {
673 double x_xi = (x[i + 1][j] - x[i - 1][j]) / (2.0 * dxi);
674 double x_eta = (x[i][j + 1] - x[i][j - 1]) / (2.0 * deta);
675 double y_xi = (y[i + 1][j] - y[i - 1][j]) / (2.0 * dxi);
676 double y_eta = (y[i][j + 1] - y[i][j - 1]) / (2.0 * deta);
677
678 double J = x_xi * y_eta - x_eta * y_xi;
679 if (J < min_jac) min_jac = J;
680
681 double len_xi = sqrt(x_xi * x_xi + y_xi * y_xi);
682 double len_eta = sqrt(x_eta * x_eta + y_eta * y_eta);
683 double ar = (len_xi > len_eta) ? len_xi / max(len_eta, 1e-30)
684 : len_eta / max(len_xi, 1e-30);
685 if (ar > max_ar) max_ar = ar;
686
687 double dot = x_xi * x_eta + y_xi * y_eta;
688 double cos_angle = dot / max(len_xi * len_eta, 1e-30);
689 cos_angle = max(-1.0, min(1.0, cos_angle));
690 double angle_deg = acos(fabs(cos_angle)) * 180.0 / M_PI;
691 if (angle_deg < min_orth) min_orth = angle_deg;
692
693 double skew = fabs(90.0 - angle_deg) / 90.0;
694 if (skew > max_skew) max_skew = skew;
695 }
696 }
697}

748 First layer thickness is length*(1-r)/(1-r^nop) where r=growth_ratio.
749 ====================================================================== */
750void Grid::Stack_Grid(double &length, int &nop, vector<Point> &Stacked_Grid_List,
751 double growth_ratio)
752{
753 cout << "Stacking grid with geometric stretching:\n"
754 << " Plane size: " << Grid_List.size() << "\n"
755 << " Stack length: " << length << "\n"
756 << " Points in Z: " << nop << "\n"
757 << " Growth ratio: " << growth_ratio << endl;
758
759 vector<double> z_coords(nop, 0.0);
760
761 if (fabs(growth_ratio - 1.0) < 1e-10)
762 {
763 double delz = length / (nop - 1);
764 for (int i = 0; i < nop; i++)
765 z_coords[i] = i * delz;

779 for (int i = 1; i < nop; i++)
780 z_coords[i] *= scale;
781 }

784 << " Last layer dz: " << z_coords[nop - 1] - z_coords[nop - 2] << endl;
785
786 Stacked_Grid_List.reserve(Grid_List.size() * nop);
787 Point p;
788 for (int i = 0; i < nop; i++)
789 {
790 for (unsigned int j = 0; j < Grid_List.size(); j++)
791 {

793 z_coords[i] + Grid_List[j].Get_z());

795 }
