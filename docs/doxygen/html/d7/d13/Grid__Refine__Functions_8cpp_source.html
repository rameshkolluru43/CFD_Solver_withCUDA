42// Compute per-cell gradient-based refinement indicator (density + pressure), scale-invariant.
43// indicator = |grad(rho)|*sqrt(Area) + (|grad(P)|/max(P,eps))*sqrt(Area). Stored in Gradient_Refinement_Indicator.
44void Compute_Gradient_Refinement_Indicator()
45{
46 if (Gradient_Refinement_Indicator.size() != static_cast<size_t>(No_Physical_Cells))
47 Gradient_Refinement_Indicator.resize(No_Physical_Cells, 0.0);
48
49 int Grad_Type_Rho = 0, Grad_Type_P = 4;
50 V_D grad(2, 0.0);
51 const double eps = 1e-14;
52
53 for (int i = 0; i < No_Physical_Cells; i++)
54 {
55 double ind = 0.0;
56 const double h = sqrt(Cells[i].Area);
57
58 Calculate_Gradient_At_Cell_Center(i, Grad_Type_Rho, grad);
59 ind += sqrt(grad[0] * grad[0] + grad[1] * grad[1]) * h;
60
61 Calculate_Gradient_At_Cell_Center(i, Grad_Type_P, grad);
62 double P = Primitive_Cells[i][4];
63 if (P > eps)
64 ind += (sqrt(grad[0] * grad[0] + grad[1] * grad[1]) / P) * h;
65
66 Gradient_Refinement_Indicator[i] = ind;
67 }
68}

70// Tag cells for refinement: indicator > threshold, optionally cap by fraction (refine top N% by indicator).
71void TagRefinableCells(vector<Cell> &cells, double &threshold)
72{
73 Compute_Gradient_Refinement_Indicator();
74
75 for (int i = 0; i < No_Physical_Cells; i++)
76 cells[i].Is_Splittable = false;
77
78 if (AMR_Max_Fraction > 0.0 && AMR_Max_Fraction < 1.0)
79 {
80 // Refine top AMR_Max_Fraction of cells by indicator value
81 vector<pair<double, int>> order(No_Physical_Cells);
82 for (int i = 0; i < No_Physical_Cells; i++)
83 order[i] = {Gradient_Refinement_Indicator[i], i};
84 sort(order.begin(), order.end(), [](const pair<double, int> &a, const pair<double, int> &b)
85 { return a.first > b.first; });
86 int nRefine = static_cast<int>(AMR_Max_Fraction * No_Physical_Cells);
87 if (nRefine < 1)
88 nRefine = 1;
89 for (int k = 0; k < nRefine && k < No_Physical_Cells; k++)
90 if (order[k].first >= threshold)
91 cells[order[k].second].Is_Splittable = true;
92 }
93 else
94 {
95 for (int i = 0; i < No_Physical_Cells; i++)
96 if (Gradient_Refinement_Indicator[i] > threshold)
97 cells[i].Is_Splittable = true;
98 }
99}

114// Green-Gauss gradient: grad = (1/Area) * sum_f (phi_face * n_f * dl_f). Works for any polygon.
115void Calculate_Gradient(V_D &av, V_D &nx, V_D &ny, V_D &dl, double &inv_area, V_D &grad)
116{
117 if (grad.empty())
118 grad.resize(2, 0.0);
119 else
120 {
121 grad[0] = 0.0;
122 grad[1] = 0.0;
123 }
124 const int n = static_cast<int>(av.size());
125 for (int i = 0; i < n; ++i)
126 {
127 grad[0] += av[i] * nx[i] * dl[i];
128 grad[1] += av[i] * ny[i] * dl[i];
129 }
130 grad[0] *= inv_area;
131 grad[1] *= inv_area;
132}
133// Function to identify secondary neighbors for second gradients
143void Identify_Neighbours_For_Second_Gradients(int &Current_Cell_Index)
144{
145 /* Secondary_Neighbours = {SW, SE, NE, NW} diagonal stencil for co-volume
146 face gradients. Face map: 0=left, 1=bottom, 2=right, 3=top. */
147 if (Cells[Current_Cell_Index].Secondary_Neighbours.empty())
148 Cells[Current_Cell_Index].Secondary_Neighbours.resize(4, -1);
149 else
150 {
151 for (int i = 0; i < 4; ++i)
152 Cells[Current_Cell_Index].Secondary_Neighbours[i] = -1;
153 }
154
155 const auto &Neighbours = Cells[Current_Cell_Index].Neighbours;
156 if (Neighbours.size() < 4)
157 return;
158
159 const int N_L = Neighbours[0];
160 const int N_B = Neighbours[1];
161 const int N_R = Neighbours[2];
162 const int N_T = Neighbours[3];
163
164 const int n_all = static_cast<int>(Cells.size());
165 auto is_valid = [n_all](int c) -> bool { return c >= 0 && c < n_all; };
166 auto is_phys = [](int c) -> bool { return c >= 0 && c < No_Physical_Cells; };
167
168 /* Neighbour-of-neighbour; never return Current_Cell_Index or -1 silently as 0. */
169 auto getNeighbour = [&](int cellIndex, int direction) -> int {
170 if (!is_valid(cellIndex) || direction < 0 || direction >= 4)
171 return -1;
172 if (static_cast<int>(Cells[cellIndex].Neighbours.size()) <= direction)
173 return -1;
174 const int n = Cells[cellIndex].Neighbours[direction];
175 if (!is_valid(n) || n == Current_Cell_Index)
176 return -1;
177 return n;
178 };
179
180 auto pick = [&](int a, int b) -> int {
181 if (is_valid(a) && a != Current_Cell_Index)
182 return a;
183 if (is_valid(b) && b != Current_Cell_Index)
184 return b;
185 return Current_Cell_Index; /* last resort: collapse stencil */
186 };
187
188 const bool ghost_L = !is_phys(N_L);
189 const bool ghost_B = !is_phys(N_B);
190 const bool ghost_R = !is_phys(N_R);
191 const bool ghost_T = !is_phys(N_T);
192
193 int SW = -1, SE = -1, NE = -1, NW = -1;
194
195 if (ghost_L && ghost_B)
196 {
197 /* Wall/exit bottom-left corner (e.g. half-cylinder cell 0). */
198 SW = pick(N_B, N_L);
199 SE = pick(getNeighbour(N_R, 1), N_B);
200 NE = pick(getNeighbour(N_R, 3), getNeighbour(N_T, 2));
201 NW = pick(getNeighbour(N_T, 0), N_L);
202 }
203 else if (ghost_B && ghost_R)
204 {
205 /* Wall/exit bottom-right corner (e.g. cell 479). */
206 SW = pick(getNeighbour(N_L, 1), N_B);
207 SE = pick(N_B, N_R);
208 NE = pick(getNeighbour(N_T, 2), N_R);
209 NW = pick(getNeighbour(N_L, 3), getNeighbour(N_T, 0));
210 }
211 else if (ghost_R && ghost_T)
212 {
213 /* Inlet/exit top-right corner. */
214 SW = pick(getNeighbour(N_L, 1), getNeighbour(N_B, 0));
215 SE = pick(getNeighbour(N_B, 2), N_R);
216 NE = pick(N_T, N_R);
217 NW = pick(getNeighbour(N_L, 3), N_T);
218 }
219 else if (ghost_L && ghost_T)
220 {
221 /* Inlet/exit top-left corner. */
222 SW = pick(getNeighbour(N_B, 0), N_L);
223 SE = pick(getNeighbour(N_R, 1), getNeighbour(N_B, 2));
224 NE = pick(getNeighbour(N_R, 3), N_T);
225 NW = pick(N_L, N_T);
226 }
227 else
228 {
229 /* Interior / single-face boundary: walk around primary faces. */
230 SW = pick(getNeighbour(N_L, 1), getNeighbour(N_B, 0));
231 SE = pick(getNeighbour(N_B, 2), getNeighbour(N_R, 1));
232 NE = pick(getNeighbour(N_R, 3), getNeighbour(N_T, 2));
233 NW = pick(getNeighbour(N_T, 0), getNeighbour(N_L, 3));
234 }
235
236 Cells[Current_Cell_Index].Secondary_Neighbours[0] = SW;
237 Cells[Current_Cell_Index].Secondary_Neighbours[1] = SE;
238 Cells[Current_Cell_Index].Secondary_Neighbours[2] = NE;
239 Cells[Current_Cell_Index].Secondary_Neighbours[3] = NW;
240}

334void Calculate_Gradients_At_Cell_Centers()
335{
336 // Loop over all physical cells
337 V_D gradient(2, 0.0);
338 for (int Current_Cell_Index = 0; Current_Cell_Index < No_Physical_Cells; ++Current_Cell_Index)
339 {
340 // Calculate gradients for each variable
341 for (int Grad_Type = 0; Grad_Type < 5; ++Grad_Type)
342 {
343 // Calculate gradient at cell center
344 Calculate_Gradient_At_Cell_Center(Current_Cell_Index, Grad_Type, gradient);
345 // Grad_type = 0, density
346 // Grad_type = 1, u velocity
347 // Grad_type = 2, v velocity
348 // Grad_type = 3, temperature
349 // Gradent = 4, pressure
350 // Store gradient in the appropriate cell
351 // without if else store the appropriate gradients in rho_Gradient, u_Gradient, v_Gradient, T_Gradient,P_Gradient
352
353 if (Grad_Type == 0)
354 {
355 Rho_Gradient = gradient;
356 }
357 else if (Grad_Type == 1)
358 {
359 u_Gradient = gradient;
360 }
361 else if (Grad_Type == 2)
362 {
363 v_Gradient = gradient;
364 }
365 else if (Grad_Type == 3)
366 {
367 T_Gradient = gradient;
368 }
369 else if (Grad_Type == 4)
370 {
371 P_Gradient = gradient;
372 }
373 else
374 {
375 cout << "Invalid Grad_Type: " << Grad_Type << endl;
376 throw runtime_error("Calculate_Green_Gauss_Gradient: invalid Grad_Type " + to_string(Grad_Type));
377 }
378 }
379 }

382// Green-Gauss gradient at cell center: phi_face = 0.5*(phi_c + phi_neighbor), grad = (1/Area)*sum_f (phi_face * n_f * dl_f).
383// Works for any polygon (tri/quad); uses this cell's Face_Normals and Face_Areas.
384void Calculate_Gradient_At_Cell_Center(int &Current_Cell_Index, int &Grad_Type, V_D &grad)
385{
386 grad.assign(2, 0.0);
387 const Cell &cell = Cells[Current_Cell_Index];
388 const int nF = (cell.numFaces > 0) ? cell.numFaces : static_cast<int>(cell.Face_Areas.size());
389 if (nF <= 0 || cell.Area <= 0.0)
390 return;
391
392 const double phi_c = Primitive_Cells[Current_Cell_Index][Grad_Type];
393 for (int f = 0; f < nF; f++)
394 {
395 const int idx = f * 2;
396 const double nx_f = cell.Face_Normals[idx + 0];
397 const double ny_f = cell.Face_Normals[idx + 1];
398 const double dl_f = cell.Face_Areas[f];
399 int neigh = (f < static_cast<int>(cell.Neighbours.size())) ? cell.Neighbours[f] : -1;
400 double phi_n = phi_c;
401 if (neigh >= 0 && neigh < No_Physical_Cells)
402 phi_n = Primitive_Cells[neigh][Grad_Type];
403 const double phi_face = 0.5 * (phi_c + phi_n);
404 grad[0] += phi_face * nx_f * dl_f;
405 grad[1] += phi_face * ny_f * dl_f;
406 }
407 grad[0] *= cell.Inv_Area;
408 grad[1] *= cell.Inv_Area;
409}

542 // Vertices of the face are b and c from the primary cell, with cell center as current cell
543 // Current cell, Neighbour 3, Neighbour 4, Neighbour 7
544 b = Calculate_Vertex_Average({weights[0], weights[3], weights[4], weights[7]}, {cell_averages[0], cell_averages[3], cell_averages[4], cell_averages[7]});
545 // Current cell, Neighbour 1, Neighbour 4 , Neighbour 8
546 c = Calculate_Vertex_Average({weights[0], weights[1], weights[4], weights[8]}, {cell_averages[0], cell_averages[1], cell_averages[4], cell_averages[8]});
547 // Calculate face averages from the verticies of co volume cells
548 av[0] = 0.5 * (cell_averages[0] + b); // Average for Face 0
549 av[1] = 0.5 * (cell_averages[0] + c); // Average for Face 1
550 av[2] = 0.5 * (cell_averages[4] + c); // Average for Face 2
551 av[3] = 0.5 * (cell_averages[4] + b); // Average for Face 3
552 break;
553 }
554
555 V_D gradient(2, 0.0);
556 // Calculate gradients
557 Calculate_Gradient(av, nx, ny, dl, inv_area, gradient);
558
559 // Assign gradients based on Grad_Type
560 switch (Grad_Type)
561 {
562 case 0:
563 Rho_Gradient[0] = gradient[0];
564 Rho_Gradient[1] = gradient[1];
565 break;
566 case 1:
567 u_Gradient[0] = gradient[0];
568 u_Gradient[1] = gradient[1];
569 break;
570 case 2:
571 v_Gradient[0] = gradient[0];
572 v_Gradient[1] = gradient[1];
573 break;
574 case 3:
575 T_Gradient[0] = gradient[0];
576 T_Gradient[1] = gradient[1];
577 break;
578 case 4:
579 P_Gradient[0] = gradient[0];
580 P_Gradient[1] = gradient[1];
581 break;
582 }
583}

585void Viscous_Flux_on_Face(const int &Cell_No, const int &Face_No)
586{
587 // cout<<"Calculating Viscous flux for Cell No\t"<<Cell_No<<"\t on face \t"<<Face_No<<endl;
588 double u11, u12, u21, u22, T11, T12, T21, T22;
589 double v1 = 0.0, v2 = 0.0, mu = 0.0;
590 int index = Face_No * 2, Grad_Type = 0, N_Cell_No = 0;
591
592 Qx = 0.0;
593 Qy = 0.0;
594
595 // Normals and Face Area (in 2D Length of the face)
596
597 nx = Cells[Cell_No].Face_Normals[index];
598 ny = Cells[Cell_No].Face_Normals[index + 1];
599 dl = Cells[Cell_No].Face_Areas[Face_No];
600
601 // Fetching the Corresponding Neighbours for the given cell and given face
602 N_Cell_No = Cells[Cell_No].Neighbours[Face_No];
603
604 // Find the arthematic average of the velocities and Temperature on the Face
605
606 v1 = 0.5 * (Primitive_Cells[Cell_No][1] + Primitive_Cells[N_Cell_No][1]);
607 v2 = 0.5 * (Primitive_Cells[Cell_No][2] + Primitive_Cells[N_Cell_No][2]);
608 mu = 0.5 * (Primitive_Cells[Cell_No][8] + Primitive_Cells[N_Cell_No][8]);
609
610 // cout<<"Velocities on Face\t"<<v1<<"\t"<<v2<<endl;
611
612 Grad_Type = 1; // U velocity Gradient // Finds the u Velocity gradient on the face required
613 Calculate_Gradient_On_Face(Cell_No, Grad_Type, Face_No);
614 u11 = u_Gradient[0];
615 u12 = u_Gradient[1];
616 Grad_Type = 2; // V Velocity Gradient
617 Calculate_Gradient_On_Face(Cell_No, Grad_Type, Face_No);
618 u21 = v_Gradient[0];
619 u22 = v_Gradient[1];
620 Grad_Type = 3; // Temperature Gradient
621 Calculate_Gradient_On_Face(Cell_No, Grad_Type, Face_No);
622 // cout<<"Gradients\t"<<Cell_No<<"\t"<<u11<<"\t"<<u12<<"\t"<<u21<<"\t"<<u22<<"\t"<<T_Gradient[0]<<"\t"<<T_Gradient[1]<<endl;
623 // cout<<T<<"\t"<<mu<<"\t"<<Re<<"\t"<<Pr<<endl;
624 T11 = (2.0 / 3.0) * mu * Inv_Re * (2.0 * u11 - u22);
625 T12 = mu * Inv_Re * (u12 + u21);
626 T21 = T12;
627 T22 = (2.0 / 3.0) * mu * Inv_Re * (2.0 * u22 - u11);
628 // cout<<T11<<"\t"<<T12<<"\t"<<T21<<"\t"<<T22<<endl;
629 // K1 = mu/(gamma_M_1*M_ref*M_ref*Re*Pr);
630 Qx = K1 * T_Gradient[0];
631 Qy = K1 * T_Gradient[1];
632
633 Cells_Viscous_Flux[Cell_No][0] += 0.0;
634 Cells_Viscous_Flux[Cell_No][1] += (T11 * nx + T21 * ny) * dl; // Ti1Ai
635 Cells_Viscous_Flux[Cell_No][2] += (T12 * nx + T22 * ny) * dl; // Ti2Ai
636 Cells_Viscous_Flux[Cell_No][3] += ((T11 * v1 + T12 * v2 + Qx) * nx * dl + (T21 * v1 + T22 * v2 + Qy) * ny * dl);
637
638 if (isnan(T11) or isnan(T21) or isnan(T22) or isnan(Qx) or isnan(Qy))
639 {
640
641 cout << Cell_No << "\t" << Face_No << "\t" << T11 << "\t" << T12 << "\t" << T21 << "\t" << T22 << "\t" << Qx << "\t" << Qy << endl;
642 cout << mu << "\t" << Inv_Re << "\t" << K1 << endl;
643 cout << u11 << "\t" << u12 << "\t" << u21 << "\t" << u22 << endl;
644 throw runtime_error("Viscous flux calculation produced non-finite values");
645 }
646}

649void Evaluate_Viscous_Fluxes()
650{
651 /* Refresh SGS viscosity into Primitive_Cells[*][8] before host/CUDA viscous flux. */
652 if (Is_LES)
653 Solve_LES_Step(Min_dt);
654
655#ifdef USE_CUDA
656 if (Evaluate_Viscous_Fluxes_CUDA())
657 return;
658#endif
659
660 // cout<< "Evaluating Viscous Fluxes\n";
661 for (int Current_Cell_Index = 0; Current_Cell_Index < No_Physical_Cells; Current_Cell_Index++)
662 {
663 // Resetting all viscous flux to zero
664 for (int i = 0; i < Cells_Viscous_Flux[Current_Cell_Index].size(); i++)
665 {
666 Cells_Viscous_Flux[Current_Cell_Index][i] = 0.0;
667 }

677// Apply gradient-based adaptive refinement: compute indicator, tag cells. Returns true if mesh was changed (future: actual refinement).
678bool Apply_Adaptive_Refinement()
679{
680 TagRefinableCells(Cells, AMR_Gradient_Threshold);
681 int nTagged = 0;
682 for (int i = 0; i < No_Physical_Cells; i++)
683 if (Cells[i].Is_Splittable)
684 nTagged++;
685 if (nTagged <= 0)
686 return false;
687
688 cout << "AMR: " << nTagged << " cells tagged for refinement (gradient threshold=" << AMR_Gradient_Threshold << ")" << endl;
689
690 // Refine by switching to the next available structured grid level (Nx,Ny)->(2Nx-1,2Ny-1)
691 // and transferring U by nearest-cell interpolation.
692 static int amr_refinements_done = 0;
693 const int AMR_MAX_REFINEMENTS = 2;
694 if (amr_refinements_done >= AMR_MAX_REFINEMENTS)
695 {
696 cout << "AMR: max refinement levels reached (" << AMR_MAX_REFINEMENTS << ")." << endl;
697 return false;
698 }
699
700 const int old_nx = meshParams.nx;
701 const int old_ny = meshParams.ny;
702 const int new_nx = 2 * old_nx - 1;
703 const int new_ny = 2 * old_ny - 1;
704
705 std::regex re("_(\\d+)_(\\d+)\\.(txt|vtk)$");
706 std::smatch m;
707 std::string candidate = Grid_File;
708 if (std::regex_search(Grid_File, m, re))
709 {
710 candidate = std::regex_replace(Grid_File, re, "_" + std::to_string(new_nx) + "_" + std::to_string(new_ny) + "." + m[3].str());
711 }
712 else
713 {
714 cout << "AMR: unable to infer next grid filename from " << Grid_File << endl;
715 return false;
716 }
717
718 if (!std::filesystem::exists(candidate))
719 {
720 cout << "AMR: no finer grid file found: " << candidate << endl;
721 return false;
722 }
723
724 // Backup old solution on physical cells.
725 const int old_n_phys = No_Physical_Cells;
726 vector<V_D> old_centers(old_n_phys, V_D(2, 0.0));
727 vector<V_D> old_u(old_n_phys, V_D(4, 0.0));
728 for (int i = 0; i < old_n_phys; ++i)
729 {
730 old_centers[i] = Cells[i].Cell_Center;
731 old_u[i] = U_Cells[i];
732 }
733
734 cout << "AMR: refining grid " << old_nx << "x" << old_ny << " -> " << new_nx << "x" << new_ny << endl;
735
736 // Rebuild mesh on finer grid.
737 Grid_File = candidate;
738 meshParams.nx = new_nx;
739 meshParams.ny = new_ny;
740 Cells.clear();
741 Boundary_Cells.clear();
742 Co_Volume_Cells.clear();
743 Total_No_Cells = 0;
744 No_Physical_Cells = 0;
745 No_Ghost_Cells = 0;
746 Inlet_Cells_List.clear();
747 Exit_Cells_List.clear();
748 Wall_Cells_List.clear();
749 Symmetry_Cells_List.clear();
750 if (!Form_Cells(Grid_File))
751 {
752 cout << "AMR: Form_Cells failed on refined grid, keeping current grid." << endl;
753 return false;
754 }
755
756 // Reinitialize solver storage for new grid.
757 Cells_Net_Flux.clear();
758 Cells_DelU.clear();
759 Cells_Face_Boundary_Type.clear();
760 U_Cells.clear();
761 U_Cells_RK_1.clear();
762 U_Cells_RK_2.clear();
763 Primitive_Cells.clear();
764 Cells_Viscous_Flux.clear();
765 CF.clear();
766 Initialize(Test_Case);
767
768 // Nearest-cell interpolation of conservative variables onto refined physical cells.
769 for (int i = 0; i < No_Physical_Cells; ++i)
770 {
771 const double cx = Cells[i].Cell_Center[0];
772 const double cy = Cells[i].Cell_Center[1];
773 int best = 0;
774 double best_d2 = std::numeric_limits<double>::max();
775 for (int j = 0; j < old_n_phys; ++j)
776 {
777 const double dx = cx - old_centers[j][0];
778 const double dy = cy - old_centers[j][1];
779 const double d2 = dx * dx + dy * dy;
780 if (d2 < best_d2)
781 {
782 best_d2 = d2;
783 best = j;
784 }
785 }
786 U_Cells[i] = old_u[best];
787 Calculate_Primitive_Variables(i, U_Cells[i], Primitive_Cells[i]);
788 }
789
790 amr_refinements_done++;
791 cout << "AMR: refinement applied using grid file " << Grid_File << endl;
792 return true;
793}
