66/* Van Leer (1974) harmonic mean: more compressive than MinMod, smooth ψ. */
67void VanLeerSlope2(double a, double b, double &phi)
68{
69 const double ab = a * b;
70 if (ab <= 0.0)
71 {
72 phi = 0.0;
73 return;
74 }
75 const double den = a + b;
76 if (fabs(den) < 1e-30)
77 phi = 0.0;
78 else
79 phi = 2.0 * ab / den;
80}

82/* Superbee: most compressive second-order TVD limiter (steeper shocks, can add noise). */
83void SuperbeeSlope2(double a, double b, double &phi)
84{
85 if (a * b <= 0.0)
86 {
87 phi = 0.0;
88 return;
89 }
90 const double s = (a > 0.0) ? 1.0 : -1.0;
91 const double aa = fabs(a), bb = fabs(b);
92 phi = s * max(min(2.0 * aa, bb), min(aa, 2.0 * bb));
93}

95/* van Albada (1982): smooth, between MinMod and Superbee — good accuracy on smooth flows. */
96void VanAlbadaSlope2(double a, double b, double &phi)
97{
98 if (a * b <= 0.0)
99 {
100 phi = 0.0;
101 return;
102 }
103 const double a2 = a * a, b2 = b * b;
104 const double den = a2 + b2;
105 if (den < 1e-30)
106 phi = 0.0;
107 else
108 phi = (a * b2 + b * a2) / den;
109}

144bool least_squares_gradient(const int cell_index, const int var, double &gx, double &gy)
145{
146 gx = 0.0;
147 gy = 0.0;
148 if (!valid_cell_index(cell_index))
149 return false;
150
151 double a00 = 0.0, a01 = 0.0, a11 = 0.0;
152 double b0 = 0.0, b1 = 0.0;
153 const Cell &cell = Cells[cell_index];
154 const double u0 = U_Cells[cell_index][var];
155 const int n_faces = std::min(static_cast<int>(cell.Neighbours.size()), cell.numFaces > 0 ? cell.numFaces : static_cast<int>(cell.Neighbours.size()));
156
157 for (int f = 0; f < n_faces; ++f)
158 {
159 const int nb = cell.Neighbours[f];
160 if (!valid_cell_index(nb))
161 continue;
162
163 double dx = 0.0, dy = 0.0;
164 center_to_neighbor_vector(cell_index, f, dx, dy);
165 const double r2 = dx * dx + dy * dy;
166 if (r2 < kSmall)
167 continue;
168
169 const double w = 1.0 / r2;
170 const double du = U_Cells[nb][var] - u0;
171 a00 += w * dx * dx;
172 a01 += w * dx * dy;
173 a11 += w * dy * dy;
174 b0 += w * dx * du;
175 b1 += w * dy * du;
176 }
177
178 const double det = a00 * a11 - a01 * a01;
179 if (fabs(det) < 1.0e-30)
180 return false;
181
182 gx = (b0 * a11 - b1 * a01) / det;
183 gy = (a00 * b1 - a01 * b0) / det;
184 return true;
185}

206double bounded_tvd_phi(const int cell_index, const int var, const double delta)
207{
208 if (fabs(delta) < kSmall)
209 return 1.0;
210
211 double u_min = 0.0, u_max = 0.0;
212 neighbor_bounds(cell_index, var, u_min, u_max);
213 const double u0 = U_Cells[cell_index][var];
214 const double bound = (delta > 0.0) ? (u_max - u0) : (u_min - u0);
215 const double r = std::max(0.0, bound / delta);
216 double phi = std::min(1.0, r);
217
218 switch (Limiter_Case)
219 {
220 case 1: // Superbee
221 phi = std::max(std::min(2.0 * r, 1.0), std::min(r, 2.0));
222 break;
223 case 3: // van Leer
224 phi = (r > 0.0) ? (2.0 * r / (1.0 + r)) : 0.0;
225 break;
226 case 4: // van Albada
227 phi = (r > 0.0) ? ((r * r + r) / (r * r + 1.0)) : 0.0;
228 break;
229 default: // MinMod / MCD / fallback
230 break;
231 }
232
233 // Preserve the local extrema bound even for compressive limiter curves.
234 return std::max(0.0, std::min(phi, std::min(1.0, r)));
235}

242double Venkatakrishnan_Phi_MUSCL2D(int cell_index, int var, double Delta_plus)
243{
244 if (fabs(Delta_plus) < kSmall)
245 return 1.0;
246
247 double u_min = 0.0, u_max = 0.0;
248 neighbor_bounds(cell_index, var, u_min, u_max);
249 const double u_c = U_Cells[cell_index][var];
250 const double y = (Delta_plus > 0.0) ? (u_max - u_c) : (u_min - u_c);
251 double h2 = Cells[cell_index].Area;
252 if (h2 <= 0.0)
253 h2 = Cell_Minimum_Length * Cell_Minimum_Length;
254 if (h2 <= 0.0)
255 h2 = 1.0;
256
257 const double eps2 = Venkat_K * Venkat_K * Venkat_K * h2 * h2;
258 const double num = y * y + eps2 + 2.0 * y * Delta_plus;
259 const double den = y * y + 2.0 * Delta_plus * Delta_plus + y * Delta_plus + eps2;
260 if (fabs(den) < 1.0e-30)
261 return 1.0;
262 return std::max(0.0, std::min(1.0, num / den));
263}
