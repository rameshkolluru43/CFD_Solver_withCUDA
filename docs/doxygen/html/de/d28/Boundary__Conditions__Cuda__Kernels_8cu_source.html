28__device__ void calculate_primitive_device(
29 const double* U,
30 double* primitive)
31{
32 double rho = U[0];
33 if (rho < 1e-12) rho = 1e-12;
34
35 double u = U[1] / rho;
36 double v = U[2] / rho;
37 double E = U[3];
38
39 double kinetic_energy = 0.5 * rho * (u * u + v * v);
40 double p = GAMMA_M1 * (E - kinetic_energy);
41 if (p < 1e-12) p = 1e-12;
42
43 double T = p / (rho * R_GC);
44
45 primitive[0] = rho;
46 primitive[1] = u;
47 primitive[2] = v;
48 primitive[3] = p;
49 primitive[4] = T;
50}

53 const double* primitive,
54 double* U)
55{
56 double rho = primitive[0];

58 double v = primitive[2];
59 double p = primitive[3];
60
61 U[0] = rho;
62 U[1] = rho * u;
63 U[2] = rho * v;
64 U[3] = p / GAMMA_M1 + 0.5 * rho * (u * u + v * v);

257__global__ void farfield_bc_kernel(
258 double *U_cells,
259 const int *farfield_cell_list,
260 const double *face_normals,
261 const FarfieldCondition_GPU bc_data,
262 int num_farfield_cells,
263 double gamma)
264{
265 int idx = blockIdx.x * blockDim.x + threadIdx.x;
266 if (idx >= num_farfield_cells) return;
267
268 int cell = farfield_cell_list[idx];
269 int base = cell * 4;
270
271 double prim[5];
272 calculate_primitive_device(&U_cells[base], prim);
273
274 double rho_i = prim[0];
275 double u_i = prim[1];
276 double v_i = prim[2];
277 double p_i = prim[3];
278
279 double rho_inf = bc_data.Rho_inf;
280 double u_inf = bc_data.u_inf;
281 double v_inf = bc_data.v_inf;
282 double p_inf = bc_data.P_inf;
283
284 double nx = face_normals[cell * 8 + 0];
285 double ny = face_normals[cell * 8 + 1];
286
287 double a_i = sqrt(GAMMA_VAL * p_i / rho_i);
288 double a_inf = sqrt(GAMMA_VAL * p_inf / rho_inf);
289
290 double v_n_i = u_i * nx + v_i * ny;
291 double v_n_inf = u_inf * nx + v_inf * ny;
292
293 double R_plus = v_n_i + 2.0 * a_i / GAMMA_M1;
294 double R_minus = v_n_inf - 2.0 * a_inf / GAMMA_M1;
295
296 double v_n_b = 0.5 * (R_plus + R_minus);
297 double a_b = 0.25 * GAMMA_M1 * (R_plus - R_minus);
298 if (a_b < 1e-12) a_b = 1e-12;
299
300 double rho_b, u_b, v_b, p_b;
301
302 if (v_n_b >= 0.0) {
303 /* Outflow: entropy from interior */
304 double s_i = p_i / pow(rho_i, GAMMA_VAL);
305 rho_b = pow(a_b * a_b / (GAMMA_VAL * s_i), 1.0 / GAMMA_M1);
306 u_b = u_i + (v_n_b - v_n_i) * nx;
307 v_b = v_i + (v_n_b - v_n_i) * ny;
308 p_b = rho_b * a_b * a_b / GAMMA_VAL;
309 } else {
310 /* Inflow: entropy from freestream */
311 double s_inf = p_inf / pow(rho_inf, GAMMA_VAL);
312 rho_b = pow(a_b * a_b / (GAMMA_VAL * s_inf), 1.0 / GAMMA_M1);
313 u_b = u_inf + (v_n_b - v_n_inf) * nx;
314 v_b = v_inf + (v_n_b - v_n_inf) * ny;
315 p_b = rho_b * a_b * a_b / GAMMA_VAL;
316 }
317
318 double v_mag_sq = u_b * u_b + v_b * v_b;
319
320 U_cells[base + 0] = rho_b;
321 U_cells[base + 1] = rho_b * u_b;
322 U_cells[base + 2] = rho_b * v_b;
323 U_cells[base + 3] = p_b / GAMMA_M1 + 0.5 * rho_b * v_mag_sq;
324}
