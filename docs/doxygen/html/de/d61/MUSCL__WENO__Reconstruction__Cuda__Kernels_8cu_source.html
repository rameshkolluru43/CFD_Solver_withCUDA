29__global__ void muscl_reconstruction_kernel(
30 const double* U_cells,
31 const int* cell_neighbors,
32 const double* cell_distances,
33 const double* limited_gradients, // From limiter kernels
34 double* Q_left, // Output: left states at faces
35 double* Q_right, // Output: right states at faces
36 int num_cells,
37 double kappa // MUSCL parameter: -1 (2nd upwind), 0 (Fromm), 0.5 (Quick), 1 (central)
38) {
39 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
40 if (cell_idx >= num_cells) return;
41
42 for (int face = 0; face < 4; face++) {
43 int neighbor_idx = cell_neighbors[cell_idx * 4 + face];
44 if (neighbor_idx < 0 || neighbor_idx >= num_cells) {
45 // Boundary face - use cell-centered values
46 for (int var = 0; var < 4; var++) {
47 int face_offset = (cell_idx * 4 + face) * 4 + var;
48 Q_left[face_offset] = U_cells[cell_idx * 4 + var];
49 Q_right[face_offset] = U_cells[cell_idx * 4 + var];
50 }
51 continue;
52 }
53
54 double distance = cell_distances[cell_idx * 4 + face];
55
56 for (int var = 0; var < 4; var++) {
57 // Get cell values
58 double U_i = U_cells[cell_idx * 4 + var];
59 double U_j = U_cells[neighbor_idx * 4 + var];
60
61 // Get limited gradients (phi already applied)
62 int grad_offset_L = (cell_idx * 4 + face) * 4 + var;
63 int grad_offset_R = (neighbor_idx * 4 + face) * 4 + var;
64
65 double phi_L = limited_gradients[grad_offset_L];
66 double phi_R = limited_gradients[grad_offset_R];
67
68 // MUSCL reconstruction
69 // Left state (from current cell)
70 double slope_L = (U_j - U_i) / distance;
71 double Q_L = U_i + 0.5 * kappa * phi_L * slope_L * distance;
72
73 // Right state (from neighbor cell)
74 double slope_R = (U_i - U_j) / distance;
75 double Q_R = U_j + 0.5 * kappa * phi_R * slope_R * distance;
76
77 // Store reconstructed values
78 int face_offset = (cell_idx * 4 + face) * 4 + var;
79 Q_left[face_offset] = Q_L;
80 Q_right[face_offset] = Q_R;
81 }
82 }
83}

92__device__ double weno5_smoothness_indicator_device(
93 double u1, double u2, double u3
94) {
95 double term1 = u1 - 2.0 * u2 + u3;
96 double term2 = u1 - 4.0 * u2 + 3.0 * u3;
97 return (13.0 / 12.0) * term1 * term1 + 0.25 * term2 * term2;
98}

126__global__ void weno5_reconstruction_kernel(
127 const double* U_cells,
128 const int* cell_neighbors,
129 double* Q_left,
130 double* Q_right,
131 int num_cells,
132 double epsilon,
133 int p
134) {
135 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
136 if (cell_idx >= num_cells) return;
137
138 for (int face = 0; face < 4; face++) {
139 int stencil_base = (cell_idx * 4 + face) * 5;
140 int idx_m2 = cell_neighbors[stencil_base + 0]; // i-2
141 int idx_m1 = cell_neighbors[stencil_base + 1]; // i-1
142 int idx_0 = cell_idx; // i
143 int idx_p1 = cell_neighbors[stencil_base + 2]; // i+1
144 int idx_p2 = cell_neighbors[stencil_base + 3]; // i+2
145 int idx_p3 = cell_neighbors[stencil_base + 4]; // i+3
146
147 bool valid_left = (idx_m2 >= 0 && idx_m2 < num_cells &&
148 idx_m1 >= 0 && idx_m1 < num_cells &&
149 idx_p1 >= 0 && idx_p1 < num_cells &&
150 idx_p2 >= 0 && idx_p2 < num_cells);
151 bool valid_right = (idx_m1 >= 0 && idx_m1 < num_cells &&
152 idx_p1 >= 0 && idx_p1 < num_cells &&
153 idx_p2 >= 0 && idx_p2 < num_cells &&
154 idx_p3 >= 0 && idx_p3 < num_cells);
155
156 if (!valid_left && !valid_right) {
157 for (int var = 0; var < 4; var++) {
158 int face_offset = (cell_idx * 4 + face) * 4 + var;
159 Q_left[face_offset] = U_cells[cell_idx * 4 + var];
160 Q_right[face_offset] = (idx_p1 >= 0 && idx_p1 < num_cells)
161 ? U_cells[idx_p1 * 4 + var]
162 : U_cells[cell_idx * 4 + var];
163 }
164 continue;
165 }
166
167 for (int var = 0; var < 4; var++) {
168 double u_0 = U_cells[idx_0 * 4 + var];
169
170 // ============ LEFT STATE (u^- at i+1/2) ============
171 // Stencil: {i-2, i-1, i, i+1, i+2}
172 double Q_L;
173 if (valid_left) {
174 double u_m2 = U_cells[idx_m2 * 4 + var];
175 double u_m1 = U_cells[idx_m1 * 4 + var];
176 double u_p1 = U_cells[idx_p1 * 4 + var];
177 double u_p2 = U_cells[idx_p2 * 4 + var];
178
179 double v0_L = ( 2.0 * u_0 + 5.0 * u_p1 - u_p2) / 6.0;
180 double v1_L = ( -u_m1 + 5.0 * u_0 + 2.0 * u_p1) / 6.0;
181 double v2_L = ( 2.0 * u_m2 - 7.0 * u_m1 + 11.0 * u_0) / 6.0;
182
183 double beta0_L = weno5_smoothness_indicator_device(u_0, u_p1, u_p2);
184 double beta1_L = weno5_smoothness_indicator_device(u_m1, u_0, u_p1);
185 double beta2_L = weno5_smoothness_indicator_device(u_m2, u_m1, u_0);
186
187 double alpha0_L = (3.0 / 10.0) / pow(epsilon + beta0_L, p);
188 double alpha1_L = (6.0 / 10.0) / pow(epsilon + beta1_L, p);
189 double alpha2_L = (1.0 / 10.0) / pow(epsilon + beta2_L, p);
190 double sum_L = alpha0_L + alpha1_L + alpha2_L;
191
192 Q_L = (alpha0_L * v0_L + alpha1_L * v1_L + alpha2_L * v2_L) / sum_L;
193 if (!isfinite(Q_L)) Q_L = u_0;
194 } else {
195 Q_L = u_0;
196 }
197
198 // ============ RIGHT STATE (u^+ at i+1/2) ============
199 // Stencil: {i-1, i, i+1, i+2, i+3}
200 double Q_R;
201 if (valid_right) {
202 double u_m1 = U_cells[idx_m1 * 4 + var];
203 double u_p1 = U_cells[idx_p1 * 4 + var];
204 double u_p2 = U_cells[idx_p2 * 4 + var];
205 double u_p3 = U_cells[idx_p3 * 4 + var];
206
207 double v0_R = (11.0 * u_p1 - 7.0 * u_p2 + 2.0 * u_p3) / 6.0;
208 double v1_R = ( 2.0 * u_0 + 5.0 * u_p1 - u_p2) / 6.0;
209 double v2_R = ( -u_m1 + 5.0 * u_0 + 2.0 * u_p1) / 6.0;
210
211 double beta0_R = weno5_smoothness_indicator_device(u_p1, u_p2, u_p3);
212 double beta1_R = weno5_smoothness_indicator_device(u_0, u_p1, u_p2);
213 double beta2_R = weno5_smoothness_indicator_device(u_m1, u_0, u_p1);
214
215 double alpha0_R = (1.0 / 10.0) / pow(epsilon + beta0_R, p);
216 double alpha1_R = (6.0 / 10.0) / pow(epsilon + beta1_R, p);
217 double alpha2_R = (3.0 / 10.0) / pow(epsilon + beta2_R, p);
218 double sum_R = alpha0_R + alpha1_R + alpha2_R;
219
220 Q_R = (alpha0_R * v0_R + alpha1_R * v1_R + alpha2_R * v2_R) / sum_R;
221 if (!isfinite(Q_R)) Q_R = u_p1;
222 } else {
223 Q_R = (idx_p1 >= 0 && idx_p1 < num_cells)
224 ? U_cells[idx_p1 * 4 + var] : u_0;
225 }
226
227 int face_offset = (cell_idx * 4 + face) * 4 + var;
228 Q_left[face_offset] = Q_L;
229 Q_right[face_offset] = Q_R;
230 }
231 }
232}

251__global__ void weno5_characteristic_reconstruction_kernel(
252 const double* U_cells,
253 const int* cell_neighbors,
254 const double* L_matrices,
255 const double* R_matrices,
256 double* Q_left,
257 double* Q_right,
258 int num_cells,
259 double epsilon,
260 int p
261) {
262 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
263 if (cell_idx >= num_cells) return;
264
265 for (int face = 0; face < 4; face++) {
266 int stencil_base = (cell_idx * 4 + face) * 5;
267 int idx_m2 = cell_neighbors[stencil_base + 0];
268 int idx_m1 = cell_neighbors[stencil_base + 1];
269 int idx_0 = cell_idx;
270 int idx_p1 = cell_neighbors[stencil_base + 2];
271 int idx_p2 = cell_neighbors[stencil_base + 3];
272 int idx_p3 = cell_neighbors[stencil_base + 4];
273
274 bool valid_left = (idx_m2 >= 0 && idx_m2 < num_cells &&
275 idx_m1 >= 0 && idx_m1 < num_cells &&
276 idx_p1 >= 0 && idx_p1 < num_cells &&
277 idx_p2 >= 0 && idx_p2 < num_cells);
278 bool valid_right = (idx_m1 >= 0 && idx_m1 < num_cells &&
279 idx_p1 >= 0 && idx_p1 < num_cells &&
280 idx_p2 >= 0 && idx_p2 < num_cells &&
281 idx_p3 >= 0 && idx_p3 < num_cells);
282
283 if (!valid_left && !valid_right) {
284 int face_offset = (cell_idx * 4 + face) * 4;
285 for (int var = 0; var < 4; var++) {
286 Q_left[face_offset + var] = U_cells[cell_idx * 4 + var];
287 Q_right[face_offset + var] = (idx_p1 >= 0 && idx_p1 < num_cells)
288 ? U_cells[idx_p1 * 4 + var]
289 : U_cells[cell_idx * 4 + var];
290 }
291 continue;
292 }
293
294 int matrix_offset = (cell_idx * 4 + face) * 16;
295
296 // Transform all stencil points to characteristic space
297 double w_m2[4] = {0}, w_m1[4] = {0}, w_0[4] = {0};
298 double w_p1[4] = {0}, w_p2[4] = {0}, w_p3[4] = {0};
299
300 for (int i = 0; i < 4; i++) {
301 for (int j = 0; j < 4; j++) {
302 double L_ij = L_matrices[matrix_offset + i * 4 + j];
303 w_0[i] += L_ij * U_cells[idx_0 * 4 + j];
304 if (valid_left) {
305 w_m2[i] += L_ij * U_cells[idx_m2 * 4 + j];
306 }
307 if (valid_left || valid_right) {
308 w_m1[i] += L_ij * U_cells[idx_m1 * 4 + j];
309 w_p1[i] += L_ij * U_cells[idx_p1 * 4 + j];
310 w_p2[i] += L_ij * U_cells[idx_p2 * 4 + j];
311 }
312 if (valid_right) {
313 w_p3[i] += L_ij * U_cells[idx_p3 * 4 + j];
314 }
315 }
316 }
317
318 double w_L[4], w_R[4];
319
320 for (int var = 0; var < 4; var++) {
321 // ============ LEFT STATE ============
322 // Stencil: {w_m2, w_m1, w_0, w_p1, w_p2}
323 if (valid_left) {
324 double v0_L = ( 2.0 * w_0[var] + 5.0 * w_p1[var] - w_p2[var]) / 6.0;
325 double v1_L = ( -w_m1[var] + 5.0 * w_0[var] + 2.0 * w_p1[var]) / 6.0;
326 double v2_L = ( 2.0 * w_m2[var] - 7.0 * w_m1[var] + 11.0 * w_0[var]) / 6.0;
327
328 double b0_L = weno5_smoothness_indicator_device(w_0[var], w_p1[var], w_p2[var]);
329 double b1_L = weno5_smoothness_indicator_device(w_m1[var], w_0[var], w_p1[var]);
330 double b2_L = weno5_smoothness_indicator_device(w_m2[var], w_m1[var], w_0[var]);
331
332 double a0_L = (3.0 / 10.0) / pow(epsilon + b0_L, p);
333 double a1_L = (6.0 / 10.0) / pow(epsilon + b1_L, p);
334 double a2_L = (1.0 / 10.0) / pow(epsilon + b2_L, p);
335 double sum_L = a0_L + a1_L + a2_L;
336
337 w_L[var] = (a0_L * v0_L + a1_L * v1_L + a2_L * v2_L) / sum_L;
338 } else {
339 w_L[var] = w_0[var];
340 }
341
342 // ============ RIGHT STATE ============
343 // Stencil: {w_m1, w_0, w_p1, w_p2, w_p3}
344 if (valid_right) {
345 double v0_R = (11.0 * w_p1[var] - 7.0 * w_p2[var] + 2.0 * w_p3[var]) / 6.0;
346 double v1_R = ( 2.0 * w_0[var] + 5.0 * w_p1[var] - w_p2[var]) / 6.0;
347 double v2_R = ( -w_m1[var] + 5.0 * w_0[var] + 2.0 * w_p1[var]) / 6.0;
348
349 double b0_R = weno5_smoothness_indicator_device(w_p1[var], w_p2[var], w_p3[var]);
350 double b1_R = weno5_smoothness_indicator_device(w_0[var], w_p1[var], w_p2[var]);
351 double b2_R = weno5_smoothness_indicator_device(w_m1[var], w_0[var], w_p1[var]);
352
353 double a0_R = (1.0 / 10.0) / pow(epsilon + b0_R, p);
354 double a1_R = (6.0 / 10.0) / pow(epsilon + b1_R, p);
355 double a2_R = (3.0 / 10.0) / pow(epsilon + b2_R, p);
356 double sum_R = a0_R + a1_R + a2_R;
357
358 w_R[var] = (a0_R * v0_R + a1_R * v1_R + a2_R * v2_R) / sum_R;
359 } else {
360 w_R[var] = w_p1[var];
361 }
362 }
363
364 // Transform back to conservative variables
365 double QL[4] = {0}, QR[4] = {0};
366 for (int i = 0; i < 4; i++) {
367 for (int j = 0; j < 4; j++) {
368 double R_ij = R_matrices[matrix_offset + i * 4 + j];
369 QL[i] += R_ij * w_L[j];
370 QR[i] += R_ij * w_R[j];
371 }
372 }
373
374 int face_offset = (cell_idx * 4 + face) * 4;
375 for (int var = 0; var < 4; var++) {
376 Q_left[face_offset + var] = isfinite(QL[var]) ? QL[var] : U_cells[cell_idx * 4 + var];
377 Q_right[face_offset + var] = isfinite(QR[var]) ? QR[var]
378 : ((idx_p1 >= 0 && idx_p1 < num_cells) ? U_cells[idx_p1 * 4 + var] : U_cells[cell_idx * 4 + var]);
379 }
380 }
