161__device__ double venkatakrishnan_limiter_device(
162 double delta_plus, // Forward difference
163 double delta_minus, // Backward difference
164 double delta_j, // Cell size measure
165 double K // Constant (typically 0.3-5.0)
166) {
167 if (fabs(delta_plus) < 1e-12) return 1.0;
168
169 double epsilon_sq = (K * delta_j) * (K * delta_j) * (K * delta_j);
170 double num = (delta_plus * delta_plus + epsilon_sq) * delta_minus +
171 2.0 * delta_minus * delta_minus * delta_plus;
172 double den = delta_plus * delta_plus + 2.0 * delta_minus * delta_minus +
173 delta_plus * delta_minus + epsilon_sq;
174
175 if (fabs(den) < 1e-12) return 1.0;
176
177 return num / den;
178}

184__device__ double barth_jespersen_limiter_device(
185 double u_cell, // Cell center value
186 double u_max, // Maximum value in neighborhood
187 double u_min, // Minimum value in neighborhood
188 double du_unlimited // Unlimited gradient * distance
189) {
190 if (fabs(du_unlimited) < 1e-12) return 1.0;
191
192 double phi = 1.0;
193
194 if (du_unlimited > 0.0) {
195 phi = fmin(1.0, (u_max - u_cell) / du_unlimited);
196 } else if (du_unlimited < 0.0) {
197 phi = fmin(1.0, (u_min - u_cell) / du_unlimited);
198 }
199
200 return fmax(0.0, phi);
201}

211__global__ void minmod_limiter_kernel(
212 const double* U_cells, // [num_cells * 4] - Conservative variables
213 const int* cell_neighbors, // [num_cells * 4] - Neighbor indices
214 const double* cell_distances, // [num_cells * 4] - Distances to neighbors
215 double* limited_gradients, // [num_cells * 4 * 4] - Output limited gradients
216 int num_cells,
217 double limiter_zeta, // Limiter parameter (typically 1.0-2.0)
218 int use_three_arg // 0=2-arg, 1=3-arg MinMod
219) {
220 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
221 if (cell_idx >= num_cells) return;
222
223 // Process each face (0=left, 1=bottom, 2=right, 3=top)
224 for (int face = 0; face < 4; face++) {
225 int neighbor_1 = cell_neighbors[cell_idx * 4 + face];
226
227 // Skip if no neighbor
228 if (neighbor_1 < 0) {
229 for (int var = 0; var < 4; var++) {
230 limited_gradients[(cell_idx * 4 + face) * 4 + var] = 0.0;
231 }
232 continue;
233 }
234
235 double d1 = cell_distances[cell_idx * 4 + face];
236
237 // Get second neighbor for 3-arg MinMod
238 int neighbor_2 = (face == 0 || face == 1) ?
239 cell_neighbors[cell_idx * 4 + face] :
240 cell_idx;
241 double d2 = d1;
242
243 if (use_three_arg && neighbor_1 < num_cells) {
244 neighbor_2 = cell_neighbors[neighbor_1 * 4 + face];
245 if (neighbor_2 >= 0 && neighbor_2 < num_cells) {
246 d2 = cell_distances[neighbor_1 * 4 + face];
247 }
248 }
249
250 // Process each conservative variable
251 for (int var = 0; var < 4; var++) {
252 double u_cell = U_cells[cell_idx * 4 + var];
253 double u_n1 = U_cells[neighbor_1 * 4 + var];
254
255 // Compute slopes
256 double slope1 = limiter_zeta * (u_cell - u_n1) / d1;
257
258 double phi;
259 if (use_three_arg && neighbor_2 >= 0 && neighbor_2 < num_cells) {
260 double u_n2 = U_cells[neighbor_2 * 4 + var];
261 double slope2 = limiter_zeta * (u_n1 - u_n2) / d2;
262 double slope3 = limiter_zeta * (u_cell - u_n2) / (d1 + d2);
263
264 phi = minmod_limiter_3arg_device(slope1, slope2, slope3);
265 } else {
266 // Use interior slope as second argument
267 double slope2 = limiter_zeta * (u_n1 - u_cell) / d1;
268 phi = minmod_limiter_2arg_device(slope1, slope2);
269 }
270
271 limited_gradients[(cell_idx * 4 + face) * 4 + var] = phi;
272 }
273 }
274}

443__global__ void barth_jespersen_limiter_kernel(
444 const double* U_cells,
445 const int* cell_neighbors,
446 const double* cell_distances,
447 const double* unlimited_gradients, // Pre-computed unlimited gradients
448 double* limited_gradients,
449 int num_cells
450) {
451 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
452 if (cell_idx >= num_cells) return;
453
454 // For each variable, find min/max in neighborhood
455 for (int var = 0; var < 4; var++) {
456 double u_cell = U_cells[cell_idx * 4 + var];
457 double u_max = u_cell;
458 double u_min = u_cell;
459
460 // Find min/max among neighbors
461 for (int face = 0; face < 4; face++) {
462 int neighbor = cell_neighbors[cell_idx * 4 + face];
463 if (neighbor >= 0 && neighbor < num_cells) {
464 double u_n = U_cells[neighbor * 4 + var];
465 u_max = fmax(u_max, u_n);
466 u_min = fmin(u_min, u_n);
467 }
468 }
469
470 // Compute limiter for each face
471 for (int face = 0; face < 4; face++) {
472 int neighbor = cell_neighbors[cell_idx * 4 + face];
473
474 if (neighbor < 0) {
475 limited_gradients[(cell_idx * 4 + face) * 4 + var] = 0.0;
476 continue;
477 }
478
479 double d1 = cell_distances[cell_idx * 4 + face];
480 double du_unlimited = unlimited_gradients[(cell_idx * 4 + face) * 4 + var] * d1;
481
482 double phi = barth_jespersen_limiter_device(u_cell, u_max, u_min, du_unlimited);
483
484 limited_gradients[(cell_idx * 4 + face) * 4 + var] = phi;
485 }
486 }

492__global__ void generic_limiter_kernel(
493 const double* U_cells,
494 const int* cell_neighbors,
495 const double* cell_distances,
496 double* limited_gradients,
497 int num_cells,
498 double limiter_zeta,
499 int limiter_type // 0=MinMod, 1=VanLeer, 2=Superbee, 3=VanAlbada
500) {
501 int cell_idx = blockIdx.x * blockDim.x + threadIdx.x;
502 if (cell_idx >= num_cells) return;
503
504 for (int face = 0; face < 4; face++) {
505 int neighbor = cell_neighbors[cell_idx * 4 + face];
506
507 if (neighbor < 0) {
508 for (int var = 0; var < 4; var++) {
509 limited_gradients[(cell_idx * 4 + face) * 4 + var] = 0.0;
510 }
511 continue;
512 }
513
514 double d1 = cell_distances[cell_idx * 4 + face];
515
516 for (int var = 0; var < 4; var++) {
517 double u_cell = U_cells[cell_idx * 4 + var];
518 double u_n = U_cells[neighbor * 4 + var];
519
520 double slope_forward = limiter_zeta * (u_cell - u_n) / d1;
521 double slope_backward = limiter_zeta * (u_n - u_cell) / d1;
522
523 double phi;
524 switch (limiter_type) {
525 case 0: // MinMod
526 phi = minmod_limiter_2arg_device(slope_forward, slope_backward);
527 break;
528 case 1: // Van Leer
529 phi = vanleer_limiter_device(slope_forward, slope_backward);
530 break;
531 case 2: // Superbee
532 phi = superbee_limiter_device(slope_forward, slope_backward);
533 break;
534 case 3: // Van Albada
535 phi = vanalbada_limiter_device(slope_forward, slope_backward);
536 break;
537 default:
538 phi = minmod_limiter_2arg_device(slope_forward, slope_backward);
539 }
540
541 limited_gradients[(cell_idx * 4 + face) * 4 + var] = phi;
542 }
