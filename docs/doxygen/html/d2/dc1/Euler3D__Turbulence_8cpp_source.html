15inline double strain_rate_mag(const double g[12])
16{
17 const double sxx = g[0], syy = g[4], szz = g[8];
18 const double sxy = 0.5 * (g[1] + g[3]);
19 const double sxz = 0.5 * (g[2] + g[6]);
20 const double syz = 0.5 * (g[5] + g[7]);
21 const double twoSijSij =
22 2.0 * (sxx * sxx + syy * syy + szz * szz) + 4.0 * (sxy * sxy + sxz * sxz + syz * syz);
23 return std::sqrt(std::max(twoSijSij, 0.0));
24}

26inline double wale_sd_mag(const double g[12])
27{
28 const double G[3][3] = {{g[0], g[1], g[2]}, {g[3], g[4], g[5]}, {g[6], g[7], g[8]}};
29 double G2[3][3] = {{0, 0, 0}, {0, 0, 0}, {0, 0, 0}};
30 for (int i = 0; i < 3; ++i)
31 for (int j = 0; j < 3; ++j)
32 for (int k = 0; k < 3; ++k)
33 G2[i][j] += G[i][k] * G[k][j];
34 const double tr = G2[0][0] + G2[1][1] + G2[2][2];
35 double Sd2 = 0.0;
36 for (int i = 0; i < 3; ++i)
37 for (int j = 0; j < 3; ++j)
38 {
39 double sd = 0.5 * (G2[i][j] + G2[j][i]);
40 if (i == j)
41 sd -= tr / 3.0;
42 Sd2 += sd * sd;
43 }
44 return std::sqrt(std::max(Sd2, 0.0));
45}

47inline double vreman_op(const double g[12], double delta)
48{
49 const double a[3][3] = {{g[0], g[3], g[6]}, {g[1], g[4], g[7]}, {g[2], g[5], g[8]}};
50 double alpha2 = 0.0;
51 double beta[3][3];
52 for (int i = 0; i < 3; ++i)
53 for (int j = 0; j < 3; ++j)
54 {
55 double s = 0.0;
56 for (int m = 0; m < 3; ++m)
57 {
58 s += a[m][i] * a[m][j];
59 alpha2 += a[i][j] * a[i][j];
60 }
61 beta[i][j] = delta * delta * s;
62 }
63 const double B = beta[0][0] * beta[1][1] - beta[0][1] * beta[0][1] +
64 beta[0][0] * beta[2][2] - beta[0][2] * beta[0][2] +
65 beta[1][1] * beta[2][2] - beta[1][2] * beta[1][2];
66 if (alpha2 < 1.0e-30 || B < 0.0)
67 return 0.0;
68 return std::sqrt(B / alpha2);
69}

73double mu_sgs_from_grad(int turb_model, double rho, const double g[12], double delta,
74 double y_wall, double mu_lam, const Config &cfg)
75{
76 if (turb_model <= 1)
77 return 0.0;
78 rho = std::max(rho, 1.0e-14);
79 const double nu_lam = mu_lam / rho;
80 const double S = strain_rate_mag(g);
81 delta = std::max(delta, 1.0e-14);
82 y_wall = std::max(y_wall, 0.0);
83
84 if (turb_model == 2)
85 {
86 const double u_tau = 0.04 * std::max(std::fabs(cfg.freestream_u), 1.0e-6);
87 const double yplus = y_wall * u_tau / std::max(nu_lam, 1.0e-14);
88 const double damp = 1.0 - std::exp(-yplus / std::max(cfg.A_plus, 1.0));
89 const double mix = cfg.kappa_von * y_wall * damp;
90 return rho * mix * mix * S;
91 }
92 if (turb_model == 3)
93 {
94 const double u_tau = 0.04 * std::max(std::fabs(cfg.freestream_u), 1.0e-6);
95 const double yplus = y_wall * u_tau / std::max(nu_lam, 1.0e-14);
96 const double fvd = 1.0 - std::exp(-yplus / std::max(cfg.A_plus, 1.0));
97 const double ls = cfg.Cs * delta * fvd;
98 return rho * ls * ls * S;
99 }
100 if (turb_model == 4)
101 {
102 const double Sd = wale_sd_mag(g);
103 const double num = std::pow(Sd, 1.5);
104 const double den = std::pow(S, 2.5) + std::pow(Sd, 1.25);
105 if (den < 1.0e-30)
106 return 0.0;
107 const double cw = cfg.Cw * delta;
108 return rho * cw * cw * num / std::max(den, 1.0e-30);
109 }
110 if (turb_model == 5)
111 return rho * cfg.Cv * vreman_op(g, delta);
112 return 0.0;
113}
