35__device__ double movers_alpha(const double d_u, const double d_f, const double lambda_max, const double lambda_min)
36{
37 const double eps = 1e-10;
38 if (fabs(d_f) < eps && fabs(d_u) > eps)
39 return 0.0;
40 if (fabs(d_f) < eps && fabs(d_u) < eps)
41 return lambda_min;
42 if (fabs(d_f) > eps && fabs(d_u) > eps)
43 {
44 double alpha = fabs(d_f / d_u);
45 if (alpha >= lambda_max)
