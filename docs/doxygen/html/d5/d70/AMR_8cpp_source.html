109Cell make_child_from_parent(const Cell &p, const int childType, const int parentIdx, const int childID)
110{
111 // childType: 0=SW, 1=SE, 2=NE, 3=NW
112 Cell ch = p;
113 ch.cellID = childID;
114 ch.AMR_Parent = parentIdx;
115 ch.AMR_Level = p.AMR_Level + 1;
116 ch.AMR_IsLeaf = true;
117 ch.AMR_Children.clear();
118 ch.ParentCellID = parentIdx;
119 ch.Is_Splittable = false;
120 ch.numFaces = 4;
121 ch.numNodes = 4;
122 ch.Neighbours.assign(4, -1);
123 ch.Secondary_Neighbours.assign(4, -1);
124 ch.faceID.assign(4, -1);
125 ch.Face_Boundary_Kind.assign(4, 0);
126 ch.Face_Neighbour_List.assign(4, vector<int>());
127 ch.Cell_Center_Distances.assign(4, 1.0);
128 ch.Face_Areas.assign(4, 1.0);
129 ch.Face_Normals.assign(8, 0.0);
130 ch.Area = p.Area * 0.25;
131 ch.Inv_Area = (ch.Area > 0.0) ? (1.0 / ch.Area) : 0.0;
132 ch.del_t = p.del_t;
133
134 // Map parent face normals: 0-left,1-bottom,2-right,3-top.
135 for (int f = 0; f < 4; ++f)
136 {
137 ch.Face_Normals[2 * f + 0] = p.Face_Normals[2 * f + 0];
138 ch.Face_Normals[2 * f + 1] = p.Face_Normals[2 * f + 1];
139 }
140
141 const double leftLen = (p.Face_Areas.size() > 0) ? p.Face_Areas[0] : 1.0;
142 const double botLen = (p.Face_Areas.size() > 1) ? p.Face_Areas[1] : 1.0;
143 const double rightLen = (p.Face_Areas.size() > 2) ? p.Face_Areas[2] : leftLen;
144 const double topLen = (p.Face_Areas.size() > 3) ? p.Face_Areas[3] : botLen;
145 const double vHalf = 0.25 * (leftLen + rightLen); // half of avg vertical span
146 const double hHalf = 0.25 * (botLen + topLen); // half of avg horizontal span
147 const double leftHalf = 0.5 * leftLen;
148 const double rightHalf = 0.5 * rightLen;
149 const double botHalf = 0.5 * botLen;
150 const double topHalf = 0.5 * topLen;
151
152 // Cell center offset using parent face normals (structured quad convention).
153 const double nxL = p.Face_Normals[0], nyL = p.Face_Normals[1];
154 const double nxB = p.Face_Normals[2], nyB = p.Face_Normals[3];
155 const double nxR = p.Face_Normals[4], nyR = p.Face_Normals[5];
156 const double nxT = p.Face_Normals[6], nyT = p.Face_Normals[7];
157 const double tx = 0.5 * (nxR - nxL);
158 const double ty = 0.5 * (nyR - nyL);
159 const double sx = 0.5 * (nxT - nxB);
160 const double sy = 0.5 * (nyT - nyB);
161 const double scale = 0.25 * sqrt(std::max(p.Area, 1e-12));
162 double ox = 0.0, oy = 0.0;
163 if (childType == 0) // SW
164 {
165 ox = -(tx + sx) * scale;
166 oy = -(ty + sy) * scale;
167 }
168 else if (childType == 1) // SE
169 {
170 ox = (tx - sx) * scale;
171 oy = (ty - sy) * scale;
172 }
173 else if (childType == 2) // NE
174 {
175 ox = (tx + sx) * scale;
176 oy = (ty + sy) * scale;
177 }
178 else // NW
179 {
180 ox = (-tx + sx) * scale;
181 oy = (-ty + sy) * scale;
182 }
183 if (ch.Cell_Center.size() < 2)
184 ch.Cell_Center.assign(2, 0.0);
185 ch.Cell_Center[0] = p.Cell_Center[0] + ox;
186 ch.Cell_Center[1] = p.Cell_Center[1] + oy;
187
188 // Initialize face lengths & boundary kinds from parent topology.
189 if (childType == 0) // SW
190 {
191 ch.Face_Areas[0] = leftHalf;
192 ch.Face_Areas[1] = botHalf;
193 ch.Face_Areas[2] = vHalf;
194 ch.Face_Areas[3] = hHalf;
195 ch.Face_Boundary_Kind[0] = p.Face_Boundary_Kind.size() > 0 ? p.Face_Boundary_Kind[0] : 0;
196 ch.Face_Boundary_Kind[1] = p.Face_Boundary_Kind.size() > 1 ? p.Face_Boundary_Kind[1] : 0;
197 }
198 else if (childType == 1) // SE
199 {
200 ch.Face_Areas[0] = vHalf;
201 ch.Face_Areas[1] = botHalf;
202 ch.Face_Areas[2] = rightHalf;
203 ch.Face_Areas[3] = hHalf;
204 ch.Face_Boundary_Kind[1] = p.Face_Boundary_Kind.size() > 1 ? p.Face_Boundary_Kind[1] : 0;
205 ch.Face_Boundary_Kind[2] = p.Face_Boundary_Kind.size() > 2 ? p.Face_Boundary_Kind[2] : 0;
206 }
207 else if (childType == 2) // NE
208 {
209 ch.Face_Areas[0] = vHalf;
210 ch.Face_Areas[1] = hHalf;
211 ch.Face_Areas[2] = rightHalf;
212 ch.Face_Areas[3] = topHalf;
213 ch.Face_Boundary_Kind[2] = p.Face_Boundary_Kind.size() > 2 ? p.Face_Boundary_Kind[2] : 0;
214 ch.Face_Boundary_Kind[3] = p.Face_Boundary_Kind.size() > 3 ? p.Face_Boundary_Kind[3] : 0;
215 }
216 else // NW
217 {
218 ch.Face_Areas[0] = leftHalf;
219 ch.Face_Areas[1] = hHalf;
220 ch.Face_Areas[2] = vHalf;
221 ch.Face_Areas[3] = topHalf;
222 ch.Face_Boundary_Kind[0] = p.Face_Boundary_Kind.size() > 0 ? p.Face_Boundary_Kind[0] : 0;
223 ch.Face_Boundary_Kind[3] = p.Face_Boundary_Kind.size() > 3 ? p.Face_Boundary_Kind[3] : 0;
224 }
225
226 return ch;
227}

231{

446bool AMR_Merge_Tagged_Cells()
447{
448 bool changed = false;
449 for (int p = 0; p < No_Physical_Cells; ++p)
450 {
451 if (!Cells[p].Is_Splittable)
452 continue;
453 if (Cells[p].AMR_IsLeaf || Cells[p].AMR_Children.size() != 4)
454 continue;
455
456 vector<int> kids = Cells[p].AMR_Children;
457 bool valid = true;
458 for (const int k : kids)
459 {
460 if (k < 0 || k >= No_Physical_Cells || !Cells[k].AMR_IsLeaf)
461 {
462 valid = false;
463 break;
464 }
465 }
466 if (!valid)
467 continue;
468
469 // Redirect external neighbors from children back to parent.
470 for (const int k : kids)
471 {
472 const auto &N = Cells[k].Neighbours;
473 for (int f = 0; f < static_cast<int>(N.size()); ++f)
474 {
475 const int neigh = N[f];
476 bool is_sibling = false;
477 for (const int s : kids)
478 if (neigh == s)
479 is_sibling = true;
480 if (!is_sibling && neigh >= 0 && neigh < No_Physical_Cells)
481 remap_neighbor_reference_all(neigh, kids, p);
482 }
483 }
484
485 // Restore coarse neighbors on parent from children boundary faces.
486 Cells[p].Neighbours[0] = pick_first_valid_neighbor(Cells[kids[0]].Neighbours[0], Cells[kids[3]].Neighbours[0]);
487 Cells[p].Neighbours[1] = pick_first_valid_neighbor(Cells[kids[0]].Neighbours[1], Cells[kids[1]].Neighbours[1]);
488 Cells[p].Neighbours[2] = pick_first_valid_neighbor(Cells[kids[1]].Neighbours[2], Cells[kids[2]].Neighbours[2]);
489 Cells[p].Neighbours[3] = pick_first_valid_neighbor(Cells[kids[3]].Neighbours[3], Cells[kids[2]].Neighbours[3]);
490
491 // Restrict child states to parent by simple average.
492 for (int c = 0; c < NUM_FLUX_COMPONENTS; ++c)
493 U_Cells[p][c] = 0.25 * (U_Cells[kids[0]][c] + U_Cells[kids[1]][c] + U_Cells[kids[2]][c] + U_Cells[kids[3]][c]);
494 Primitive_Cells[p] = Primitive_Cells[kids[0]];
495
496 Cells[p].AMR_IsLeaf = true;
497 Cells[p].Is_Splittable = false;
498 for (const int k : kids)
499 {
500 Cells[k].AMR_IsLeaf = false;
501 Cells[k].Is_Splittable = false;
502 }
503 changed = true;
504 }
505 return changed;
506}
